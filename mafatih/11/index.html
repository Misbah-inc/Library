مفاتیح الجنان جدید (نگارش آسان) · سوره های قرآن در مفاتیح الجنان ۱۴

سوره الواقعة

این بخش تنها سوره‌های برگزیدهٔ مفاتیح است. متن کامل قرآن کریم با ترجمه ←

بِسْمِ اللّه الرَّحْمَنِ الرَّحِيمِ

به نام خداوند بخشنده مهربان

إِذَا وَقَعَتِ الْوَاقِعَةُ﴿1﴾

آن گاه که آن واقعه (واقعه قیامت) واقع شود (1)

لَيْسَ لِوَقْعَتِهَا كَاذِبَةٌ ﴿2﴾

که در واقع شدن آن دروغی نیست، (2)

خَافِضَةٌ رَافِعَةٌ ﴿3﴾

پایین آورنده است و بالابرنده. (3)

إِذَا رُجَّتِ الْأَرْضُ رَجًّا ﴿4﴾

آن گاه که این زمین سخت لرزانده شود. (4)

وَبُسَّتِ الْجِبَالُ بَسًّا ﴿5﴾

و کوه ها به شدت سیر داده شوند و خرد گردند. (5)

فَكَانَتْ هَبَاءً مُنْبَثًّا ﴿6﴾

پس به حالت غبار پراکنده درآیند (آن گاه مومنان رستگار و کافران زیانکار گردند). (6)

وَكُنْتُمْ أَزْوَاجًا ثَلَاثَةً ﴿7﴾

و شما (مکلّفان) به سه گروه تقسیم شوید (7)

فَأَصْحَابُ الْمَيْمَنَةِ مَا أَصْحَابُ الْمَيْمَنَةِ ﴿8﴾

(نخست) اصحاب یمین چه هستند اصحاب یمین؟. (8)

وَأَصْحَابُ الْمَشْأَمَةِ مَا أَصْحَابُ الْمَشْأَمَةِ ﴿9﴾

و (دوم) اصحاب شمال چه هستند اصحاب شمال. (9)

وَالسَّابِقُونَ السَّابِقُونَ ﴿10﴾

و (سوم) پیشگامان (در ایمان) سبقت گیران (در تقوی). (10)

أُولَئِكَ الْمُقَرَّبُونَ ﴿11﴾

آنها مقربان (درگاه) اند (11)

فِي جَنَّاتِ النَّعِيمِ ﴿12﴾

در بهشت ها و باغ های پرنعمت خواهند بود. (12)

ثُلَّةٌ مِنَ الْأَوَّلِينَ ﴿13﴾

گروه انبوهی از پیشینیان (معصومین از گذشتگان) (13)

وَقَلِيلٌ مِنَ الْآخِرِينَ ﴿14﴾

و اندکی از پسینیان (معصومین امت اخیر)اند (14)

عَلَى سُرُرٍ مَوْضُونَةٍ ﴿15﴾

بر تخت هایی بافته شده (از طلا) در ردیف شده، (15)

مُتَّكِئِينَ عَلَيْهَا مُتَقَابِلِينَ ﴿16﴾

تکیه داده بر آنها رو به روی یکدیگر. (16)

يَطُوفُ عَلَيْهِمْ وِلْدَانٌ مُخَلَّدُونَ ﴿17﴾

همواره به گرد آنها پسرانی جاودانی (با دستبند و گوشواره دار) می گردند. (17)

بِأَكْوَابٍ وَ أَبَارِيقَ وَ كَأْسٍ مِنْ مَعِينٍ ﴿18﴾

با تنگ ها و ابریق ها و جام هایی از شرابی روان (18)

لَا يُصَدَّعُونَ عَنْهَا وَ لَا يُنْزِفُونَ ﴿19﴾

که از آن نه سردرد می گیرند و نه عقلشان زایل می گردد. (19)

وَفَاكِهَةٍ مِمَّا يَتَخَيَّرُونَ ﴿20﴾

و با میوه از آنچه انتخاب کنند. (20)

وَلَحْمِ طَيْرٍ مِمَّا يَشْتَهُونَ ﴿21﴾

و گوشت پرنده از هر چه میل نمایند. (21)

وَحُورٌ عِينٌ ﴿22﴾

و (برای آنها در آنجا) زنانی است سفیدپوست و درشت چشم و زیبا. (22)

كَأَمْثَالِ اللُّؤْلُؤِ الْمَكْنُونِ ﴿23﴾

همانند مرواریدی نهان در صدف. (23)

جَزَاءً بِمَا كَانُوا يَعْمَلُونَ ﴿24﴾

به پاداش آنچه (از عمل های شایسته) به جا می آوردند. (24)

لَا يَسْمَعُونَ فِيهَا لَغْوًا

در آن بهشت ها هرگز سخنی لغو

ص ۱۵

وَلَا تَأْثِيمًا ﴿25﴾

و گناه آور نمی شنوند. (25)

إِلَّا قِيلًا سَلَامًا سَلَامًا ﴿26﴾

مگر سخن سلام (علیک) سلام (علیک). (26)

وَأَصْحَابُ الْيَمِينِ مَا أَصْحَابُ الْيَمِينِ ﴿27﴾

و یاران دست راست، چه هستند یاران دست راست؟ (27)

فِي سِدْرٍ مَخْضُودٍ ﴿28﴾

در زیر درختان سدری بی خار (28)

وَطَلْحٍ مَنْضُودٍ ﴿29﴾

و موزی که میوه هایش روی هم نشسته (29)

وَظِلٍّ مَمْدُودٍ ﴿30﴾

و سایه ای کشیده شده دائم (30)

وَمَاءٍ مَسْكُوبٍ ﴿31﴾

و آبی ریزان (از آبشارها) (31)

وَفَاكِهَةٍ كَثِيرَةٍ ﴿32﴾

و میوه هایی فراوان (32)

لَا مَقْطُوعَةٍ وَ لَا مَمْنُوعَةٍ ﴿33﴾

که نه تمام شدنی است و نه ممنوع از استفاده است (33)

وَفُرُشٍ مَرْفُوعَةٍ ﴿34﴾

و فرش هایی عالی و روی هم گسترده، یا همسرانی والا قدر در اخلاق و جمال (34)

إِنَّا أَنْشَأْنَاهُنَّ إِنْشَاءً ﴿35﴾

که البته ما آنها را به آفرینش خاصی آفریدیم (35)

فَجَعَلْنَاهُنَّ أَبْكَارًا ﴿36﴾

پس آنها را (برای همیشه) دوشیزه قرار داده ایم (36)

عُرُبًا أَتْرَابًا ﴿37﴾

و دوستان و عشق ورزان به همسر، (یا همسالان با یکدیگر یا با همسرانشان) (37)

لِأَصْحَابِ الْيَمِينِ ﴿38﴾

(اینها همه) برای یاران دست راست است (38)

ثُلَّةٌ مِنَ الْأَوَّلِينَ ﴿39﴾

که گروهی انبوه از پیشینیان (از امت های گذشته) (39)

وَثُلَّةٌ مِنَ الْآخِرِينَ ﴿40﴾

و گروه انبوهی از متأخران (از آخرین امت) اند (40)

وَأَصْحَابُ الشِّمَالِ مَا أَصْحَابُ الشِّمَالِ ﴿41﴾

و یاران دست چپ، چه هستند یاران دست چپ؟ (41)

فِي سَمُومٍ وَ حَمِيمٍ ﴿42﴾

در (میان) حرارت نفوذکننده آتش و آب جوشان (42)

وَظِلٍّ مِنْ يَحْمُومٍ ﴿43﴾

و سایه ای از دود غلیظ سیاه اند (43)

لَا بَارِدٍ وَ لَا كَرِيمٍ ﴿44﴾

که نه خنک است و نه سودبخش (برای استراحت) (44)

إِنَّهُمْ كَانُوا قَبْلَ ذَلِكَ مُتْرَفِينَ ﴿45﴾

حقّا که آنها پیش از این (در دنیا) از صاحب نعمت های خوشگذران بودند (45)

وَكَانُوا يُصِرُّونَ عَلَى الْحِنْثِ الْعَظِيمِ ﴿46﴾

و همواره بر تخلّفات بزرگ اصرار می ورزیدند (46)

وَكَانُوا يَقُولُونَ أَئِذَا مِتْنَا وَ كُنَّا تُرَابًا وَ عِظَامًا أَإِنَّا لَمَبْعُوثُونَ ﴿47﴾

و می گفتند: آیا هنگامی که ما مردیم و خاک و پاره های استخوان شدیم آیا حتما ما برانگیخته می شویم؟ (47)

أَوَآبَاؤُنَا الْأَوَّلُونَ ﴿48﴾

و آیا نیاکان ما (نیز برانگیخته می شوند) (48)

قُلْ إِنَّ الْأَوَّلِينَ وَ الْآخِرِينَ ﴿49﴾

بگو: بی تردید اولین و آخرین (49)

لَمَجْمُوعُونَ إِلَى مِيقَاتِ يَوْمٍ مَعْلُومٍ ﴿50﴾

حتما به سوی وعده گاه روز معینی (روز قیامت) گردآوری خواهند شد (50)

ثُمَّ إِنَّكُمْ أَيُّهَا الضَّالُّونَ الْمُكَذِّبُونَ ﴿51﴾

سپس شما ای گمراهان انکارکننده (رسالت و قیامت) (51)

لَآكِلُونَ مِنْ شَجَرٍ مِنْ زَقُّومٍ ﴿52﴾

حتما از درختی (بد منظر و بد طعم و بو) که زقّوم است خواهید خورد (52)

فَمَالِئُونَ مِنْهَا الْبُطُونَ ﴿53﴾

پس شکم ها را از آن پر خواهید نمود (53)

فَشَارِبُونَ عَلَيْهِ مِنَ الْحَمِيمِ ﴿54﴾

پس روی آن آب جوشان خواهید نوشید (54)

ص ۱۶

فَشَارِبُونَ شُرْبَ الْهِيمِ ﴿55﴾

پس همانند شتران عطش زده می نوشید (55)

هَذَا نُزُلُهُمْ يَوْمَ الدِّينِ ﴿56﴾

این در روز جزا پذیرایی ابتدایی آنهاست (56)

نَحْنُ خَلَقْنَاكُمْ فَلَوْلَا تُصَدِّقُونَ ﴿57﴾

ما شما را آفریدیم پس چرا (تجدید خلقتان در آخرت را) تصدیق نمی کنید؟ (57)

أَفَرَأَيْتُمْ مَا تُمْنُونَ ﴿58﴾

آیا آنچه از نطفه (در رحم) می ریزید دیده اید (58)

أَأَنْتُمْ تَخْلُقُونَهُ أم نَحْنُ الْخَالِقُونَ ﴿59﴾

آیا شما آن را می آفرینید یا ما آفریننده ایم (59)

نَحْنُ قَدَّرْنَا بَيْنَكُمُ الْمَوْتَ وَ مَا نَحْنُ بِمَسْبُوقِينَ ﴿60﴾

ماییم که مرگ را در میان شما مقدّر کردیم- و هرگز ما در اراده مغلوب نمی شویم (60)

عَلَى أَنْ نُبَدِّلَ أَمْثَالَكُمْ وَ نُنْشِئَكُمْ فِي مَا لَا تَعْلَمُونَ ﴿61﴾

تا (در دنیا) امثالتان را به جای شما قرار دهیم و شما را در صورت ها و حالاتی که نمی دانید مجددا ایجاد نماییم (61)

وَلَقَدْ عَلِمْتُمُ النَّشْأَةَ الْأُولَى فَلَوْلَا تَذَكَّرُونَ ﴿62﴾

و به یقین شما خلقت نخستین (در این دنیا) را دانستید پس چرا متذکّر نمی شوید؟ (62)

أَفَرَأَيْتُمْ مَا تَحْرُثُونَ ﴿63﴾

آیا آنچه را کشت می کنید دیده اید (63)

أَأَنْتُمْ تَزْرَعُونَهُ أم نَحْنُ الزَّارِعُونَ ﴿64﴾

آیا شما آن را می رویانید یا ما رویاننده ایم (64)

لَوْ نَشَاءُ لَجَعَلْنَاهُ حُطَامًا فَظَلْتُمْ تَفَكَّهُونَ ﴿65﴾

اگر اراده حتمی کنیم مسلّما آن را خار و خاشاک می گردانیم، پس شما از روی تعجب و تأسف گویید (65)

إِنَّا لَمُغْرَمُونَ ﴿66﴾

حقّا که ما غرامت زده ایم (66)

بَلْ نَحْنُ مَحْرُومُونَ ﴿67﴾

بلکه ما محروم و بدبختیم. (67)

أَفَرَأَيْتُمُ الْمَاءَ الَّذِي تَشْرَبُونَ ﴿68﴾

آیا آبی را که می نوشید دیده اید. (68)

أَأَنْتُمْ أَنْزَلْتُمُوهُ مِنَ الْمُزْنِ أم نَحْنُ الْمُنْزِلُونَ ﴿69﴾

آیا شما آن را از ابر سفید باران زا فرود آوردید یا ما فرو فرستنده ایم (69)

لَوْ نَشَاءُ جَعَلْنَاهُ أُجَاجًا فَلَوْلَا تَشْكُرُونَ ﴿70﴾

اگر بخواهیم آن را تلخ می گردانیم، پس چرا سپاس نمی گزارید؟ (70)

أَفَرَأَيْتُمُ النَّارَ الَّتِي تُورُونَ ﴿71﴾

آیا آن آتشی را که برمی افروزید دیده اید. (71)

أَأَنْتُمْ أَنْشَأْتُمْ شَجَرَتَهَا أم نَحْنُ الْمُنْشِئُونَ ﴿72﴾

آیا شما درخت آن را ایجاد کرده اید یا ما ایجادکننده ایم (72)

نَحْنُ جَعَلْنَاهَا تَذْكِرَةً وَ مَتَاعًا لِلْمُقْوِينَ ﴿73﴾

ما آن را وسیله تذکر (آتش قیامت) و برخورداری برای مسافران قرار داده ایم. (73)

فَسَبِّحْ بِاسْمِ رَبِّكَ الْعَظِيمِ ﴿74﴾

پس به نام پروردگار بزرگت تسبیح گوی. (74)

فَلَا أُقْسِمُ بِمَوَاقِعِ النُّجُومِ ﴿75﴾

پس سوگند به جایگاه ستارگان (در جوّ وسیع) (75)

وَإِنَّهُ لَقَسَمٌ لَوْ تَعْلَمُونَ عَظِيمٌ ﴿76﴾

و البته اگر بدانید این سوگندی است بزرگ. (76)

ص ۱۷

إِنَّهُ لَقُرْآنٌ كَرِيمٌ ﴿77﴾

که به یقین این قرآنی است کریم (77)

فِي كِتَابٍ مَكْنُونٍ ﴿78﴾

که در نوشته ای مستور است (78)

لَا يَمَسُّهُ إِلَّا الْمُطَهَّرُونَ ﴿79﴾

که آن (لوح نوشته) را لمس نکند جز فرشتگان پاکیزه شدگان. (79)

تَنْزِيلٌ مِنْ رَبِّ الْعَالَمِينَ ﴿80﴾

فرو فرستاده شده از جانب پروردگار جهانیان است. (80)

أَفَبِهَذَا الْحَدِيثِ أَنْتُمْ مُدْهِنُونَ ﴿81﴾

آیا با این سخن به سستی و سبکی برخورد می کنید؟ (81)

وَتَجْعَلُونَ رِزْقَكُمْ أَنَّكُمْ تُكَذِّبُونَ ﴿82﴾

و روزی تان را (از آن) این قرار می دهید که آن را تکذیب می نمایید (82)

فَلَوْلَا إِذَا بَلَغَتِ الْحُلْقُومَ ﴿83﴾

پس چرا آن گاه که جان به گلوگاه می رسد (83)

وَأَنْتُمْ حِينَئِذٍ تَنْظُرُونَ ﴿84﴾

و شما در آن حال (به وضع محتضر) می نگرید (84)

وَنَحْنُ أَقْرَبُ إِلَيْهِ مِنْكُمْ وَ لَكِنْ لَا تُبْصِرُونَ ﴿85﴾

و ما به آن محتضر از شما نزدیکتریم و لکن نمی بینید. (85)

فَلَوْلَا إِنْ كُنْتُمْ غَيْرَ مَدِينِينَ ﴿86﴾

(آری) پس چرا اگر شما مجازات شدنی نیستید (86)

تَرْجِعُونَهَا إِنْ كُنْتُمْ صَادِقِينَ ﴿87﴾

آن جان را (به بدن او) بازنمی گردانید اگر راستگویید؟ (87)

فَأَمَّا إِنْ كَانَ مِنَ الْمُقَرَّبِينَ ﴿88﴾

پس اگر (آن محتضر) از مقربان باشد (88)

فَرَوْحٌ وَ رَيْحَانٌ وَ جَنَّتُ نَعِيمٍ ﴿89﴾

پس (او را) آسایش و روزی پاک و بهشت پرنعمت است (89)

وَأَمَّا إِنْ كَانَ مِنْ أَصْحَابِ الْيَمِينِ ﴿90﴾

و اما اگر از یاران دست راست باشد (90)

فَسَلَامٌ لَكَ مِنْ أَصْحَابِ الْيَمِينِ ﴿91﴾

پس (به او گفته می شود) سلامی است تو را از جانب یاران دست راست. (91)

وَأَمَّا إِنْ كَانَ مِنَ الْمُكَذِّبِينَ الضَّالِّينَ ﴿92﴾

و اما اگر از تکذیب کنندگان گمراه باشد (92)

فَنُزُلٌ مِنْ حَمِيمٍ ﴿93﴾

پس پذیرایی ابتدایی از آب جوشان است. (93)

وَتَصْلِيَةُ جَحِيمٍ ﴿94﴾

و وارد شدن در جهنم و چشیدن حرارت آن. (94)

إِنَّ هَذَا لَهُوَ حَقُّ الْيَقِينِ ﴿95﴾

به یقین این (مطالب) خود حقّی است یقینی و غیر قابل شک. (95)

فَسَبِّحْ بِاسْمِ رَبِّكَ الْعَظِيمِ ﴿96﴾

پس به نام پروردگار بزرگ خود تسبیح گو. (96)
