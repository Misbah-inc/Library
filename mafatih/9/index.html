سوره الرحمن

این بخش تنها سوره‌های برگزیدهٔ مفاتیح است. متن کامل قرآن کریم با ترجمه ←

بِسْمِ اللّه الرَّحْمَنِ الرَّحِيمِ

به نام خداوند بخشنده مهربان

الرَّحْمَنُ﴿1﴾

خداوند رحمان،(1)

عَلَّمَ الْقُرْآنَ ﴿2﴾

قرآن را یاد داد. (2)

خَلَقَ الْإِنْسَانَ ﴿3﴾

انسان را بیافرید (3)

عَلَّمَهُ الْبَيَانَ ﴿4﴾

او را راه بیان آموخت. (4)

الشَّمْسُ وَ الْقَمَرُ بِحُسْبَانٍ ﴿5﴾

ورشید و ماه با حسابی معین در جریانند. (5)

وَالنَّجْمُ وَ الشَّجَرُ يَسْجُدَانِ ﴿6﴾

و گیاه و درخت (در برابر خدا تکوینا) سجده می کنند. (6)

وَالسَّمَاءَ رَفَعَهَا وَ وَضَعَ الْمِيزَانَ ﴿7﴾

و آسمان را برافراشت، و قانون و میزان را بنهاد. (7)

أَلَّا تَطْغَوْا فِي الْمِيزَانِ ﴿8﴾

(و امر نمود) که در سنجش طغیان نکنید. (8)

وَأَقِيمُوا الْوَزْنَ بِالْقِسْطِ وَ لَا تُخْسِرُوا الْمِيزَانَ ﴿9﴾

و وزن را به عدالت برپا دارید و ترازو را کم مگذارید. (9)

وَالْأَرْضَ وَضَعَهَا لِلْأَنَامِ ﴿10﴾

و این زمین را برای همه خلق بیافرید. (10)

ص ۱۰

فِيهَا فَاكِهَةٌ وَ النَّخْلُ ذَاتُ الْأَكْمَامِ ﴿11﴾

در آن هر نوع میوه، و درخت خرماست که دارای خوشه های غلاف دار است. (11)

وَالْحَبُّ ذُو الْعَصْفِ وَ الرَّيْحَانُ ﴿12﴾

نیز دانه با ساق و برگ و گیاهان خوشبو (12)

فَبِأَيِّ آلَاءِ رَبِّكُمَا تُكَذِّبَانِ ﴿13﴾

پس (ای گروه انس و جن) کدام یک از نعمت های پروردگارتان را تکذیب و انکار می کنید؟ (13)

خَلَقَ الْإِنْسَانَ مِنْ صَلْصَالٍ كَالْفَخَّارِ ﴿14﴾

انسان را از گل خشکیده ای مانند سفال آفرید. (14)

وَخَلَقَ الْجَانَّ مِنْ مَارِجٍ مِنْ نَارٍ ﴿15﴾

و جن را از شعله ای سفید و سرخ و سیاه از آتش آفرید. (15)

فَبِأَيِّ آلَاءِ رَبِّكُمَا تُكَذِّبَانِ ﴿16﴾

پس کدام یک از نعمت های پروردگارتان را تکذیب و انکار می کنید؟ (16)

رَبُّ الْمَشْرِقَيْنِ وَ رَبُّ الْمَغْرِبَيْنِ ﴿17﴾

(او) پروردگار دو مشرق و پروردگار دو مغرب است. (17)

فَبِأَيِّ آلَاءِ رَبِّكُمَا تُكَذِّبَانِ ﴿18﴾

پس کدام یک از نعمت های پروردگارتان را تکذیب و انکار می کنید؟ (18)

مَرَجَ الْبَحْرَيْنِ يَلْتَقِيَانِ ﴿19﴾

دو دریا را که با هم برخورد دارند به هم درآمیخت. (19)

بَيْنَهُمَا بَرْزَخٌ لَا يَبْغِيَانِ ﴿20﴾

میان آن دو حد فاصلی است که بر یکدیگر (از حیث اثر) غلبه نمی کنند (20)

فَبِأَيِّ آلَاءِ رَبِّكُمَا تُكَذِّبَانِ ﴿21﴾

پس کدام یک از نعمت های پروردگارتان را تکذیب و انکار می کنید؟ (21)

يَخْرُجُ مِنْهُمَا اللُّؤْلُؤُ وَ الْمَرْجَانُ ﴿22﴾

از آن دو دریا (ی شور و شیرین) مروارید و مرجان بیرون آید. (22)

فَبِأَيِّ آلَاءِ رَبِّكُمَا تُكَذِّبَانِ ﴿23﴾

پس کدام یک از نعمت های پروردگارتان را تکذیب و انکار می کنید؟ (23)

وَلَهُ الْجَوَارِ الْمُنْشَآتُ فِي الْبَحْرِ كَالْأَعْلَامِ ﴿24﴾

و او راست کشتی های روان در دریا که همانند کوه های بلندی ساخته شده اند. (24)

فَبِأَيِّ آلَاءِ رَبِّكُمَا تُكَذِّبَانِ ﴿25﴾

پس کدام یک از نعمت های پروردگارتان را تکذیب و انکار می کنید؟ (25)

كُلُّ مَنْ عَلَيْهَا فَانٍ ﴿26﴾

هر که بر روی این (زمین) است محکوم به فناست. (26)

وَيَبْقَى وَجْهُ رَبِّكَ ذُو الْجَلَالِ وَ الْإِكْرَامِ ﴿27﴾

و تنها ذات صاحب کبریا و عظمت و برتری و دارای جود و جمال و رحمت پروردگارت باقی می ماند. (27)

فَبِأَيِّ آلَاءِ رَبِّكُمَا تُكَذِّبَانِ ﴿28﴾

پس کدام یک از نعمت های پروردگارتان را تکذیب و انکار می کنید؟ (28)

يَسْأَلُهُ مَنْ فِي السَّمَاوَاتِ وَ الْأَرْضِ ۚ كُلَّ يَوْمٍ هُوَ فِي شَأْنٍ ﴿29﴾

هر که در آسمان ها و زمین است از او درخواست می کند او هر روز و هر زمان در کاری است. (29)

فَبِأَيِّ آلَاءِ رَبِّكُمَا تُكَذِّبَانِ ﴿30﴾

پس کدام یک از نعمت های پروردگارتان را تکذیب و انکار می کنید؟ (30)

سَنَفْرُغُ لَكُمْ أَيُّهَ الثَّقَلَانِ ﴿31﴾

به زودی به (حساب) شما می پردازیم ای دو گروه انس و جن (31)

فَبِأَيِّ آلَاءِ رَبِّكُمَا تُكَذِّبَانِ ﴿32﴾

پس کدام یک از نعمت های پروردگارتان را تکذیب و انکار می کنید؟ (32)

يَا مَعْشَرَ الْجِنِّ وَ الْإِنْسِ

ای گروه جن و انس،

ص ۱۱

إِنِ اسْتَطَعْتُمْ أَنْ تَنْفُذُوا مِنْ أَقْطَارِ السَّمَاوَاتِ وَ الْأَرْضِ فَانْفُذُوا ۚ لَا تَنْفُذُونَ إِلَّا بِسُلْطَانٍ ﴿33﴾

اگر توانستید از کرانه های آسمان ها و زمین (به بیرون) بگذرید (تا از کیفر اعمالتان فرار کنید) بگذرید، (اما) نخواهید گذشت جز با قدرتی (که آن را ندارید). (33)

فَبِأَيِّ آلَاءِ رَبِّكُمَا تُكَذِّبَانِ ﴿34﴾

پس کدام یک از نعمت های پروردگارتان را تکذیب و انکار می کنید؟ (34)

يُرْسَلُ عَلَيْكُمَا شُوَاظٌ مِنْ نَارٍ وَ نُحَاسٌ فَلَا تَنْتَصِرَانِ ﴿35﴾

بر (سر) شما شعله ای از آتش و دودی غلیظ، یا پاره های مس گداخته، فرو فرستاده می شود، پس هرگز از خود دفاع نتوانید کرد (35)

فَبِأَيِّ آلَاءِ رَبِّكُمَا تُكَذِّبَانِ ﴿36﴾

پس کدام یک از نعمت های پروردگارتان را تکذیب و انکار می کنید؟ (36)

فَإِذَا انْشَقَّتِ السَّمَاءُ فَكَانَتْ وَرْدَةً كَالدِّهَانِ ﴿37﴾

پس آن گاه که آسمان بشکافد و همانند چرم سرخ (یا روغن مذاب) گلگون گردد. (37)

فَبِأَيِّ آلَاءِ رَبِّكُمَا تُكَذِّبَانِ ﴿38﴾

پس کدام یک از نعمت های پروردگارتان را تکذیب و انکار می کنید؟ (38)

فَيَوْمَئِذٍ لَا يُسْأَلُ عَنْ ذَنْبِهِ إِنْسٌ وَ لَا جَانٌّ ﴿39﴾

پس در آن روز هیچ انس و جنّی از گناهش پرسش نمی شود. (39)

فَبِأَيِّ آلَاءِ رَبِّكُمَا تُكَذِّبَانِ ﴿40﴾

پس کدام یک از نعمت های پروردگارتان را تکذیب و انکار می کنید؟ (40)

يُعْرَفُ الْمُجْرِمُونَ بِسِيمَاهُمْ فَيُؤْخَذُ بِالنَّوَاصِي وَ الْأَقْدَامِ ﴿41﴾

(زیرا) گنهکاران به سیمایشان شناخته می شوند، پس آنها را از موهای جلو سر و قدم ها می گیرند (و به دوزخ می افکنند) (41)

فَبِأَيِّ آلَاءِ رَبِّكُمَا تُكَذِّبَانِ ﴿42﴾

پس کدام یک از نعمت های پروردگارتان را تکذیب و انکار می کنید؟ (42)

هَذِهِ جَهَنَّمُ الَّتِي يُكَذِّبُ بِهَا الْمُجْرِمُونَ ﴿43﴾

این همان جهنمی است که گنهکاران آن را تکذیب و انکار می کردند. (43)

يَطُوفُونَ بَيْنَهَا وَ بَيْنَ حَمِيمٍ آنٍ ﴿44﴾

آنها در میان آن آتش و آب جوشان دور می زنند (44)

فَبِأَيِّ آلَاءِ رَبِّكُمَا تُكَذِّبَانِ ﴿45﴾

پس کدام یک از نعمت های پروردگارتان را تکذیب و انکار می کنید؟ (45)

وَلِمَنْ خَافَ مَقَامَ رَبِّهِ جَنَّتَانِ ﴿46﴾

و برای کسی که از تسلط و احاطه پروردگارش، یا از قیام خود در حضور پروردگارش بترسد دو بهشت است. (46)

فَبِأَيِّ آلَاءِ رَبِّكُمَا تُكَذِّبَانِ ﴿47﴾

پس کدام یک از نعمت های پروردگارتان را تکذیب و انکار می کنید؟ (47)

ذَوَاتَا أَفْنَانٍ ﴿48﴾

که هر دو دارای انواع درختان و شاخساران است. (48)

فَبِأَيِّ آلَاءِ رَبِّكُمَا تُكَذِّبَانِ ﴿49﴾

پس کدام یک از نعمت های پروردگارتان را تکذیب و انکار می کنید؟ (49)

فِيهِمَا عَيْنَانِ تَجْرِيَانِ ﴿50﴾

در آن دو (بهشت) دو چشمه پیوسته در جریان است. (50)

فَبِأَيِّ آلَاءِ رَبِّكُمَا تُكَذِّبَانِ ﴿51﴾

پس کدام یک از نعمت های پروردگارتان را تکذیب و انکار می کنید؟ (51)

فِيهِمَا مِنْ كُلِّ فَاكِهَةٍ زَوْجَانِ ﴿52﴾

در آن دو از هر میوه ای دو صنف است. (52)

فَبِأَيِّ آلَاءِ رَبِّكُمَا تُكَذِّبَانِ ﴿53﴾

پس کدام یک از نعمت های پروردگارتان را تکذیب و انکار می کنید؟ (53)

مُتَّكِئِينَ

ص ۱۲

عَلَى فُرُشٍ بَطَائِنُهَا مِنْ إِسْتَبْرَقٍ ۚ وَ جَنَى الْجَنَّتَيْنِ دَانٍ ﴿54﴾

در حالی که بر روی بسترهایی که آستر داخل آنها از ابریشم ستبر است تکیه زده اند، و میوه هر دو باغ نزدیک و در دسترس است. (54)

فَبِأَيِّ آلَاءِ رَبِّكُمَا تُكَذِّبَانِ ﴿55﴾

پس کدام یک از نعمت های پروردگارتان را تکذیب و انکار می کنید؟ (55)

فِيهِنَّ قَاصِرَاتُ الطَّرْفِ لَمْ يَطْمِثْهُنَّ إِنْسٌ قَبْلَهُمْ وَ لَا جَانٌّ ﴿56﴾

در آن بهشت ها زنانی هستند که چشم از دیدن غیر همسر کوته بینان دارند و جز او به کسی عشق نورزند، نه انسانی پیش از آنها با آنان تماس گرفته و نه جنّی. (56)

فَبِأَيِّ آلَاءِ رَبِّكُمَا تُكَذِّبَانِ ﴿57﴾

پس کدام یک از نعمت های پروردگارتان را تکذیب و انکار می کنید؟ (57)

كَأَنَّهُنَّ الْيَاقُوتُ وَ الْمَرْجَانُ ﴿58﴾

گویی یاقوت و مرجانند. (58)

فَبِأَيِّ آلَاءِ رَبِّكُمَا تُكَذِّبَانِ ﴿59﴾

پس کدام یک از نعمت های پروردگارتان را تکذیب و انکار می کنید؟ (59)

هَلْ جَزَاءُ الْإِحْسَانِ إِلَّا الْإِحْسَانُ ﴿60﴾

آیا پاداش نیکی جز نیکی است؟ (60)

فَبِأَيِّ آلَاءِ رَبِّكُمَا تُكَذِّبَانِ ﴿61﴾

پس کدام یک از نعمت های پروردگارتان را تکذیب و انکار می کنید؟ (61)

وَمِنْ دُونِهِمَا جَنَّتَانِ ﴿62﴾

و فروتر از آن دو بهشت دیگر است. (62)

فَبِأَيِّ آلَاءِ رَبِّكُمَا تُكَذِّبَانِ ﴿63﴾

پس کدام یک از نعمت های پروردگارتان را تکذیب و انکار می کنید؟ (63)

مُدْهَامَّتَانِ ﴿64﴾

که هر دو از شدت سرسبزی به سیاهی می زنند. (64)

فَبِأَيِّ آلَاءِ رَبِّكُمَا تُكَذِّبَانِ ﴿65﴾

پس کدام یک از نعمت های پروردگارتان را تکذیب و انکار می کنید؟ (65)

فِيهِمَا عَيْنَانِ نَضَّاخَتَانِ ﴿66﴾

در آن دو، دو چشمه جوشان و در فوران است. (66)

فَبِأَيِّ آلَاءِ رَبِّكُمَا تُكَذِّبَانِ ﴿67﴾

پس کدام یک از نعمت های پروردگارتان را تکذیب و انکار می کنید؟ (67)

فِيهِمَا فَاكِهَةٌ وَ نَخْلٌ وَ رُمَّانٌ ﴿68﴾

در آن دو میوه و درخت خرما و انار است. (68)

فَبِأَيِّ آلَاءِ رَبِّكُمَا تُكَذِّبَانِ ﴿69﴾

پس کدام یک از نعمت های پروردگارتان را تکذیب و انکار می کنید؟ (69)

فِيهِنَّ خَيْرَاتٌ حِسَانٌ ﴿70﴾

در آنها زنانی نیکو سیرت و زیبا صورت است. (70)

فَبِأَيِّ آلَاءِ رَبِّكُمَا تُكَذِّبَانِ ﴿71﴾

پس کدام یک از نعمت های پروردگارتان را تکذیب و انکار می کنید؟ (71)

حُورٌ مَقْصُورَاتٌ فِي الْخِيَامِ ﴿72﴾

حوریانی پرده نشین در میان خیمه ها. (72)

فَبِأَيِّ آلَاءِ رَبِّكُمَا تُكَذِّبَانِ ﴿73﴾

پس کدام یک از نعمت های پروردگارتان را تکذیب و انکار می کنید؟ (73)

لَمْ يَطْمِثْهُنَّ إِنْسٌ قَبْلَهُمْ وَ لَا جَانٌّ ﴿74﴾

که نه انسانی پیش از آنها با آنان تماس گرفته و نه جنّی. (74)

فَبِأَيِّ آلَاءِ رَبِّكُمَا تُكَذِّبَانِ ﴿75﴾

پس کدام یک از نعمت های پروردگارتان را تکذیب و انکار می کنید؟ (75)

مُتَّكِئِينَ عَلَى رَفْرَفٍ خُضْرٍ وَ عَبْقَرِيٍّ حِسَانٍ ﴿76﴾

در حالی که بر بالش های سبز و بسترها و فرش های زیبا تکیه زده اند. (76)

فَبِأَيِّ آلَاءِ رَبِّكُمَا تُكَذِّبَانِ ﴿77﴾

پس کدام یک از نعمت های پروردگارتان را تکذیب و انکار می کنید؟ (77)

تَبَارَكَ اسْمُ رَبِّكَ ذِي الْجَلَالِ وَ الْإِكْرَامِ ﴿78﴾

پرخیر و برکت است نام پروردگارت که صاحب کبریا و عظمت و برتری و جود و جمال و رحمت است. (78)
