٢٢١

بِسْمِ ٱللَّهِ ٱلرَّحْمَٰنِ ٱلرَّحِيمِ

الٓر ۚ كِتَٰبٌ أُحْكِمَتْ ءَايَٰتُهُۥ ثُمَّ فُصِّلَتْ مِن لَّدُنْ حَكِيمٍ خَبِيرٍ۝١

1الۤر - یہ وہ کتاب ہے جس کی آیتیں محکم بنائی گئی ہیں اور ایک صاحبِ علم و حکمت کی طرف سے تفصیل کے ساتھ بیان کی گئی ہیں

أَلَّا تَعْبُدُوٓا۟ إِلَّا ٱللَّهَ ۚ إِنَّنِى لَكُم مِّنْهُ نَذِيرٌ وَبَشِيرٌ۝٢

2کہ اللہ کے علاوہ کسی کی عبادت نہ کرو -میں اسی کی طرف سے ڈرانے والا اور بشارت دینے والا ہوں

وَأَنِ ٱسْتَغْفِرُوا۟ رَبَّكُمْ ثُمَّ تُوبُوٓا۟ إِلَيْهِ يُمَتِّعْكُم مَّتَٰعًا حَسَنًا إِلَىٰٓ أَجَلٍ مُّسَمًّى وَيُؤْتِ كُلَّ ذِى فَضْلٍ فَضْلَهُۥ ۖ وَإِن تَوَلَّوْا۟ فَإِنِّىٓ أَخَافُ عَلَيْكُمْ عَذَابَ يَوْمٍ كَبِيرٍ۝٣

3اور اپنے رب سے استغفار کرو پھر اس کی طرف متوجہ ہوجاؤ وہ تم کو مقررہ مدّت میں بہترین فائدہ عطا کرے گا اور صاحبِ فضل کو اس کے فضل کا حق دے گا اور میں تمہارے بارے میں ایک بڑے دن کے عذاب سے خوفزدہ ہوں

إِلَى ٱللَّهِ مَرْجِعُكُمْ ۖ وَهُوَ عَلَىٰ كُلِّ شَىْءٍ قَدِيرٌ۝٤

4تم سب کی بازگشت خدا ہی کی طرف ہے اور وہ ہر شے پر قدرت رکھنے والا ہے

أَلَآ إِنَّهُمْ يَثْنُونَ صُدُورَهُمْ لِيَسْتَخْفُوا۟ مِنْهُ ۚ أَلَا حِينَ يَسْتَغْشُونَ ثِيَابَهُمْ يَعْلَمُ مَا يُسِرُّونَ وَمَا يُعْلِنُونَ ۚ إِنَّهُۥ عَلِيمٌۢ بِذَاتِ ٱلصُّدُورِ۝٥

5آگاہ ہوجاؤ کہ یہ لوگ اپنے سینوں کو دہرائے لے رہے ہیں کہ اس طرح پیغمبر سے چھپ جائیں تو آگاہ رہیں کہ یہ جب اپنے کپڑوں کو خوب لپیٹ لیتے ہیں تو اس وقت بھی وہ ان کے ظاہر و باطن دونوں کو جانتا ہے کہ وہ تمام سینوں کے رازوں کا جاننے والا ہے

٢٢٢

وَمَا مِن دَآبَّةٍ فِى ٱلْأَرْضِ إِلَّا عَلَى ٱللَّهِ رِزْقُهَا وَيَعْلَمُ مُسْتَقَرَّهَا وَمُسْتَوْدَعَهَا ۚ كُلٌّ فِى كِتَٰبٍ مُّبِينٍ۝٦

6اور زمین پر چلنے والی کوئی مخلوق ایسی نہیں ہے جس کا رزق خدا کے ذمہ نہ ہو -وہ ہر ایک کے سونپے جانے کی جگہ اور اس کے قرار کی منزل کو جانتا ہے اور سب کچھ کتاب مبین میں محفوظ ہے

وَهُوَ ٱلَّذِى خَلَقَ ٱلسَّمَٰوَٰتِ وَٱلْأَرْضَ فِى سِتَّةِ أَيَّامٍ وَكَانَ عَرْشُهُۥ عَلَى ٱلْمَآءِ لِيَبْلُوَكُمْ أَيُّكُمْ أَحْسَنُ عَمَلًا ۗ وَلَئِن قُلْتَ إِنَّكُم مَّبْعُوثُونَ مِنۢ بَعْدِ ٱلْمَوْتِ لَيَقُولَنَّ ٱلَّذِينَ كَفَرُوٓا۟ إِنْ هَٰذَآ إِلَّا سِحْرٌ مُّبِينٌ۝٧

7اور وہی وہ ہے جس نے آسمانوں اور زمین کو چھ دنوں میں پیدا کیا ہے اور اس کا تخت اقتدار پانی پر تھا تاکہ تمہیں آزمائے کہ تم میں سب سے بہتر عمل کرنے والا کون ہے اور اگر آپ کہیں گے کہ تم لوگ موت کے بعد پھر اٹھائے جانے والے ہو تو یہ کافر کہیں گے کہ یہ تو صرف ایک کھلا ہوا جادو ہے

وَلَئِنْ أَخَّرْنَا عَنْهُمُ ٱلْعَذَابَ إِلَىٰٓ أُمَّةٍ مَّعْدُودَةٍ لَّيَقُولُنَّ مَا يَحْبِسُهُۥٓ ۗ أَلَا يَوْمَ يَأْتِيهِمْ لَيْسَ مَصْرُوفًا عَنْهُمْ وَحَاقَ بِهِم مَّا كَانُوا۟ بِهِۦ يَسْتَهْزِءُونَ۝٨

8اور اگر ہم ان کے عذاب کو ایک معینہ مدّت کے لئے ٹال دیں تو طنز کریں گے کہ عذاب کو کس چیز نے روک لیا ہے -آگاہ ہوجاؤ کہ جس دن عذاب آجائے گا تو پھر پلٹنے والا نہیں ہے اور پھر وہ عذاب ان کو ہر طرف سے گھیر لے گا جس کا یہ مذاق اڑا رہے تھے

وَلَئِنْ أَذَقْنَا ٱلْإِنسَٰنَ مِنَّا رَحْمَةً ثُمَّ نَزَعْنَٰهَا مِنْهُ إِنَّهُۥ لَيَـُٔوسٌ كَفُورٌ۝٩

9اور اگر ہم انسان کو رحمت دے کر چھین لیتے ہیں تو مایوس ہوجاتا ہے اور کفر کرنے لگتا ہے

وَلَئِنْ أَذَقْنَٰهُ نَعْمَآءَ بَعْدَ ضَرَّآءَ مَسَّتْهُ لَيَقُولَنَّ ذَهَبَ ٱلسَّيِّـَٔاتُ عَنِّىٓ ۚ إِنَّهُۥ لَفَرِحٌ فَخُورٌ۝١٠

10اور اگر تکلیف پہنچنے کے بعد نعمت اور آرام کا مزہ چکھا دیتے ہیں تو کہتا ہے کہ اب تو ہماری ساری برائیاں چلی گئیں اور وہ خوش ہوکر اکڑنے لگتا ہے

إِلَّا ٱلَّذِينَ صَبَرُوا۟ وَعَمِلُوا۟ ٱلصَّٰلِحَٰتِ أُو۟لَٰٓئِكَ لَهُم مَّغْفِرَةٌ وَأَجْرٌ كَبِيرٌ۝١١

11علاوہ ان لوگوں کے جو ایمان لائے ہیں اور انہوں نے نیک اعمال کئے ہیں کہ ان کے لئے مغفرت ہے اور بہت بڑا اجر بھی ہے

فَلَعَلَّكَ تَارِكٌۢ بَعْضَ مَا يُوحَىٰٓ إِلَيْكَ وَضَآئِقٌۢ بِهِۦ صَدْرُكَ أَن يَقُولُوا۟ لَوْلَآ أُنزِلَ عَلَيْهِ كَنزٌ أَوْ جَآءَ مَعَهُۥ مَلَكٌ ۚ إِنَّمَآ أَنتَ نَذِيرٌ ۚ وَٱللَّهُ عَلَىٰ كُلِّ شَىْءٍ وَكِيلٌ۝١٢

12پس کیا تم ہماری وحی کے بعض حصوں کو اس لئے ترک کرنے والے ہو یا اس سے تمہارا سینہ اس لئے تنگ ہوا ہے کہ یہ لوگ کہیں گے کہ ان کے اوپر خزانہ کیوں نہیں نازل ہوا یا ان کے ساتھ َمُلک کیوں نہیں آیا ...تو آپ صرف عذاب الٰہی سے ڈرانے والے ہیں اور اللہ ہر شئے کا نگراں اور ذمہ دار ہے

٢٢٣

أَمْ يَقُولُونَ ٱفْتَرَىٰهُ ۖ قُلْ فَأْتُوا۟ بِعَشْرِ سُوَرٍ مِّثْلِهِۦ مُفْتَرَيَٰتٍ وَٱدْعُوا۟ مَنِ ٱسْتَطَعْتُم مِّن دُونِ ٱللَّهِ إِن كُنتُمْ صَٰدِقِينَ۝١٣

13کیا یہ لوگ کہتے ہیں کہ یہ قرآن بندے نے گڑھ لیا ہے تو کہہ دیجئے کہ اس کے جیسے دس سورہ گڑھ کر تم بھی لے آؤ اور اللہ کے علاوہ جس کو چاہو اپنی مدد کے لئے بلالو اگر تم اپنی بات میں سچے ہو

فَإِلَّمْ يَسْتَجِيبُوا۟ لَكُمْ فَٱعْلَمُوٓا۟ أَنَّمَآ أُنزِلَ بِعِلْمِ ٱللَّهِ وَأَن لَّآ إِلَٰهَ إِلَّا هُوَ ۖ فَهَلْ أَنتُم مُّسْلِمُونَ۝١٤

14پھر اگر یہ آپ کی بات قبول نہ کریں تو تم سب سمجھ لو کہ جو کچھ نازل کیا گیا ہے سب خدا کے علم سے ہے اور اس کے علاوہ کوئی خدا نہیں ہے تو کیا اب تم اسلام لانے والے ہو

مَن كَانَ يُرِيدُ ٱلْحَيَوٰةَ ٱلدُّنْيَا وَزِينَتَهَا نُوَفِّ إِلَيْهِمْ أَعْمَٰلَهُمْ فِيهَا وَهُمْ فِيهَا لَا يُبْخَسُونَ۝١٥

15جو شخص زندگانی دنیا اور اس کی زینت ہی چاہتا ہے ہم اس کے اعمال کا پورا پورا حساب یہیں کردیتے ہیں اور کسی طرح کی کمی نہیں کرتے ہیں

أُو۟لَٰٓئِكَ ٱلَّذِينَ لَيْسَ لَهُمْ فِى ٱلْـَٔاخِرَةِ إِلَّا ٱلنَّارُ ۖ وَحَبِطَ مَا صَنَعُوا۟ فِيهَا وَبَٰطِلٌ مَّا كَانُوا۟ يَعْمَلُونَ۝١٦

16اور یہی وہ ہیں جن کے لئے آخرت میں جہنم ّکے علاوہ کچھ نہیں ہے اور ان کے سارے کاروبار برباد ہوگئے ہیں اور سارے اعمال باطل و بے اثر ہوگئے ہیں

أَفَمَن كَانَ عَلَىٰ بَيِّنَةٍ مِّن رَّبِّهِۦ وَيَتْلُوهُ شَاهِدٌ مِّنْهُ وَمِن قَبْلِهِۦ كِتَٰبُ مُوسَىٰٓ إِمَامًا وَرَحْمَةً ۚ أُو۟لَٰٓئِكَ يُؤْمِنُونَ بِهِۦ ۚ وَمَن يَكْفُرْ بِهِۦ مِنَ ٱلْأَحْزَابِ فَٱلنَّارُ مَوْعِدُهُۥ ۚ فَلَا تَكُ فِى مِرْيَةٍ مِّنْهُ ۚ إِنَّهُ ٱلْحَقُّ مِن رَّبِّكَ وَلَٰكِنَّ أَكْثَرَ ٱلنَّاسِ لَا يُؤْمِنُونَ۝١٧

17کیا جو شخص اپنے رب کی طرف سے کھلی دلیل رکھتا ہے اور اس کے پیچھے اس کا گواہ بھی ہے اور اس کے پہلے موسٰی کی کتاب گواہی دے رہی ہے جو قوم کے لئے پیشوا اور رحمت تھی -وہ افترا کرے گا بیشک صاحبانِ ایمان اسی پر ایمان رکھتے ہیں اور جو لوگ اس کا انکار کرتے ہیں ان کاٹھکانہ جہنم ّہے تو خبردار تم اس قرآن کی طرف سے شک میں مبتلا نہ ہونا -یہ خدا کی طرف سے برحق ہے اگرچہ اکثر لوگ اس پر ایمان نہیں لاتے ہیں

وَمَنْ أَظْلَمُ مِمَّنِ ٱفْتَرَىٰ عَلَى ٱللَّهِ كَذِبًا ۚ أُو۟لَٰٓئِكَ يُعْرَضُونَ عَلَىٰ رَبِّهِمْ وَيَقُولُ ٱلْأَشْهَٰدُ هَٰٓؤُلَآءِ ٱلَّذِينَ كَذَبُوا۟ عَلَىٰ رَبِّهِمْ ۚ أَلَا لَعْنَةُ ٱللَّهِ عَلَى ٱلظَّٰلِمِينَ۝١٨

18اور اس سے بڑا ظالم کون ہے جو اللہ پر جھوٹا الزام لگاتا ہے. یہی وہ لوگ ہیں جو خدا کے سامنے پیش کئے جائیں گے تو سارے گواہ گواہی دیں گے کہ ان لوگوں نے خدا کے بارے میں غلط بیانی سے کام لیا ہے تو آگاہ ہوجاؤ کہ ظالمین پر خدا کی لعنت ہے

ٱلَّذِينَ يَصُدُّونَ عَن سَبِيلِ ٱللَّهِ وَيَبْغُونَهَا عِوَجًا وَهُم بِٱلْـَٔاخِرَةِ هُمْ كَٰفِرُونَ۝١٩

19جو راسِ خدا سے روکتے ہیں اور اس میں کجی پیدا کرنا چاہتے ہیں اور آخرت کے بارے میں کفر اور انکار کرنے والے ہیں

٢٢٤

أُو۟لَٰٓئِكَ لَمْ يَكُونُوا۟ مُعْجِزِينَ فِى ٱلْأَرْضِ وَمَا كَانَ لَهُم مِّن دُونِ ٱللَّهِ مِنْ أَوْلِيَآءَ ۘ يُضَٰعَفُ لَهُمُ ٱلْعَذَابُ ۚ مَا كَانُوا۟ يَسْتَطِيعُونَ ٱلسَّمْعَ وَمَا كَانُوا۟ يُبْصِرُونَ۝٢٠

20یہ لوگ نہ روئے زمین میں خدا کو عاجز کرسکتے ہیں اور نہ خد اکے علاوہ ان کا کوئی ناصر و مددگار ہے ان کا عذاب دگنا کردیا جائے گا کہ یہ نہ حق بات سن سکتے تھے اور نہ اس کے منظر عام کو دیکھ سکتے تھے

أُو۟لَٰٓئِكَ ٱلَّذِينَ خَسِرُوٓا۟ أَنفُسَهُمْ وَضَلَّ عَنْهُم مَّا كَانُوا۟ يَفْتَرُونَ۝٢١

21یہی وہ لوگ ہیں جنہوں نے خود اپنے نفس کو خسارہ میں مبتلا کیا اور ان سے وہ بھی گم ہوگئے جن کا افترا کیا کرتے تھے

لَا جَرَمَ أَنَّهُمْ فِى ٱلْـَٔاخِرَةِ هُمُ ٱلْأَخْسَرُونَ۝٢٢

22یقینا یہ لوگ آخرت میں بہت بڑا گھاٹا اٹھانے والے ہیں

إِنَّ ٱلَّذِينَ ءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّٰلِحَٰتِ وَأَخْبَتُوٓا۟ إِلَىٰ رَبِّهِمْ أُو۟لَٰٓئِكَ أَصْحَٰبُ ٱلْجَنَّةِ ۖ هُمْ فِيهَا خَٰلِدُونَ۝٢٣

23بیشک جو لوگ ایمان لے آئے اور انہوں نے نیک اعمال انجام دئیے اور اپنے رب کی بارگاہ میں عاجزی سے پیش آئے وہی اہل جّنت ہیں اور اس میں ہمیشہ رہنے والے ہیں

مَثَلُ ٱلْفَرِيقَيْنِ كَٱلْأَعْمَىٰ وَٱلْأَصَمِّ وَٱلْبَصِيرِ وَٱلسَّمِيعِ ۚ هَلْ يَسْتَوِيَانِ مَثَلًا ۚ أَفَلَا تَذَكَّرُونَ۝٢٤

24کافر اور مسلمان کی مثال اندھے بہرے اور دیکھنے سننے والے کی ہے تو کیا یہ دونوں مثال کے اعتبار سے برابر ہوسکتے ہیں تمہیں ہوش کیوں نہیں آتا ہے

وَلَقَدْ أَرْسَلْنَا نُوحًا إِلَىٰ قَوْمِهِۦٓ إِنِّى لَكُمْ نَذِيرٌ مُّبِينٌ۝٢٥

25اور ہم نے نوح کو ان کی قوم کی طرف اس پیغام کے ساتھ بھیجاکہ میں تمہارے لئے کھلے ہوئے عذاب الٰہی سے ڈرانے والا ہوں

أَن لَّا تَعْبُدُوٓا۟ إِلَّا ٱللَّهَ ۖ إِنِّىٓ أَخَافُ عَلَيْكُمْ عَذَابَ يَوْمٍ أَلِيمٍ۝٢٦

26اور یہ کہ خبردار تم اللہ کے علاوہ کسی کی عبادت نہ کرنا کہ میں تمہارے بارے میں دردناک دن کے عذاب کا خوف رکھتا ہوں

فَقَالَ ٱلْمَلَأُ ٱلَّذِينَ كَفَرُوا۟ مِن قَوْمِهِۦ مَا نَرَىٰكَ إِلَّا بَشَرًا مِّثْلَنَا وَمَا نَرَىٰكَ ٱتَّبَعَكَ إِلَّا ٱلَّذِينَ هُمْ أَرَاذِلُنَا بَادِىَ ٱلرَّأْىِ وَمَا نَرَىٰ لَكُمْ عَلَيْنَا مِن فَضْلٍۭ بَلْ نَظُنُّكُمْ كَٰذِبِينَ۝٢٧

27تو ان کی قوم کے بڑے لوگ جنہوں نے کفر اختیار کرلیا تھا -انہوں نے کہا کہ ہم تو تم کو اپنا ہی جیسا ایک انسان سمجھ رہے ہیں اور تمہارے اتباع کرنے والوں کو دیکھتے ہیں کہ وہ ہمارے پست طبقہ کے سادہ لوح افراد ہیں. ہم تم میں اپنے اوپر کوئی فضیلت نہیں دیکھتے ہیں بلکہ تمہیں جھوٹا خیال کرتے ہیں

قَالَ يَٰقَوْمِ أَرَءَيْتُمْ إِن كُنتُ عَلَىٰ بَيِّنَةٍ مِّن رَّبِّى وَءَاتَىٰنِى رَحْمَةً مِّنْ عِندِهِۦ فَعُمِّيَتْ عَلَيْكُمْ أَنُلْزِمُكُمُوهَا وَأَنتُمْ لَهَا كَٰرِهُونَ۝٢٨

28انہوں نے جواب دیا کہ اے قوم تمہارا کیا خیال ہے کہ اگر میں اپنے پروردگار کی طرف سے دلیل رکھتا ہوں اور وہ مجھے اپنی طرف سے وہ رحمت عطا کردے جو تمہیں دکھائی نہ دے تو کیا میں ناگواری کے باوجود زبردستی تمہارے اوپر لادسکتا ہوں

٢٢٥

وَيَٰقَوْمِ لَآ أَسْـَٔلُكُمْ عَلَيْهِ مَالًا ۖ إِنْ أَجْرِىَ إِلَّا عَلَى ٱللَّهِ ۚ وَمَآ أَنَا۠ بِطَارِدِ ٱلَّذِينَ ءَامَنُوٓا۟ ۚ إِنَّهُم مُّلَٰقُوا۟ رَبِّهِمْ وَلَٰكِنِّىٓ أَرَىٰكُمْ قَوْمًا تَجْهَلُونَ۝٢٩

29اے قوم میں تم سے کوئی مال تو نہیں چاہتا ہوں -میرا اجر تو اللہ کے ذمہ ہے اور میں صاحبان هایمان کو نکال بھی نہیں سکتا ہوں کہ وہ لوگ اپنے پروردگار سے ملاقات کرنے والے ہیں البتہ میں تم کو ایک جاہل قوم تصور کررہا ہوں

وَيَٰقَوْمِ مَن يَنصُرُنِى مِنَ ٱللَّهِ إِن طَرَدتُّهُمْ ۚ أَفَلَا تَذَكَّرُونَ۝٣٠

30اے قوم میں ان لوگوں کو نکال باہر کردوں تو اللہ کی طرف سے میرا مددگار کون ہوگا کیا تمہیں ہوش نہیں آتا ہے

وَلَآ أَقُولُ لَكُمْ عِندِى خَزَآئِنُ ٱللَّهِ وَلَآ أَعْلَمُ ٱلْغَيْبَ وَلَآ أَقُولُ إِنِّى مَلَكٌ وَلَآ أَقُولُ لِلَّذِينَ تَزْدَرِىٓ أَعْيُنُكُمْ لَن يُؤْتِيَهُمُ ٱللَّهُ خَيْرًا ۖ ٱللَّهُ أَعْلَمُ بِمَا فِىٓ أَنفُسِهِمْ ۖ إِنِّىٓ إِذًا لَّمِنَ ٱلظَّٰلِمِينَ۝٣١

31اور میں تم سے یہ بھی نہیں کہتا ہوں کہ میرے پاس تمام خدائی خزانے موجود ہیں اور نہ ہر غیب کے جاننے کا دعوٰی کرتا ہوں اور نہ یہ کہتا ہوں کہ میں فرشتہ ہوں اور نہ جو لوگ تمہارے نگاہوں میں ذلیل ہیں ان کے بارے میں یہ کہتا ہوں کہ خدا انہیں خیر نہ دے گا -اللہ ان کے دلوں سے خوب باخبر ہے -میں ایسا کہہ دوں گا تو ظالموں میں شمار ہوجاؤں گا

قَالُوا۟ يَٰنُوحُ قَدْ جَٰدَلْتَنَا فَأَكْثَرْتَ جِدَٰلَنَا فَأْتِنَا بِمَا تَعِدُنَآ إِن كُنتَ مِنَ ٱلصَّٰدِقِينَ۝٣٢

32ان لوگوں نے کہا کہ نوح آپ نے ہم سے جھگڑا کیا اور بہت جھگڑا کیا تو اب جس چیز کا وعدہ کررہے تھے اسے لے آؤ اگر تم اپنے دعوٰی میں سچے ہو

قَالَ إِنَّمَا يَأْتِيكُم بِهِ ٱللَّهُ إِن شَآءَ وَمَآ أَنتُم بِمُعْجِزِينَ۝٣٣

33نوح نے کہا کہ وہ تو خدا لے آئے گا اگر چاہے گا اور تم اسے عاجز بھی نہیں کرسکتے ہو

وَلَا يَنفَعُكُمْ نُصْحِىٓ إِنْ أَرَدتُّ أَنْ أَنصَحَ لَكُمْ إِن كَانَ ٱللَّهُ يُرِيدُ أَن يُغْوِيَكُمْ ۚ هُوَ رَبُّكُمْ وَإِلَيْهِ تُرْجَعُونَ۝٣٤

34اور میں تمہیں نصیحت بھی کرنا چاہوں تو میری نصیحت تمہارے کام نہیں آئے گی اگر خدا ہی تم کو گمراہی میں چھوڑ دینا چاہے -وہی تمہارا پروردگار ہے اور اسی کی طرف تم پلٹ کر جانے والے ہو

أَمْ يَقُولُونَ ٱفْتَرَىٰهُ ۖ قُلْ إِنِ ٱفْتَرَيْتُهُۥ فَعَلَىَّ إِجْرَامِى وَأَنَا۠ بَرِىٓءٌ مِّمَّا تُجْرِمُونَ۝٣٥

35کیا یہ لوگ یہ کہتے ہیں کہ انہوں نے اپنے پاس سے گڑھ لیا ہے تو آپ کہہ دیجئے کہ اگر میں نے گڑھا ہے تو اس کا جرم میرے ذمہ ہے اور میں تمہارے جرائم سے بری اور بیزار ہوں

وَأُوحِىَ إِلَىٰ نُوحٍ أَنَّهُۥ لَن يُؤْمِنَ مِن قَوْمِكَ إِلَّا مَن قَدْ ءَامَنَ فَلَا تَبْتَئِسْ بِمَا كَانُوا۟ يَفْعَلُونَ۝٣٦

36اور نوح کی طرف یہ وحی کی گئی کہ تمہاری قوم میں سے اب کوئی ایمان نہ لائے گا علاوہ ان کے جو ایمان لاچکے لہذا تم ان کے افعال سے رنجیدہ نہ ہو

وَٱصْنَعِ ٱلْفُلْكَ بِأَعْيُنِنَا وَوَحْيِنَا وَلَا تُخَٰطِبْنِى فِى ٱلَّذِينَ ظَلَمُوٓا۟ ۚ إِنَّهُم مُّغْرَقُونَ۝٣٧

37اور ہماری نگاہوں کے سامنے ہماری وحی کی نگرانی میں کشتی تیار کرو اور ظالموں کے بارے میں مجھ سے بات نہ کرو کہ یہ سب غرق ہوجانے والے ہیں

٢٢٦

وَيَصْنَعُ ٱلْفُلْكَ وَكُلَّمَا مَرَّ عَلَيْهِ مَلَأٌ مِّن قَوْمِهِۦ سَخِرُوا۟ مِنْهُ ۚ قَالَ إِن تَسْخَرُوا۟ مِنَّا فَإِنَّا نَسْخَرُ مِنكُمْ كَمَا تَسْخَرُونَ۝٣٨

38اور نوح کشتی بنارہے تھے اور جب بھی قوم کی کسی جماعت کا گزر ہوتا تھا تو ان کا مذاق اڑاتے تھے -نوح نے کہا کہ اگر تم ہمارا مذاق اڑاؤ گے تو کل ہم اسی طرح تمہارا بھی مذاق اڑائیں گے

فَسَوْفَ تَعْلَمُونَ مَن يَأْتِيهِ عَذَابٌ يُخْزِيهِ وَيَحِلُّ عَلَيْهِ عَذَابٌ مُّقِيمٌ۝٣٩

39پھر عنقریب تمہیں معلوم ہوجائے گا کہ جس کے پاس عذاب آتا ہے اسے رسوا کردیا جاتا ہے اور پھر وہ عذاب دائمی ہی ہوجاتا ہے

حَتَّىٰٓ إِذَا جَآءَ أَمْرُنَا وَفَارَ ٱلتَّنُّورُ قُلْنَا ٱحْمِلْ فِيهَا مِن كُلٍّ زَوْجَيْنِ ٱثْنَيْنِ وَأَهْلَكَ إِلَّا مَن سَبَقَ عَلَيْهِ ٱلْقَوْلُ وَمَنْ ءَامَنَ ۚ وَمَآ ءَامَنَ مَعَهُۥٓ إِلَّا قَلِيلٌ۝٤٠

40یہاں تک کہ جب ہمارا حکم آگیا اور تنور سے پانی ابلنے لگا تو ہم نے کہا کہ نوح اپنے ساتھ ہر جوڑے میں سے دو کو لے لو اور اپنے اہل کو بھی لے لو علاوہ ان کے جن کے بارے میں ہلاکت کا فیصلہ ہوچکا ہے اور صاحبان هایمان کو بھی لے لو اور ان کے ساتھ ایمان والے بہت ہی کم تھے

وَقَالَ ٱرْكَبُوا۟ فِيهَا بِسْمِ ٱللَّهِ مَجْر۪ىٰهَا وَمُرْسَىٰهَآ ۚ إِنَّ رَبِّى لَغَفُورٌ رَّحِيمٌ۝٤١

41نوح نے کہا کہ اب تم سب کشتی میں سوار ہوجاؤ خدا کے نام کے سہارے اس کا بہاؤ بھی ہے اور ٹھہراؤ بھی اور بیشک میرا پروردگار بڑا بخشنے والا مہربان ہے

وَهِىَ تَجْرِى بِهِمْ فِى مَوْجٍ كَٱلْجِبَالِ وَنَادَىٰ نُوحٌ ٱبْنَهُۥ وَكَانَ فِى مَعْزِلٍ يَٰبُنَىَّ ٱرْكَب مَّعَنَا وَلَا تَكُن مَّعَ ٱلْكَٰفِرِينَ۝٤٢

42اور وہ کشتی انہیں لے کر پہاڑوں جیسی موجوں کے درمیان چلی جارہی تھی کہ نوح نے اپنے فرزند کو آواز دی جو الگ جگہ پر تھا کہ فرزند ہمارے ساتھ کشتی میں سوار ہوجا اور کافروں میں نہ ہوجا

قَالَ سَـَٔاوِىٓ إِلَىٰ جَبَلٍ يَعْصِمُنِى مِنَ ٱلْمَآءِ ۚ قَالَ لَا عَاصِمَ ٱلْيَوْمَ مِنْ أَمْرِ ٱللَّهِ إِلَّا مَن رَّحِمَ ۚ وَحَالَ بَيْنَهُمَا ٱلْمَوْجُ فَكَانَ مِنَ ٱلْمُغْرَقِينَ۝٤٣

43اس نے کہا کہ میں عنقریب پہاڑ پر پناہ لے لوں گا وہ مجھے پانی سے بچالے گا -نوح نے کہا کہ آج حکم خدا سے کوئی بچانے والا نہیں ہے سوائے اس کے جس پر خود خدا رحم کرے اور پھر دونوں کے درمیان موج حائل ہوگئی اور وہ ڈوبنے والوں میں شامل ہوگیا

وَقِيلَ يَٰٓأَرْضُ ٱبْلَعِى مَآءَكِ وَيَٰسَمَآءُ أَقْلِعِى وَغِيضَ ٱلْمَآءُ وَقُضِىَ ٱلْأَمْرُ وَٱسْتَوَتْ عَلَى ٱلْجُودِىِّ ۖ وَقِيلَ بُعْدًا لِّلْقَوْمِ ٱلظَّٰلِمِينَ۝٤٤

44اور قدرت کا حکم ہوا کہ اے زمین اپنے پانی کو نگل لے اور اے آسمان اپنے پانی کو روک لے .اور پھر پانی گھٹ گیا اور کام تمام کردیا گیا اور کشتی کوئہ جودی پر ٹھہر گئی اور آواز آئی کہ ہلاکت قوم هظالمین کے لئے ہے

وَنَادَىٰ نُوحٌ رَّبَّهُۥ فَقَالَ رَبِّ إِنَّ ٱبْنِى مِنْ أَهْلِى وَإِنَّ وَعْدَكَ ٱلْحَقُّ وَأَنتَ أَحْكَمُ ٱلْحَٰكِمِينَ۝٤٥

45اور نوح نے اپنے پروردگار کو پکارا کہ پروردگار میرا فرزند میرے اہل میں سے ہے اور تیرا وعدہ اہل کو بچانے کا برحق ہے اور تو بہترین فیصلہ کرنے والا ہے

٢٢٧

قَالَ يَٰنُوحُ إِنَّهُۥ لَيْسَ مِنْ أَهْلِكَ ۖ إِنَّهُۥ عَمَلٌ غَيْرُ صَٰلِحٍ ۖ فَلَا تَسْـَٔلْنِ مَا لَيْسَ لَكَ بِهِۦ عِلْمٌ ۖ إِنِّىٓ أَعِظُكَ أَن تَكُونَ مِنَ ٱلْجَٰهِلِينَ۝٤٦

46ارشاد ہوا کہ نوح یہ تمہارے اہل سے نہیں ہے یہ عمل هغیر صالح ہے لہذا مجھ سے اس چیز کے بارے میں سوال نہ کرو جس کا تمہیں علم نہیں ہے -میں تمہیں نصیحت کرتا ہوں کہ تمہارا شمار جاہلوں میں نہ ہوجائے

قَالَ رَبِّ إِنِّىٓ أَعُوذُ بِكَ أَنْ أَسْـَٔلَكَ مَا لَيْسَ لِى بِهِۦ عِلْمٌ ۖ وَإِلَّا تَغْفِرْ لِى وَتَرْحَمْنِىٓ أَكُن مِّنَ ٱلْخَٰسِرِينَ۝٤٧

47نوح نے کہا کہ خدایا میں اس بات سے پناہ مانگتا ہوں کہ اس چیز کا سوال کروں جس کا علم نہ ہو اور اگر تو مجھے معاف نہ کرے گا اور مجھ پر رحم نہ کرے گا تو میں خسارہ والوں میں ہوجاؤں گا

قِيلَ يَٰنُوحُ ٱهْبِطْ بِسَلَٰمٍ مِّنَّا وَبَرَكَٰتٍ عَلَيْكَ وَعَلَىٰٓ أُمَمٍ مِّمَّن مَّعَكَ ۚ وَأُمَمٌ سَنُمَتِّعُهُمْ ثُمَّ يَمَسُّهُم مِّنَّا عَذَابٌ أَلِيمٌ۝٤٨

48ارشاد ہوا کہ نوح ہماری طرف سے سلامتی اور برکتوں کے ساتھ کشتی سے اترو یہ سلامتی اور برکت تمہارے ساتھ کی قوم پر ہے اور کچھ قومیں ہیں جنہیں ہم پہلے راحت دیں گے اس کے بعد ہماری طرف سے دردناک عذاب ملے گا

تِلْكَ مِنْ أَنۢبَآءِ ٱلْغَيْبِ نُوحِيهَآ إِلَيْكَ ۖ مَا كُنتَ تَعْلَمُهَآ أَنتَ وَلَا قَوْمُكَ مِن قَبْلِ هَٰذَا ۖ فَٱصْبِرْ ۖ إِنَّ ٱلْعَٰقِبَةَ لِلْمُتَّقِينَ۝٤٩

49پیغمبر علیھ السّلامیہ غیب کی خبریں ہیں جن کی ہم آپ کی طرف وحی کررہے ہیں جن کا علم نہ آپ کو تھا اور نہ آپ کی قوم کو لہذا آپ صبر کریں کہ انجام صاحبان هتقویٰ کے ہاتھ میں ہے

وَإِلَىٰ عَادٍ أَخَاهُمْ هُودًا ۚ قَالَ يَٰقَوْمِ ٱعْبُدُوا۟ ٱللَّهَ مَا لَكُم مِّنْ إِلَٰهٍ غَيْرُهُۥٓ ۖ إِنْ أَنتُمْ إِلَّا مُفْتَرُونَ۝٥٠

50اور ہم نے قوم عاد کی طرف ان کے بھائی ہود کو بھیجا تو انہوں نے کہا قوم والو اللہ کی عبادت کرو-اس کے علاوہ تمہارا کوئی معبود نہیں ہے.تم صرف افترا کرنے والے ہو

يَٰقَوْمِ لَآ أَسْـَٔلُكُمْ عَلَيْهِ أَجْرًا ۖ إِنْ أَجْرِىَ إِلَّا عَلَى ٱلَّذِى فَطَرَنِىٓ ۚ أَفَلَا تَعْقِلُونَ۝٥١

51قوم والو میں تم سے کسی اجرت کا سوال نہیں کرتا میرا اجر تو اس پروردگار کے ذمہ ہے جس نے مجھے پیدا کیا ہے کیا تم عقل استعمال نہیں کرتے ہو

وَيَٰقَوْمِ ٱسْتَغْفِرُوا۟ رَبَّكُمْ ثُمَّ تُوبُوٓا۟ إِلَيْهِ يُرْسِلِ ٱلسَّمَآءَ عَلَيْكُم مِّدْرَارًا وَيَزِدْكُمْ قُوَّةً إِلَىٰ قُوَّتِكُمْ وَلَا تَتَوَلَّوْا۟ مُجْرِمِينَ۝٥٢

52اے قوم خدا سے استغفار کرو اس کے بعد اس کی طرف ہمہ تن متوجہ ہوجاؤ وہ آسمان سے موسلادھار پانی برسائے گا اور تمہاری موجودہ قوت میں قوت کا اضافہ کردے گا اور خبردار مجرموں کی طرح منہ نہ پھیرلینا

قَالُوا۟ يَٰهُودُ مَا جِئْتَنَا بِبَيِّنَةٍ وَمَا نَحْنُ بِتَارِكِىٓ ءَالِهَتِنَا عَن قَوْلِكَ وَمَا نَحْنُ لَكَ بِمُؤْمِنِينَ۝٥٣

53ان لوگوں نے کہا اے ہود تم کوئی معجزہ تو لائے نہیں اور ہم صرف تمہارے کہنے پر اپنے خداؤں کو چھوڑنے والے اور تمہاری بات پر ایمان لانے والے نہیں ہیں

٢٢٨

إِن نَّقُولُ إِلَّا ٱعْتَرَىٰكَ بَعْضُ ءَالِهَتِنَا بِسُوٓءٍ ۗ قَالَ إِنِّىٓ أُشْهِدُ ٱللَّهَ وَٱشْهَدُوٓا۟ أَنِّى بَرِىٓءٌ مِّمَّا تُشْرِكُونَ۝٥٤

54ہم صرف یہ کہنا چاہتے ہیں کہ ہمارے خداؤں میں سے کسی نے آپ کو دیوانہ بنادیا ہے -ہود نے کہا کہ میں خدا کو گواہ بناکر کہتا ہوں اور تم بھی گواہ رہنا کہ میں تمہارے شرک سے بیزار ہوں

مِن دُونِهِۦ ۖ فَكِيدُونِى جَمِيعًا ثُمَّ لَا تُنظِرُونِ۝٥٥

55لہذا تم سب مل کر میرے ساتھ مکاری کرو اور مجھے مہلت نہ دو

إِنِّى تَوَكَّلْتُ عَلَى ٱللَّهِ رَبِّى وَرَبِّكُم ۚ مَّا مِن دَآبَّةٍ إِلَّا هُوَ ءَاخِذٌۢ بِنَاصِيَتِهَآ ۚ إِنَّ رَبِّى عَلَىٰ صِرَٰطٍ مُّسْتَقِيمٍ۝٥٦

56میرا اعتماد پروردگار پر ہے جو میرا اور تمہارا سب کا خدا ہے اور کوئی زمین پر چلنے والا ایسا نہیںہے جس کی پیشانی اس کے قبضہ میں نہ ہو میرے پروردگار کا راستہ بالکل سیدھا ہے

فَإِن تَوَلَّوْا۟ فَقَدْ أَبْلَغْتُكُم مَّآ أُرْسِلْتُ بِهِۦٓ إِلَيْكُمْ ۚ وَيَسْتَخْلِفُ رَبِّى قَوْمًا غَيْرَكُمْ وَلَا تَضُرُّونَهُۥ شَيْـًٔا ۚ إِنَّ رَبِّى عَلَىٰ كُلِّ شَىْءٍ حَفِيظٌ۝٥٧

57اس کے بعد بھی انحراف کرو تو میں نے خدائی پیغام کو پہنچادیا ہے اب خدا تمہاری جگہ پر دوسری قوموں کو لے آئے گا اور تم اس کا کچھ نہیں بگاڑ سکتے ہو بیشک میرا پروردگار ہر شے کا نگراں ہے

وَلَمَّا جَآءَ أَمْرُنَا نَجَّيْنَا هُودًا وَٱلَّذِينَ ءَامَنُوا۟ مَعَهُۥ بِرَحْمَةٍ مِّنَّا وَنَجَّيْنَٰهُم مِّنْ عَذَابٍ غَلِيظٍ۝٥٨

58اور جب ہمارا حکم آگیا تو ہم نے ہود اور ان کے ساتھ ایمان لانے والوں کو اپنی رحمت سے بچالیااور انہیں سخت عذاب سے نجات دے دی

وَتِلْكَ عَادٌ ۖ جَحَدُوا۟ بِـَٔايَٰتِ رَبِّهِمْ وَعَصَوْا۟ رُسُلَهُۥ وَٱتَّبَعُوٓا۟ أَمْرَ كُلِّ جَبَّارٍ عَنِيدٍ۝٥٩

59یہ قوم عاد ہے جس نے پروردگار کی آیتوں کا انکار کیا اس کے رسولوں کی نافرمانی کی اور ہر ظالم و سرکش کا اتباع کرلیا

وَأُتْبِعُوا۟ فِى هَٰذِهِ ٱلدُّنْيَا لَعْنَةً وَيَوْمَ ٱلْقِيَٰمَةِ ۗ أَلَآ إِنَّ عَادًا كَفَرُوا۟ رَبَّهُمْ ۗ أَلَا بُعْدًا لِّعَادٍ قَوْمِ هُودٍ۝٦٠

60اور اس دنیا میں بھی اور قیامت میں بھی ان کے پیچھے لعنت لگادی گئی ہے -آگاہ ہوجاؤ کہ عاد نے اپنے پررودگار کا کفر کیا تو اب ہود کی قوم عاد کے لئے ہلاکت ہی ہلاکت ہے

وَإِلَىٰ ثَمُودَ أَخَاهُمْ صَٰلِحًا ۚ قَالَ يَٰقَوْمِ ٱعْبُدُوا۟ ٱللَّهَ مَا لَكُم مِّنْ إِلَٰهٍ غَيْرُهُۥ ۖ هُوَ أَنشَأَكُم مِّنَ ٱلْأَرْضِ وَٱسْتَعْمَرَكُمْ فِيهَا فَٱسْتَغْفِرُوهُ ثُمَّ تُوبُوٓا۟ إِلَيْهِ ۚ إِنَّ رَبِّى قَرِيبٌ مُّجِيبٌ۝٦١

61اور ہم نے قوم ثمود کی طرف ان کے بھائی صالح کو بھیجا اور انہوں نے کہا کہ اے قوم اللہ کی عبادت کرو اس کے علاوہ کوئی خدا نہیں ہے اس نے تمہیںزمین سے پیدا کیا ہے اور اس میں آباد کیا ہے اب اس سے استغفار کرو اور اس کی طرف متوجہ ہوجاؤ کہ میرا پروردگار قریب تر اور دعاؤں کا قبول کرنے والا ہے

قَالُوا۟ يَٰصَٰلِحُ قَدْ كُنتَ فِينَا مَرْجُوًّا قَبْلَ هَٰذَآ ۖ أَتَنْهَىٰنَآ أَن نَّعْبُدَ مَا يَعْبُدُ ءَابَآؤُنَا وَإِنَّنَا لَفِى شَكٍّ مِّمَّا تَدْعُونَآ إِلَيْهِ مُرِيبٍ۝٦٢

62ان لوگوں نے کہا کہ اے صالح اس سے پہلے تم سے بڑی امیدیں وابستہ تھیں کیا تم اس بات سے روکتے ہو کہ ہم اپنے بزرگوں کے معبودوں کی پرستش کریں -ہم یقینا تمہاری دعوت کی طرف سے شک اور شبہ میں ہیں

٢٢٩

قَالَ يَٰقَوْمِ أَرَءَيْتُمْ إِن كُنتُ عَلَىٰ بَيِّنَةٍ مِّن رَّبِّى وَءَاتَىٰنِى مِنْهُ رَحْمَةً فَمَن يَنصُرُنِى مِنَ ٱللَّهِ إِنْ عَصَيْتُهُۥ ۖ فَمَا تَزِيدُونَنِى غَيْرَ تَخْسِيرٍ۝٦٣

63انہوں نے کہا اے قوم تمہارا کیا خیال ہے کہ اگر میں اپنے پروردگار کی طرف سے دلیل رکھتا ہوں اور اس نے مجھے رحمت عطا کی ہے تو اگر میں اس کی نافرمانی کروں گا تو اس کے مقابلہ میں کون میری مدد کرسکے گا تم تو گھاٹے میں اضافہ کے علاوہ کچھ نہیں کرسکتے ہو

وَيَٰقَوْمِ هَٰذِهِۦ نَاقَةُ ٱللَّهِ لَكُمْ ءَايَةً فَذَرُوهَا تَأْكُلْ فِىٓ أَرْضِ ٱللَّهِ وَلَا تَمَسُّوهَا بِسُوٓءٍ فَيَأْخُذَكُمْ عَذَابٌ قَرِيبٌ۝٦٤

64اے قوم یہ ناقہ اللہ کی طرف سے ایک نشانی ہے اسے آزاد چھوڑ دو تاکہ زمین هخدا میں چین سے کھائے اور اسے کسی طرح کی تکلیف نہ دینا کہ تمہیں جلدی ہی کوئی عذاب اپنی گرفت میں لے لے

فَعَقَرُوهَا فَقَالَ تَمَتَّعُوا۟ فِى دَارِكُمْ ثَلَٰثَةَ أَيَّامٍ ۖ ذَٰلِكَ وَعْدٌ غَيْرُ مَكْذُوبٍ۝٦٥

65اس کے بعد بھی ان لوگوں نے اس کی کونچیں کاٹ دیں تو صالح نے کہا کہ اب اپنے گھروں میں تین دن تک اور آرام کرو کہ یہ وعدہ الٰہی ہے جو غلط نہیں ہوسکتا ہے

فَلَمَّا جَآءَ أَمْرُنَا نَجَّيْنَا صَٰلِحًا وَٱلَّذِينَ ءَامَنُوا۟ مَعَهُۥ بِرَحْمَةٍ مِّنَّا وَمِنْ خِزْىِ يَوْمِئِذٍ ۗ إِنَّ رَبَّكَ هُوَ ٱلْقَوِىُّ ٱلْعَزِيزُ۝٦٦

66اس کے بعد جب ہمارا حکم عذاب آپہنچا تو ہم نے صالح اور ان کے ساتھ کے صاحبانِ ایمان کو اپنی رحمت سے اپنے عذاب اور اس دن کی رسوائی سے بچالیا کہ تمہارا پروردگار صاحب هقوت اور سب پر غالب ہے

وَأَخَذَ ٱلَّذِينَ ظَلَمُوا۟ ٱلصَّيْحَةُ فَأَصْبَحُوا۟ فِى دِيَٰرِهِمْ جَٰثِمِينَ۝٦٧

67اور ظلم کرنے والوں کو ایک چنگھاڑنے اپنی لپیٹ میں لے لیا تو وہ اپنے دیار میں اوندھے پڑے رہ گئے

كَأَن لَّمْ يَغْنَوْا۟ فِيهَآ ۗ أَلَآ إِنَّ ثَمُودَا۟ كَفَرُوا۟ رَبَّهُمْ ۗ أَلَا بُعْدًا لِّثَمُودَ۝٦٨

68جیسے کبھی یہاں بسے ہی نہیں تھے -آگاہ ہوجاؤ کہ ثمود نے اپنے رب کی نافرمانی کی ہے اور ہوشیار ہوجاؤ کہ ثمود کے لئے ہلاکت ہی ہلاکت ہے

وَلَقَدْ جَآءَتْ رُسُلُنَآ إِبْرَٰهِيمَ بِٱلْبُشْرَىٰ قَالُوا۟ سَلَٰمًا ۖ قَالَ سَلَٰمٌ ۖ فَمَا لَبِثَ أَن جَآءَ بِعِجْلٍ حَنِيذٍ۝٦٩

69اور ابراہیم کے پاس ہمارے نمائندے بشارت لے کر آئے اور آکر سلام کیا تو ابراہیم نے بھی سلام کیا اور تھوڑی دیر نہ گزری تھی کہ بھنا ہوا بچھڑا لے آئے

فَلَمَّا رَءَآ أَيْدِيَهُمْ لَا تَصِلُ إِلَيْهِ نَكِرَهُمْ وَأَوْجَسَ مِنْهُمْ خِيفَةً ۚ قَالُوا۟ لَا تَخَفْ إِنَّآ أُرْسِلْنَآ إِلَىٰ قَوْمِ لُوطٍ۝٧٠

70اور جب دیکھا کہ ان لوگوں کے ہاتھ ادھر نہیں بڑھ رہے ہیں تو تعجب کیا اور ان کی طرف سے خوف محسوس کیا انہوں نے کہا کہ آپ ڈریں نہیں ہم قوم لوط کی طرف بھیجے گئے ہیں

وَٱمْرَأَتُهُۥ قَآئِمَةٌ فَضَحِكَتْ فَبَشَّرْنَٰهَا بِإِسْحَٰقَ وَمِن وَرَآءِ إِسْحَٰقَ يَعْقُوبَ۝٧١

71ابراہیم کی زوجہ اسی جگہ کھڑی تھیں یہ سن کر ہنس پڑیں تو ہم نے انہیں اسحاق کی بشارت دے دی اور اسحاق کے بعد پھر یعقوب کی بشارت دی

٢٣٠

قَالَتْ يَٰوَيْلَتَىٰٓ ءَأَلِدُ وَأَنَا۠ عَجُوزٌ وَهَٰذَا بَعْلِى شَيْخًا ۖ إِنَّ هَٰذَا لَشَىْءٌ عَجِيبٌ۝٧٢

72تو انہوں نے کہا کہ یہ مصیبت اب میرے یہاں بچہ ہوگا جب کہ میں بھی بوڑھی ہوں اور میرے میاں بھی بوڑھے ہیں یہ تو بالکل عجیب سی بات ہے

قَالُوٓا۟ أَتَعْجَبِينَ مِنْ أَمْرِ ٱللَّهِ ۖ رَحْمَتُ ٱللَّهِ وَبَرَكَٰتُهُۥ عَلَيْكُمْ أَهْلَ ٱلْبَيْتِ ۚ إِنَّهُۥ حَمِيدٌ مَّجِيدٌ۝٧٣

73فرشتوں نے کہا کہ کیا تمہیں حکم الٰہی میں تعجب ہورہا ہے اللہ کی رحمت اور برکت تم گھر والوں پر ہے وہ قابلِ حمد اور صاحب همجد و بزرگی ہے

فَلَمَّا ذَهَبَ عَنْ إِبْرَٰهِيمَ ٱلرَّوْعُ وَجَآءَتْهُ ٱلْبُشْرَىٰ يُجَٰدِلُنَا فِى قَوْمِ لُوطٍ۝٧٤

74اس کے بعد جب ابراہیم کا خوف برطرف ہوا اور ان کے پاس بشارت بھی آچکی تو انہوں نے ہم سے قوم لوط کے بارے میں اصرار کرنا شروع کردیا

إِنَّ إِبْرَٰهِيمَ لَحَلِيمٌ أَوَّٰهٌ مُّنِيبٌ۝٧٥

75بیشک ابراہیم بہت ہی دردمند اور خدا کی طرف رجوع کرنے والے تھے

يَٰٓإِبْرَٰهِيمُ أَعْرِضْ عَنْ هَٰذَآ ۖ إِنَّهُۥ قَدْ جَآءَ أَمْرُ رَبِّكَ ۖ وَإِنَّهُمْ ءَاتِيهِمْ عَذَابٌ غَيْرُ مَرْدُودٍ۝٧٦

76ابراہیم اس بات سے اعراض کرو -اب حکم خدا آچکا ہے اور ان لوگوں تک وہ عذاب آنے والا ہے جو پلٹایا نہیں جاسکتا

وَلَمَّا جَآءَتْ رُسُلُنَا لُوطًا سِىٓءَ بِهِمْ وَضَاقَ بِهِمْ ذَرْعًا وَقَالَ هَٰذَا يَوْمٌ عَصِيبٌ۝٧٧

77اور جب ہمارے فرستادے لوط کے پاس پہنچے تو وہ ان کے خیال سے رنجیدہ ہوئے اور تنگ دل ہوگئے اور کہا کہ یہ بڑا سخت دن ہے

وَجَآءَهُۥ قَوْمُهُۥ يُهْرَعُونَ إِلَيْهِ وَمِن قَبْلُ كَانُوا۟ يَعْمَلُونَ ٱلسَّيِّـَٔاتِ ۚ قَالَ يَٰقَوْمِ هَٰٓؤُلَآءِ بَنَاتِى هُنَّ أَطْهَرُ لَكُمْ ۖ فَٱتَّقُوا۟ ٱللَّهَ وَلَا تُخْزُونِ فِى ضَيْفِىٓ ۖ أَلَيْسَ مِنكُمْ رَجُلٌ رَّشِيدٌ۝٧٨

78اور ان کی قوم دوڑتی ہوئی آگئی اور اس کے پہلے بھی یہ لوگ ایسے برے کام کررہے تھے لوط نے کہا اے قوم یہ ہماری لڑکیاں تمہارے لئے زیادہ پاکیزہ ہیں خدا سے ڈرو اور مہمانوں کے بارے میں مجھے شرمندہ نہ کرو کیا تم میں کوئی سمجھدار آدمی نہیں ہے

قَالُوا۟ لَقَدْ عَلِمْتَ مَا لَنَا فِى بَنَاتِكَ مِنْ حَقٍّ وَإِنَّكَ لَتَعْلَمُ مَا نُرِيدُ۝٧٩

79ان لوگوں نے کہا کہ آپ جانتے ہیں کہ ہمیں آپ کی لڑکیوں میں کوئی دلچسپی نہیں ہے اور آپ کو معلوم ہے کہ ہم کیا چاہتے ہیں

قَالَ لَوْ أَنَّ لِى بِكُمْ قُوَّةً أَوْ ءَاوِىٓ إِلَىٰ رُكْنٍ شَدِيدٍ۝٨٠

80لوط نے کہا کاش میرے پاس قوت ہوتی یا میں کسی مضبوط قلعہ میں پناہ لے سکتا

قَالُوا۟ يَٰلُوطُ إِنَّا رُسُلُ رَبِّكَ لَن يَصِلُوٓا۟ إِلَيْكَ ۖ فَأَسْرِ بِأَهْلِكَ بِقِطْعٍ مِّنَ ٱلَّيْلِ وَلَا يَلْتَفِتْ مِنكُمْ أَحَدٌ إِلَّا ٱمْرَأَتَكَ ۖ إِنَّهُۥ مُصِيبُهَا مَآ أَصَابَهُمْ ۚ إِنَّ مَوْعِدَهُمُ ٱلصُّبْحُ ۚ أَلَيْسَ ٱلصُّبْحُ بِقَرِيبٍ۝٨١

81تو فرشتوں نے کہا کہ ہم آپ کے پروردگار کے نمائندے ہیں یہ ظالم آپ تک ہرگز نہیں پہنچ سکتے ہیں -آپ اپنے گھر والوں کو لے کر رات کے کسے حّصے میں چلے جائیے اور کوئی شخص کسی کی طرف مڑ کر بھی نہ دیکھے سوائے آپ کی زوجہ کے کہ اس تک وہی عذاب آنے والا ہے جو قوم تک آنے والا ہے ان کا وقت مقرر ہنگام هصبح ہے اور کیا صبح قریب نہیں ہے

٢٣١

فَلَمَّا جَآءَ أَمْرُنَا جَعَلْنَا عَٰلِيَهَا سَافِلَهَا وَأَمْطَرْنَا عَلَيْهَا حِجَارَةً مِّن سِجِّيلٍ مَّنضُودٍ۝٨٢

82پھر جب ہمارا عذاب آگیا تو ہم نے زمین کو تہ و بالا کردیا اور ان پر مسلسل کھرنجے دار پتھروں کی بارش کردی

مُّسَوَّمَةً عِندَ رَبِّكَ ۖ وَمَا هِىَ مِنَ ٱلظَّٰلِمِينَ بِبَعِيدٍ۝٨٣

83جن پر خدا کی طرف سے نشان لگے ہوئے تھے اور وہ بستی ان ظالموں سے کچھ دور نہیں ہے

وَإِلَىٰ مَدْيَنَ أَخَاهُمْ شُعَيْبًا ۚ قَالَ يَٰقَوْمِ ٱعْبُدُوا۟ ٱللَّهَ مَا لَكُم مِّنْ إِلَٰهٍ غَيْرُهُۥ ۖ وَلَا تَنقُصُوا۟ ٱلْمِكْيَالَ وَٱلْمِيزَانَ ۚ إِنِّىٓ أَرَىٰكُم بِخَيْرٍ وَإِنِّىٓ أَخَافُ عَلَيْكُمْ عَذَابَ يَوْمٍ مُّحِيطٍ۝٨٤

84اور ہم نے مدین کی طرف ان کے بھائی شعیب کو بھیجا تو انہوں نے کہا کہ اے قوم اللہ کی عبادت کر کہ اس کہ علاوہ تیرا کوئی خدا نہیں ہے اور خبردار ناپ تول میں کمی نہ کرنا کہ میں تمہیں بھلائی میں دیکھ رہا ہوں اور میں تمہارے بارے میں اس دن کے عذاب سے ڈرتا ہوں جو سب کو احاطہ کرلے گا

وَيَٰقَوْمِ أَوْفُوا۟ ٱلْمِكْيَالَ وَٱلْمِيزَانَ بِٱلْقِسْطِ ۖ وَلَا تَبْخَسُوا۟ ٱلنَّاسَ أَشْيَآءَهُمْ وَلَا تَعْثَوْا۟ فِى ٱلْأَرْضِ مُفْسِدِينَ۝٨٥

85اے قوم ناپ تول میں انصاف سے کام لو اور لوگوں کو کم چیزیں مت دو اور روئے زمین میں فساد مت پھیلاتے پھرو

بَقِيَّتُ ٱللَّهِ خَيْرٌ لَّكُمْ إِن كُنتُم مُّؤْمِنِينَ ۚ وَمَآ أَنَا۠ عَلَيْكُم بِحَفِيظٍ۝٨٦

86اللہ کی طرف کا ذخیرہ تمہارے حق میں بہت بہتر ہے اگر تم صاحبِ ایمان ہو اور میں تمہارے معاملات کا نگراں اور ذمہ دار نہیں ہوں

قَالُوا۟ يَٰشُعَيْبُ أَصَلَوٰتُكَ تَأْمُرُكَ أَن نَّتْرُكَ مَا يَعْبُدُ ءَابَآؤُنَآ أَوْ أَن نَّفْعَلَ فِىٓ أَمْوَٰلِنَا مَا نَشَٰٓؤُا۟ ۖ إِنَّكَ لَأَنتَ ٱلْحَلِيمُ ٱلرَّشِيدُ۝٨٧

87ان لوگوں نے طنز کیا کہ شعیب کیا تمہاری نماز تمہیں یہ حکم دیتی ہے کہ ہم اپنے بزرگوں کے معبودوں کو چھوڑ دیں یا اپنے اموال میں اپنی مرضی کے مطابق تصرف نہ کریں تم تو بڑے بردبار اور سمجھ دار معلوم ہوتے ہو

قَالَ يَٰقَوْمِ أَرَءَيْتُمْ إِن كُنتُ عَلَىٰ بَيِّنَةٍ مِّن رَّبِّى وَرَزَقَنِى مِنْهُ رِزْقًا حَسَنًا ۚ وَمَآ أُرِيدُ أَنْ أُخَالِفَكُمْ إِلَىٰ مَآ أَنْهَىٰكُمْ عَنْهُ ۚ إِنْ أُرِيدُ إِلَّا ٱلْإِصْلَٰحَ مَا ٱسْتَطَعْتُ ۚ وَمَا تَوْفِيقِىٓ إِلَّا بِٱللَّهِ ۚ عَلَيْهِ تَوَكَّلْتُ وَإِلَيْهِ أُنِيبُ۝٨٨

88شعیب نے کہا کہ اے قوم والو تمہارا کیا خیال ہے کہ اگر میں خدا کی طرف سے روشن دلیل رکھتا ہوں اور اس نے مجھے بہترین رزق عطا کردیا ہے اور میں یہ بھی نہیں چاہتا ہوں کہ جس چیز سے تم کو روکتا ہوں خود اسی کی خلاف ورزی کروں میں تو صرف اصلاح چاہتا ہوں جہاں تک میرے امکان میں ہو -میری توفیق صرف اللہ سے وابستہ ہے اسی پر میرا اعتماد ہے اور اسی کی طرف میں توجہ کررہا ہوں

٢٣٢

وَيَٰقَوْمِ لَا يَجْرِمَنَّكُمْ شِقَاقِىٓ أَن يُصِيبَكُم مِّثْلُ مَآ أَصَابَ قَوْمَ نُوحٍ أَوْ قَوْمَ هُودٍ أَوْ قَوْمَ صَٰلِحٍ ۚ وَمَا قَوْمُ لُوطٍ مِّنكُم بِبَعِيدٍ۝٨٩

89اور اے قوم کہیں میری مخالفت تم پر ایسا عذاب نازل نہ کرادے جیسا عذاب قوم نوح علیھ السّلام,قوم ہود علیھ السّلامیا قوم صالح علیھ السّلام پر نازل ہوا تھا اور قوم لوط بھی تم سے کچھ دور نہیں ہے

وَٱسْتَغْفِرُوا۟ رَبَّكُمْ ثُمَّ تُوبُوٓا۟ إِلَيْهِ ۚ إِنَّ رَبِّى رَحِيمٌ وَدُودٌ۝٩٠

90اور اپنے پروردگار سے استغفار کرو اس کے بعد اس کی طرف متوجہ ہوجاؤ کہ بیشک میرا پروردگار بہت مہربان اور محبت کرنے والا ہے

قَالُوا۟ يَٰشُعَيْبُ مَا نَفْقَهُ كَثِيرًا مِّمَّا تَقُولُ وَإِنَّا لَنَرَىٰكَ فِينَا ضَعِيفًا ۖ وَلَوْلَا رَهْطُكَ لَرَجَمْنَٰكَ ۖ وَمَآ أَنتَ عَلَيْنَا بِعَزِيزٍ۝٩١

91ان لوگوں نے کہا کہ اے شعیب آپ کی اکثر باتیں ہماری سمجھ میں نہیں آتی ہیں اور ہم توآپ کو اپنے درمیان کمزور ہی پارہے ہیں کہ اگر آپ کا قبیلہ نہ ہوتا تو ہم آپ کو سنگسار کردیتے اور آپ ہم پر غالب نہیں آسکتے تھے

قَالَ يَٰقَوْمِ أَرَهْطِىٓ أَعَزُّ عَلَيْكُم مِّنَ ٱللَّهِ وَٱتَّخَذْتُمُوهُ وَرَآءَكُمْ ظِهْرِيًّا ۖ إِنَّ رَبِّى بِمَا تَعْمَلُونَ مُحِيطٌ۝٩٢

92شعیب نے کہا کہ کیا میرا قبیلہ تمہاری نگاہ میں اللہ سے زیادہ عزیز ہے اور تم نے اللہ کو بالکل پس پشت ڈال دیا ہے جب کہ میرا پروردگار تمہارے اعمال کا خوب احاطہ کئے ہوئے ہے

وَيَٰقَوْمِ ٱعْمَلُوا۟ عَلَىٰ مَكَانَتِكُمْ إِنِّى عَٰمِلٌ ۖ سَوْفَ تَعْلَمُونَ مَن يَأْتِيهِ عَذَابٌ يُخْزِيهِ وَمَنْ هُوَ كَٰذِبٌ ۖ وَٱرْتَقِبُوٓا۟ إِنِّى مَعَكُمْ رَقِيبٌ۝٩٣

93اور اے قوم تم اپنی جگہ پر اپنا کام کرو میں اپنا کام کررہا ہوں عنقریب جان لو گے کہ کس کے پاس عذاب آکر اسے رسوا کردیتا ہے اور کون جھوٹا ہے اور انتظار کرو کہ میں بھی تمہارے ساتھ انتظار کرنے والا ہوں

وَلَمَّا جَآءَ أَمْرُنَا نَجَّيْنَا شُعَيْبًا وَٱلَّذِينَ ءَامَنُوا۟ مَعَهُۥ بِرَحْمَةٍ مِّنَّا وَأَخَذَتِ ٱلَّذِينَ ظَلَمُوا۟ ٱلصَّيْحَةُ فَأَصْبَحُوا۟ فِى دِيَٰرِهِمْ جَٰثِمِينَ۝٩٤

94اور جب ہمارا حکم (عذاب)آگیا تو ہم نے شعیب اور ان کے ساتھ ایمان لانے والوں کو اپنی رحمت سے بچالیا اور ظلم کرنے والوں کو ایک چنگھاڑنے پکڑ لیا تو وہ اپنے دیار ہی میں الٹ پلٹ ہوگئے

كَأَن لَّمْ يَغْنَوْا۟ فِيهَآ ۗ أَلَا بُعْدًا لِّمَدْيَنَ كَمَا بَعِدَتْ ثَمُودُ۝٩٥

95جیسے کبھی یہاں بسے ہی نہیں تھے اور آگاہ ہوجاؤ کہ قوم مدین کے لئے ویسے ہی ہلاکت ہے جیسے قوم ثمود ہلاک ہوگئی تھی

وَلَقَدْ أَرْسَلْنَا مُوسَىٰ بِـَٔايَٰتِنَا وَسُلْطَٰنٍ مُّبِينٍ۝٩٦

96اور ہم نے موسٰی کو اپنی نشانیوں اور روشن دلیل کے ساتھ بھیجا

إِلَىٰ فِرْعَوْنَ وَمَلَإِي۟هِۦ فَٱتَّبَعُوٓا۟ أَمْرَ فِرْعَوْنَ ۖ وَمَآ أَمْرُ فِرْعَوْنَ بِرَشِيدٍ۝٩٧

97فرعون اور اس کی قوم کی طرف تو لوگوں نے فرعون کے حکم کا اتباع کرلیا جب کہ فرعون کا حکم عقل و ہوش والا حکم نہیں تھا

٢٣٣

يَقْدُمُ قَوْمَهُۥ يَوْمَ ٱلْقِيَٰمَةِ فَأَوْرَدَهُمُ ٱلنَّارَ ۖ وَبِئْسَ ٱلْوِرْدُ ٱلْمَوْرُودُ۝٩٨

98وہ روزِ قیامت اپنی قوم کے آگے آگے چلے گا اور انہیں جہّنم ّمیں وارد کردے گا جو بدترین وارد ہونے کی جگہ ہے

وَأُتْبِعُوا۟ فِى هَٰذِهِۦ لَعْنَةً وَيَوْمَ ٱلْقِيَٰمَةِ ۚ بِئْسَ ٱلرِّفْدُ ٱلْمَرْفُودُ۝٩٩

99ان لوگوں کے پیچھے اس دنیا میں بھی لعنت لگادی گئی ہے اور روز قیامت بھی یہ بدترین عطیہ ہے جو انہیں دیا جائے گا

ذَٰلِكَ مِنْ أَنۢبَآءِ ٱلْقُرَىٰ نَقُصُّهُۥ عَلَيْكَ ۖ مِنْهَا قَآئِمٌ وَحَصِيدٌ۝١٠٠

100یہ چند بستیوں کی خبریں ہیں جو ہم آپ سے بیان کررہے ہیں -ان میں سے بعض باقی رہ گئی ہیں اور بعض کٹ پٹ کر برابر ہو گئی ہیں

وَمَا ظَلَمْنَٰهُمْ وَلَٰكِن ظَلَمُوٓا۟ أَنفُسَهُمْ ۖ فَمَآ أَغْنَتْ عَنْهُمْ ءَالِهَتُهُمُ ٱلَّتِى يَدْعُونَ مِن دُونِ ٱللَّهِ مِن شَىْءٍ لَّمَّا جَآءَ أَمْرُ رَبِّكَ ۖ وَمَا زَادُوهُمْ غَيْرَ تَتْبِيبٍ۝١٠١

101اور ہم نے ان پر کوئی ظلم نہیں کیا ہے بلکہ انہوں نے خود اپنے اوپر ظلم کیا ہے تو عذاب کے آجانے کے بعد ان کے وہ خدا بھی کام نہ آئے جنہیں وہ خدا کو چھوڑ کر پکار رہے تھے اور ان خداؤں نے مزید ہلاکت کے علاوہ انہیں کچھ نہیں دیا

وَكَذَٰلِكَ أَخْذُ رَبِّكَ إِذَآ أَخَذَ ٱلْقُرَىٰ وَهِىَ ظَٰلِمَةٌ ۚ إِنَّ أَخْذَهُۥٓ أَلِيمٌ شَدِيدٌ۝١٠٢

102اور اسی طرح تمہارے پروردگار کی گرفت ہوتی ہے جب وہ ظلم کرنے والی بستیوں کو اپنی گرفت میں لے لیتا ہے کہ اس کی گرفت بہت ہی سخت اور دردناک ہوتی ہے

إِنَّ فِى ذَٰلِكَ لَـَٔايَةً لِّمَنْ خَافَ عَذَابَ ٱلْـَٔاخِرَةِ ۚ ذَٰلِكَ يَوْمٌ مَّجْمُوعٌ لَّهُ ٱلنَّاسُ وَذَٰلِكَ يَوْمٌ مَّشْهُودٌ۝١٠٣

103اس بات میں ان لوگوں کے لئے نشانی پائی جاتی ہے جو عذاب آخرت سے ڈرنے والے ہیں -وہ دن جس دن تمام لوگ جمع کئے جائیں گے اور وہ سب کی حاضری کا دن ہوگا

وَمَا نُؤَخِّرُهُۥٓ إِلَّا لِأَجَلٍ مَّعْدُودٍ۝١٠٤

104اور ہم اپنے عذاب کو صرف ایک معینہ مدّت کے لئے ٹال رہے ہیں

يَوْمَ يَأْتِ لَا تَكَلَّمُ نَفْسٌ إِلَّا بِإِذْنِهِۦ ۚ فَمِنْهُمْ شَقِىٌّ وَسَعِيدٌ۝١٠٥

105اس کے بعد جس دن وہ آجائے گا تو کوئی شخص بھی ِذنِ خدا کے بغیر کسی سے بات بھی نہ کرسکے گا -اس دن کچھ بدبخت ہوں گے اور کچھ نیک بخت

فَأَمَّا ٱلَّذِينَ شَقُوا۟ فَفِى ٱلنَّارِ لَهُمْ فِيهَا زَفِيرٌ وَشَهِيقٌ۝١٠٦

106پس جو لوگ بدبخت ہوں گے وہ جہّنم میں رہیں گے جہاں اُن کے لئے صرف ہائے وائے اور چیخ پکار ہوگی

خَٰلِدِينَ فِيهَا مَا دَامَتِ ٱلسَّمَٰوَٰتُ وَٱلْأَرْضُ إِلَّا مَا شَآءَ رَبُّكَ ۚ إِنَّ رَبَّكَ فَعَّالٌ لِّمَا يُرِيدُ۝١٠٧

107وہ وہیں ہمیشہ رہنے والے ہیں جب تک آسمان و زمین قائم ہیں مگر یہ کہ آپ کا پروردگار نکالنا چاہے کہ وہ جو بھی چاہے کرسکتا ہے

وَأَمَّا ٱلَّذِينَ سُعِدُوا۟ فَفِى ٱلْجَنَّةِ خَٰلِدِينَ فِيهَا مَا دَامَتِ ٱلسَّمَٰوَٰتُ وَٱلْأَرْضُ إِلَّا مَا شَآءَ رَبُّكَ ۖ عَطَآءً غَيْرَ مَجْذُوذٍ۝١٠٨

108اور جو لوگ نیک بخت ہیں وہ جنت میں ہوں گے اور وہیں ہمیشہ رہیں گے جب تک کہ آسمان و زمین قائم ہیں مگر یہ کہ پروردگار اس کے خلاف چاہے ...._یہ خدا کی ایک عطا ہے جو ختم ہونے والی نہیں ہے

٢٣٤

فَلَا تَكُ فِى مِرْيَةٍ مِّمَّا يَعْبُدُ هَٰٓؤُلَآءِ ۚ مَا يَعْبُدُونَ إِلَّا كَمَا يَعْبُدُ ءَابَآؤُهُم مِّن قَبْلُ ۚ وَإِنَّا لَمُوَفُّوهُمْ نَصِيبَهُمْ غَيْرَ مَنقُوصٍ۝١٠٩

109لہذا خدا کے علاوہ جس کی بھی یہ پرستش کرتے ہیں اس کی طرف سے آپ کسی شبہ میں نہ پڑیں یہ اسی طرح پرستش کررہے ہیں جس طرح ان کے باپ دادا کررہے تھے اور ہم انہیں پورا پورا حّصہ بغیر کسی کمی کے دیں گے

وَلَقَدْ ءَاتَيْنَا مُوسَى ٱلْكِتَٰبَ فَٱخْتُلِفَ فِيهِ ۚ وَلَوْلَا كَلِمَةٌ سَبَقَتْ مِن رَّبِّكَ لَقُضِىَ بَيْنَهُمْ ۚ وَإِنَّهُمْ لَفِى شَكٍّ مِّنْهُ مُرِيبٍ۝١١٠

110اور ہم نے موسٰی علیھ السّلام کو کتاب دی تو اس میں بھی اختلاف پیدا کردیا گیا اور اگر تمہارے پروردگار کی طرف سے پہلے بات نہ ہوگئی ہوتی تو ان کے درمیان فیصلہ کردیا جاتا اور یہ لوگ اس عذاب کی طرف سے شک میں پڑے ہوئے ہیں

وَإِنَّ كُلًّا لَّمَّا لَيُوَفِّيَنَّهُمْ رَبُّكَ أَعْمَٰلَهُمْ ۚ إِنَّهُۥ بِمَا يَعْمَلُونَ خَبِيرٌ۝١١١

111اور یقیناتمہارا پروردگار سب کے اعمال کا پورا پورا بدلہ دے گا اور وہ ان سب کے اعمال سے خوب باخبر ہے

فَٱسْتَقِمْ كَمَآ أُمِرْتَ وَمَن تَابَ مَعَكَ وَلَا تَطْغَوْا۟ ۚ إِنَّهُۥ بِمَا تَعْمَلُونَ بَصِيرٌ۝١١٢

112لہذا آپ کو جس طرح حکم دیا گیا ہے اسی طرح استقامت سے کام لیں اور وہ بھی جنہوں نے آپ کے ساتھ توبہ کرلی ہے اور کوئی کسی طرح کی زیادتی نہ کرے کہ خدا سب کے اعمال کو خوب دیکھنے والا ہے

وَلَا تَرْكَنُوٓا۟ إِلَى ٱلَّذِينَ ظَلَمُوا۟ فَتَمَسَّكُمُ ٱلنَّارُ وَمَا لَكُم مِّن دُونِ ٱللَّهِ مِنْ أَوْلِيَآءَ ثُمَّ لَا تُنصَرُونَ۝١١٣

113اور خبردار تم لوگ ظالموں کی طرف جھکاؤ اختیار نہ کرنا کہ جہّنم کی آگ تمہیں چھولے گی اور خدا کے علاوہ تمہارا کوئی سرپرست نہیں ہوگا اور تمہاری مدد بھی نہیں کی جائے گی

وَأَقِمِ ٱلصَّلَوٰةَ طَرَفَىِ ٱلنَّهَارِ وَزُلَفًا مِّنَ ٱلَّيْلِ ۚ إِنَّ ٱلْحَسَنَٰتِ يُذْهِبْنَ ٱلسَّيِّـَٔاتِ ۚ ذَٰلِكَ ذِكْرَىٰ لِلذَّٰكِرِينَ۝١١٤

114اور پیغمبر آپ دن کے دونوں حصّو ں میں اور رات گئے نماز قائم کریں کہ نیکیاں برائیوں کو ختم کردینے والی ہیں اور یہ ذک» خدا کرنے والوں کے لئے ایک نصیحت ہے

وَٱصْبِرْ فَإِنَّ ٱللَّهَ لَا يُضِيعُ أَجْرَ ٱلْمُحْسِنِينَ۝١١٥

115اور آپ صبر سے کام لیں کہ خدا نیک عمل کرنے والوں کے اجر کو ضائع نہیں کرتا ہے

فَلَوْلَا كَانَ مِنَ ٱلْقُرُونِ مِن قَبْلِكُمْ أُو۟لُوا۟ بَقِيَّةٍ يَنْهَوْنَ عَنِ ٱلْفَسَادِ فِى ٱلْأَرْضِ إِلَّا قَلِيلًا مِّمَّنْ أَنجَيْنَا مِنْهُمْ ۗ وَٱتَّبَعَ ٱلَّذِينَ ظَلَمُوا۟ مَآ أُتْرِفُوا۟ فِيهِ وَكَانُوا۟ مُجْرِمِينَ۝١١٦

116تو تمہارے پہلے والے زمانوں اور نسلوں میں ایسے صاحبان عقل کیوں نہیں پیدا ہوئے ہیں جو لوگوں کو زمین میں فساد پھیلانے سے روکتے علاوہ ان چند افراد کے جنہیں ہم نے نجات دے دی اور ظالم تو اپنے عیش ہی کے پیچھے پڑے رہے اور یہ سب کے سب مجرم تھے

وَمَا كَانَ رَبُّكَ لِيُهْلِكَ ٱلْقُرَىٰ بِظُلْمٍ وَأَهْلُهَا مُصْلِحُونَ۝١١٧

117اور آپ کے پروردگار کا کام یہ نہیں ہے کہ کسی قریہ کو ظلم کرکے تباہ کردے جب کہ اس کے رہنے والے اصلاح کرنے والے ہوں

٢٣٥

وَلَوْ شَآءَ رَبُّكَ لَجَعَلَ ٱلنَّاسَ أُمَّةً وَٰحِدَةً ۖ وَلَا يَزَالُونَ مُخْتَلِفِينَ۝١١٨

118اور اگر آپ کا پروردگار چاہ لیتا تو سارے انسانوں کو ایک قوم بنادیتا (لیکن وہ جبر نہیں کرتاہے) لہذا یہ ہمیشہ مختلف ہی رہیں گے

إِلَّا مَن رَّحِمَ رَبُّكَ ۚ وَلِذَٰلِكَ خَلَقَهُمْ ۗ وَتَمَّتْ كَلِمَةُ رَبِّكَ لَأَمْلَأَنَّ جَهَنَّمَ مِنَ ٱلْجِنَّةِ وَٱلنَّاسِ أَجْمَعِينَ۝١١٩

119علاوہ ان کے جن پر خدا نے رحم کردیا ہو اور اسی لئے انہیں پیدا کیا ہے اور آپ کے پروردگار کی یہ بات قطعی حق ہے کہ ہم جہنم ّکو انسانوں اور جنوں سے بھر کر رہیں گے

وَكُلًّا نَّقُصُّ عَلَيْكَ مِنْ أَنۢبَآءِ ٱلرُّسُلِ مَا نُثَبِّتُ بِهِۦ فُؤَادَكَ ۚ وَجَآءَكَ فِى هَٰذِهِ ٱلْحَقُّ وَمَوْعِظَةٌ وَذِكْرَىٰ لِلْمُؤْمِنِينَ۝١٢٠

120اور ہم قدیم رسولوں کے واقعات آپ سے بیان کررہے ہیں کہ ان کے ذریعہ آپ کے دل کو مضبوط رکھیں اور ان واقعات میں حق ,نصیحت اور صاحبان هایمان کے لئے سامانِ عبرت بھی ہے

وَقُل لِّلَّذِينَ لَا يُؤْمِنُونَ ٱعْمَلُوا۟ عَلَىٰ مَكَانَتِكُمْ إِنَّا عَٰمِلُونَ۝١٢١

121اور آپ ان بے ایمانوں سے کہہ دیں کہ تم اپنی جگہ پر کام کرو اور ہم اپنی جگہ پر اپنا کام کررہے ہیں

122اور پھر انتظار کرو کہ ہم بھی انتظار کرنے والے ہیں

وَلِلَّهِ غَيْبُ ٱلسَّمَٰوَٰتِ وَٱلْأَرْضِ وَإِلَيْهِ يُرْجَعُ ٱلْأَمْرُ كُلُّهُۥ فَٱعْبُدْهُ وَتَوَكَّلْ عَلَيْهِ ۚ وَمَا رَبُّكَ بِغَٰفِلٍ عَمَّا تَعْمَلُونَ۝١٢٣

123اور اللہ ہی کے لئے آسمان اور زمین کا کل غیب ہے اور اسی کی طرف تمام امور کی بازگشت ہے لہذا آپ اسی کی عبادت کریں اور اسی پر اعتماد کریں کہ آپ کا رب لوگوں کے اعمال سے غافل نہیں ہے
