الغدير 6 ٥٣١

هذا رسولُ اللهِ حسبُکِ في غدٍ

يومَ الحسابِ إذا الخليلُ جفاکِ

ووصيُّهُ الهادي أبو حسنٍ إذا

أقبلتِ ظاميةً إليه سقاکِ

فهو المشفّعُ في المعادِ وخيرُ من

علقتْ به بعد النبيِّ يداکِ

وهو الذي للدينِ بعد خمولِهِ

حقّا أراکِ فهذّبت آراکِ

لولاه ما عُرف الهدى ونجوت من

متضايقِ الأشراکِ والإشراکِ

هو فُلکُ نوحٍ بين ممتسکٍ به

ناجٍ ومطّرحٍ مع الهلاّکِ

کم مارقٍ من مازقٍ قد غادرت

مزقاً حدود حسامه البتّاکِ ١

سل عنه بدراً حين بادرَ قاصمُ

الأملاکِ قائدَ موکبِ الأملاکِ

من صبَّ صوبَ دم الوليد ومن ترى

أخلا من الدهمِ الحماةِ حِماکِ

واسأل فوارسَها بأُحدٍ من ترى

ألقاک وجهَ الحتف عند لقاکِ

وأطاح طلحةَ عند مشتبکِ القنا

ولَواک قسراً عند نکسِ لواکِ

واسأل بخيبرَ خابريها من ترى

عفّى فناک ومن أباح فناکِ

وأذاق مرحبَکِ الردى وأحلّه

ضيقَ الشباکِ وفلَّ حدَّ شباکِ

واستخبري الأحزاب لمّا جرّدت

بيض المذاکي ٢ فوق جرد مذاکي

واستشعرت فرقاً جموعکِ إذ غدتْ

فرقاً وأدبر إذ قفاکِ قفاکِ

قد قلت حين تقدّمته عصابةٌ

جهلوا حقوقَ حقيقةِ الإدراکِ

لا تفرحي فبکثر ما استعذبتِ في

أولاکِ قد عُذِّبتِ في أُخراکِ

يا أُمّةً نقضت عهودَ نبيِّها

أفمن إلى نقضِ العهودِ دعاکِ

وصّاکِ خيراً بالوصيِّ کأنّما

متعمّداً في بغضِهِ وصّاکِ

أوَلم يقل فيه النبيُّ مبلِّغاً

هذا عليُّکِ في العلى أعلاکِف)

(١)إلى هنا القصيدة موجودة في أعيان الشيعة : 8 / 195.
(٢)جمع مذکاة وهی ما تذکي به النار من قطنة ونحوها وهی اسم آلة استعيرت للسيف بعلاقة أنّه تلتهب منه نار الحرب کما يلتهب الحطب بالمذکاة. (المؤلف)
