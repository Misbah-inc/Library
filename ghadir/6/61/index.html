الغدير 6 ٦١

وهوى إليک العذقُ ثمَّ رددتَهُ

في نخلةٍ تزهى به وتزانُ

والدوحتانِ وقد دعوتَ فأقبلا

حتى تلاقت منهما الأغصانُ

وشکا إليک الجيشُ من ظمأ به

فتفجّرت بالماءِ منک بنانُ

ورددتَ عينَ قتادةٍ من بعدِ ما

ذهبتْ فلم ينظرْ بها إنسانُ

وحکى ذراعُ الشاةِ مودعَ سمِّه

حتى کأنَّ العضوَ منه لسانُ

وعرجتَ في ظهرِ البُراقِ مجاوز ال

سبعِ الطباقِ کما يشَا الرحمنُ

والبدرُ شُقَّ وأشرقت شمسُ الضحى

بعدَ الغروب وما بها نقصانُ

وفضيلةٌ شهدَ الأنامُ بحقِّها

لا يستطيعُ جحودَها الإنسانُ

في الأرض ظلُّ الله کنتَ ولم يلُحْ

في الشمسِ ظلُّکَ إن حواک مکانُ

نُسختْ بمظهرِک المظاهرُ بعد ما

نُسختْ بملّةِ دينِک الأديانُ

وعلى نبوّتِکَ المعظَّمِ قدرُها

قامَ الدليلُ وأوضح البرهانُ

وبک استغاثَ الأنبياءُ جميعُهم

عند الشدائدِ ربَّهم ليُعانوا

أخذ الإلهُ لک العهودَ عليهمُ

من قبلِ ما سمحتْ بک الأزمانُ

وبک استغاثَ اللهَ آدمُ عندما

نُسب الخلافُ إليه والعصيانُ

وبک التجا نوحٌ وقد ماجت به

دُسُرُ السفينةِ إذ طغى الطوفان

وبک اغتدى أيّوبُ يسأل ربَّه

کشفَ البلاءِ فزالتِ الأحزانُ

وبک الخليلُ دعا الإلهَ فلم يخف

نمرودَ إذ شبّت له النيرانُ

وبک اغتدى في السجنِ يوسفُ سائلاً

ربَّ العباد وقلبه حيرانُ

وبک الکليمُ غداةَ خاطبَ ربَّهُ

سأل القبولَ فعمّه الإحسانُ

وبک المسيحُ دعا فأحيا ربُّهُ

ميتاً وقد بُليت به الأکفانُ

وبک استبانَ الحقُّ بعد خفائِهِ

حتى أطاعَکَ إنسُها والجانُ

ولو انَّني وفَّيتُ وصفَکَ حقَّه

فَنيَ الکلامُ وضاقتِ الأوزانُ

فعليک من ربِّ السلامِ سلامُهُ

والفضلُ والبرکاتُ والرضوانُ
