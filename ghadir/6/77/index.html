الغدير 6 ٧٧

فأجابه الصفيُّ المترجم بقوله :

ألا قل لشرِّ عبيدِ الإلهِ

وطاغي قريشٍ وکذّابِها

وباغي العبادِ وباغي العنادِ

وهاجي الکرامِ ومغتابِها

أأنت تُفاخرُ آلَ النبيِ

وتجحدُها فضلَ أحسابِها

بکم باهلَ المصطفى أم بهم

فردّ العداةَ بأوصابِها

أعنکم نفى الرجسَ أم عنهمُ

لطهر النفوسِ وألبابِها

أما الرجسُ والخمرُ من دابِکمْ

وفرطُ العبادةِ من دابِها

وقلتَ ورثنا ثيابَ النبيِ

فکم تجذبون بأهدابِها

وعندکَ لا يورثُ الأنبيا

فکيف حظيتم بأثوابِها

فکذّبت نفسَکَ في الحالتين

ولم تعلمِ الشهدَ من صابِها

أجدُّکَ يرضى بما قلتَهُ

وما کان يوماً بمرتابِها

وکان بصفِّينَ من حزبِهمْ

لحربِ الطغاةِ وأحزابِها

وقد شمَّر الموتُ عن ساقِه

وکشَّرتِ الحربُ عن نابِها

فأقبلَ يدعو إلى حيدرٍ

بإرغابها وبإرهابِها

وآثرَ أن ترتضيه الأنامُ

من الحکمينِ لأسبابِها

ليعطي الخلافَة أهلاً لها

فلم يرتضوه لإيجابِها

وصلّى مع الناسِ طولَ الحياةِ

وحيدر في صدر محرابِها

فهلاّ تقمّصها جدُّکم

إذا کان إذ ذاک أحرى بِها

إذا جعل الأمر شورى لهم

فهل کان من بعضِ أربابِها

أخامسَهم کان أم سادساً

وقد جليت بين خطّابِها

وقولک أنتم بنو بنتِهِ

ولکن بنو العمِّ أولى بِها

بنو البنتِ أيضاً بنو عمِّه

وذلک أدنى لأنسابِها

فدع في الخلافةِ فصلَ الخلافِ

فليست ذلولاً لرکّابِها
