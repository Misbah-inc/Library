الغدير 6 ٥٥٤

أيُقتلُ ظمآناً حسينٌ وجدُّه

إلى الناسِ من ربِّ العباد رسولُ

ويُمنَعُ شربَ الماءِ والسربُ آمنٌ

على الشربِ منها صادرٌ ونهولُ

وآلُ رسولِ اللهِ في دار غربةٍ

وآلُ زيادٍ في القصورِ نزولُ

وآلُ عليٍّ في القيودِ شواحبٌ

إذا أنّ مأسورٌ بکته ثکولُ

وآلُ أبي سفيان في عزِّ دولةٍ

تسير بهم تحت البنود خيولُ

مصابٌ أُصيب الدينُ منه بفادحٍ

تکادُ له شمُّ الجبالِ تزولُ

عليک ابنَ خيرِ المرسلينَ تأسّفي

وحزني وإن طال الزمانُ طويلُ

جللتَ فجلّ الرزءُ فيک على الورى

کذا کلُّ رزءٍ للجليلِ جليلُ

فليس بمُجدٍ فيک وجدي ولا البکا

مفيدٌ ولا الصبرُ الجميلُ جميلُ

إذا خفّ حزنُ الثاکلاتِ لسلوةٍ

فحزني على مرِّ الدهورِ ثقيلُ

وإن سَئِمَ الباکون فيک بکاءَهم

ملالاً فإنِّي للبکاء مُطيلُ

فما خفَّ من حزني عليک تأسّفي

ولا جفَّ من دمعي عليک مسيلُ

وينکر دمعي فيک من باتَ قلبُهُ

خليّا وما دمعُ الخليِّ هطولُ

وما هيَ إلاّ فيک نفسٌ نفيسةٌ

يحلّلها حَرُّ الأسى فتسيلُ

تباينَ فيک القائلون فمعجبٌ

کثيرٌ وذو حزنٍ عليکَ قليلُ

فأجرُ بني الدنيا عليک لشأنِهمْ

دنيٌّ وأجرُ المخلَصين جزيلُ

فإن فاتني إدراکُ يومِک سيّدي

وأخّرني عن نصرِ جيلِکَ جيلُ

فلي فيک أبکارٌ لوفقِ جناسِها

أصولُ بها للشامتين نُصولُ

لها رقّةُ المحزونِ فيک وخطبُها

جسيمٌ على أهلِ النفاق مَهولُ

يهيمُ بها سرُّ الوليِّ مسرّةً

وينصب منها ناصبٌ وجهولُ

لها في قلوبِ الملحدين عواسلٌ

ووقعُ نصولٍ ما لهنَّ نصولُ

بها من عليٍّ في علاکَ مناقبٌ

يقوم عليها في الکتابِ دليلُ

ينمُّ عن الأعرافِ طيِّبُ عَرفِها

فتعلقُها للعاقلين عقولُ
