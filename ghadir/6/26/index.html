الغدير 6 ٢٦

سهمُ بغي الأُلى أصابک من قب

لُ وللهِ عنک سهمٌ مصيبُ

أظهروا فيک حقد بدرٍ ومن قب

لُ دُعوا للهدى فلم يستجيبوا

يا بني أحمدٍ إلى مدحکم قل

ب الخليعيِّ مُستهامٌ طروبُ

کيف صبر امرئٍ يری الودَّ في القر

بي وجوباً وإرثُکم مغصوبُ

أنتمُ حجّةُ الإلهِ على الخل

ق وأنتم للطالبِ المطلوبُ

بولاکمْ وبغض أعدائکم تُقب

لُ أعمالُنا وتُمحى الذنوبُ

لثناکم شاهت وجوه ذوي النص

ب وشُقَّتْ من النفورِ القلوبُ

وله رحمه الله تعالى قوله :

سجعتْ فوق الغصونِ

فاقداتٌ للقرينِ

فاستهلّتْ سحبَ أجفا

ني وهزّتني شجوني

غرّدت لا شجوُها شج

وی ولا حنّت حنيني

لا ولا قلت لها

يا ورقُ بالنوح اسعديني

ما شجى الباکي طروباً

کشجى الباکي الحزينِ

حقَّ لي أبکي دماءً

عوضَ الدمعِ الهتونِ

لغريبٍ نازح الدا

ر خليٍّ من معينِ

لتريب الخدّ دامي ال

وجهِ مرضوضِ الجبينِ

ومنها :

يا بني طه وياسی

-ن وحاميمٍ ونونِ

بکمُ استعصمتُ من

شرِّ خطوبٍ تعتريني

فإذا خفتُ فأنتمْ

لنجاتي کالسفينِ

وعليکم ثقلُ ميزا

ني وأنتم تنقذوني

فاحشروا العبدَ الخليعيَ

إلى ذات اليمينِ
