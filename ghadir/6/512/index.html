الغدير 6 ٥١٢

والسيّدُالسجّادُالجبری المصري يُحمَلُ صاغراً

ويُقادُ في الأغلالِ وهو مقيّدُ

لا راحماً يشکو إليه مصابَهُ

في دارِ غربتِهِ ولا مُتودّدُ

يُهدى به وبرأسِ والدِه إلى

لکعٍ زنيمٍ کافرٍ يتمرّدُ

لا خيرَ في سفهاءِ قومٍ عبدُهمْ

ملکٌ يطاعُ وحرُّهم مستعبدُ

يا عينُ إن نفدتْ دموعُکِ فاسمحي

بدمٍ ولستُ إخالُ دمعَکِ ينفدُ

أسفاً على آلِ الرسولِ ومن بهمْ

رکنُ الهدى شرفاً يُشاد ويُعضدُ

منهم قتيلٌ لا يُجارُ ومن سُقي

سمّا وآخرُ عن حماه يشرَّدُ

ضاقت بلادُ اللهِ وهي فسيحةٌ

بهمُ وليس لهم بأرضٍ مقعدُ

متباعدون لهم بکلِّ تنوفةٍ ١

مستشهدٌ وبکلِّ أرضٍ مشهدُ

أَبَني المشاعرِ والحطيمِ ومن هُمُ

حججٌ بهم تشقى الأنامُ وتسعدُ

أقسمتُ لا ينفکُّ حزني دائماً

بکمُ ونارُ حشاشتي لا تخمدُ

بکم يميناً لا جرى في ناظري

حزناً عليکم غير دمعي مِرودُ

يفنى الزمانُ وتنقضي أيّامه

وعليُّکم بکمُ الحزينُ المکمدُ

فلجسمِهِ حللُ السقامِ ملابسٌ

ولطرفِهِ حرُّ المدامعِ أثمدُ

ولو انّني استمددتُ من عيني دماً

ويقلّ من عيني دماً يستمددُ

لم أقضِ حقَّکمُ عليَّ وکيف أن

تقضي حقوقَ المالکينَ الأعبُدُ

يا صفوةَ الجبّارِ يا مستودعي ال

-أسرارِ يا من ظلُّهم لي مقصدُ

عاهدتُکمْ في الذرِّ معرفةً بکم

ووفيتُ أيماناً بما أتعهّدُ

ووعدتموني في المعادِ شفاعةً

وعلى الصراطِ غداً يصحّ الموعدُ

فتفقّدوني في الحسابِ فإنّني

ثقةً بکم لوجوهِکمْ أتقصّدُ

کم مدحةٍ لي فيکمُ في طيِّها

حِکمٌ تفوز بها الرکاب وتنجدُف)

(١)التنوفة : البرّية لا ماء فيها ولا أنيس ، والجمع تنائف. (المؤلف)
