الغدير 6 ٥٥٢

أدلّتُها في الليلِ أضواءُ نورِها

وفي النقعِ أضواءُ السيوف دليلُ

يؤمُّ بها قصدَ المغالب أغلبٌ

فروسٌ لأشلاءِ الکماةِ أکولُ

له الخطُّ کوبٌ والجماجمُ أکؤسٌ

لديه وآذيُّ الدماءِ شمولُ ١

يرى الموتَ لا يخشاهُ والنبلُ واقعٌ

ولا يختشي وقعَ النبالِ نبيلُ

صئولٌ إذا کرَّ الکميُّ مناجزٌ

بليغُ إذا فاه البليغُ قؤولُ

له من عليٍّ في الخطوبِ شجاعةٌ

ومن أحمدٍ عند الخطابة قيلُ

إذا شمختْ في ذروةِ المجدِ هاشمٌ

فعمّاه منها جعفرٌ وعقيلُ

کفاه علوّا في البريّةِ أنّه

لأحمدَ والطهرِ البتولِ سليلُ

فما کلُّ جدٍّ في الرجالِ محمدٌ

ولا کلُّ أُمٍّ في النساءِ بتولُ

حسينٌ أخو المجدِ المنيفِ ومن له

فخارٌ إذا عُدَّ الفخارُ أثيلُ

أرى الموتَ عذباً في لهاک وصابُهُ

لغيرک مکروه المذاق وبيلُ

فما مرَّ ذو باس إلى مرِّ باسِهِ

على مهلٍ إلاّ وأنت عجولُ

کأنّ الأعاديَ حين صُلْتَ مبارزاً

کثيبٌ ذرته الريحُ وهو مهيلُ

وما نهل الخطيُّ منک ولا الظبا

ولا علَّ إلاّ وهو منک عليلُ

بنفسي وأهلي عافرَ الخطِّ حولَهُ

لدي الطفِّ من آل الرسولِ قبيلُ

کأنّ حسيناً فيهمُ بدرُ هالةٍ

کواکبُها حول السماکِ حلولُ

قضى ظامياً والماءُ طامٍ تصدّه

شرارُ الورى عن وردِهِ ونغولُ

وحُزَّ وريدُ السبطِ دون ورودِهِ

وغالته من أيدي الحوادثِ غولُ

وآب جواد السبط يهتف ناعياً

وقد ملأ البيداء منه صهيلُ

فلمّا سمعنَ الطاهراتُ نعيّه

لراکبه والسرجُ منه يميلُ

برزن سليباتِ الحليِّ نوادباً

لهنَّ على الندبِ الکريمِ عويلُج.

(١)الآذىّ : الموج.
