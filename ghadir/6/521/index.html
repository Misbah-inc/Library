الغدير 6 ٥٢١

أعلامُ دينٍ رُسَّخٍ لهمُ

في نشر کلِّ فضيلةٍ صدرُ

فکفاهمُ فخراً إذا افتخروا

ما دام حيّا فيهمُ الفخرُ

وصلوا نهارَهمُ بليلهمْ

نظراً وما لوصالِهم هجرُ

وطووا على مَضضٍ سرائرَهمْ

صبراً وليس لطيّها نشرُ

حتى يفضَّ ختامَها وبکمْ

يطفى بُعيد شرارِها الشرُّ

يا غائبينَ متى بقربکمُ

من بَعد وهن يُجبرُ الکسرُ

الفيءُ مقتسَمٌ لغيرِکمُ

وأکفُّکمْ من فيئکمْ صفرُ

والمال حلٌّ للعصاةِ ويُح

-رَمُه الکرامُ السادةُ الغرُّ

فنصيبهمْ منه الأعمُّ على

عصيانِهمْ ونصيبُکمْ نزرُ

يُمسون في أمنٍ وليس لهم

من طارقٍ ١ يغتالهم حذرُ

ويکاد من خوفٍ ومن جزعٍ

بکمُ يضيق البرُّ والبحرُ

ويقول بعد سبعة أبيات :

وإذا ذُکرتمْ في محافلهمْ

فوجوههمْ مُربِدةٌ صُفرُ

يتميّزون لذکرکمْ حنقاً

وعيونُهم مُزورّةٌ خزرُ

وعلى المنابرِ في بيوتکمُ

لأُولي الضلالةِ والعمى ذِکرُ

حال يسوء ذوي النهى وبه

يستبشر المتجاهل الغمرُ

ويصفِّقون على أکفّهمُ

فرحاً إذا ما أقبل العشرُ

جعلوه من أهنى مواسمهم

لا مرحباً بک أيّها الشهرُ

تلک الأناملُ من دمائِکمْ

يوم الطفوف خضيبةٌ حمرُ

فتوارثَ الهمجُ الخضابَ فمن

کفر تولّدَ ذلک الکفرُ

نبکي فيضحکُهمْ مصابکُمُ

وسرورُهمْ بمصابِکمْ نکرُف)

(١)في بعض النسخ : من طارقٍ يغشاهم خدر. (المؤلف)
