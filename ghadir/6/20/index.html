الغدير 6 ٢٠

إن لم أُبلّغ ما قد أُمرِتُ به

وکنتُ من خلقکم على حذرِ

وقال إن لم تفعلْ محوتُکَ من

حکمِ النبيِّين فاخشَ واعتبرِ

إن خفت من کيدِهم عصمتُکَ فاس

تبشر فإنّي لَخيرُ منتصرِ

أقمْ عليّا عليهمُ علماً

فقد تخيّرتُه من البشرِ

ثمَّ تلا آيةَ البلاغ لهم

والسمع يعنو لها مع البصرِ

وقال قد آن أن أُجيب إلى

داعي المنايا وقد مضى عمري

ألستُ أولى منکم بأنفسکم

قلنا بلى فاقضِ حاکماً ومُرِ

فقالَ والناسُ محدقون به

ما بين مصغٍ وبين منتظرِ

من کنتُ مولىً له فحيدرةٌ

مولاه يقفو به على أثري

يا ربّ فانصر من کان ناصرَهُ

واخذل عداه کخذلِ مقتدرِ

فقمت لمّا عرفتُ موضعَهُ

من ربِّه وهو خيرةُ الخيرِ

فقلت يا خيرةَ الأنامِ بخٍ

جاءتک منقادةً على قدرِ

أصبحتَ مولىً لنا وکنتَ أخاً

فافخر فقد حزتَ خيرَ مفتخرِ

ويقول فيها :

تالله ما ذنبُ من يقيسُ إلى

نعلک من قدّموا بمغتفرِ

أنکر قومٌ عيد الغدير وما

فيه على المؤمنين من نکرِ

حکّمک اللهُ في العبادِ به

وسرتَ فيهم بأحسنِ السِّيرِ

وأکملَ اللهُ فيه دينَهمُ

کما أتانا في محکمِ السُّورِ

نعتُکَ في محکمِ الکتابِ وفي

التوراةِ بادٍ والسّفر والزُّبُرِ

عليک عرضُ العباد تقضی على

من شئت منهم بالنفع والضررِ

تُظمئ قوماً عند الورود کما

تروي أُناساً بالوِردِ والصَدرِ

يا ملجأ الخائف اللهيفِ ويا

کنزَ الموالي وخير مدّخرِ
