الغدير 11 ٤٣٢

هذا نظامک أم درٌّ بمنتسقِ

أم الدراري التي لاحتْ على الأفقِ

وذا کلامُک أم سحرٌ به سُلِبَتْ

نهى العقول فتتلو سورةَ الفلقِ

وذا بيانُک أم صهباءُ شعشعَها

أغنُّ ذو مقلةٍ مکحولة الحدقِ

بتاج کلِّ مليکٍ منه لامعة

وجيد کلّ مجيدٍ منه في أنقِ

روضٌ من الزهر والأنوارُ زاهيةٌ

کأنجمِ الأُفق في اللألاءِ والنمقِ

وذي حمائمُ ألفاظٍ سجعنَ ضحىً

على الخمائلِ غبَّ العارضِ الغدقِ

رسالةٌ کفراديسِ الجنانِ بها

من کلّ مؤتلقٍ يلفى ومنتشقِ

کأنّما الألِفاتُ المائلاتُ بها

غصونُ بانٍ على أيدٍ من الورقِ

تعلو منابرَها الهمزاتُ صادحةً

کالوِرقِ ناحت على الأفنانِ من حرقِ

ميماتُها کثغورٍ يبتسمن بها

يزري على الدرِّ إذ يزهي على العنقِ

فطرسُها کبياض الصبح من يققٍ

ونِقسها ١ کسوادِ الليلِ في

غسقِ

يا ذا الرسالةِ قد أرسلتَ معجزةً

ودّت بلاغتها الدعوى من الفرقِ

ويا مليک ذوي الآدابِ قاطبةً

ويا إماماً هدانا أوضحَ الطرقِ

من ذا يعارض ما قد صاغ فکرک من

حلى البيانِ ومن يقفوک في السبقِ

أنت المجلّي بمضمارِ العلومِ إذا

أضحى قرومُ أولي التحقيقِ في قلقِ

صلّى أئمّةُ أهلِ الفضلِ خلفَکَ يا

مولى الموالي وربّ المنطق الذلقِ

مسلّمين لما قد حُزت من أدبٍ

مصدّقين بما شرّفت من خلقِ

مهلاً فباعي من التقصير في قصرٍ

وأنت في الطول والإحسانِ ذو عمقِ

سبحان بارئ هذي الذاتِ من هممٍ

سبحان فاطر ذا الإنسانِ من علقِ

يا ليت شعري هل شبهٌ يُرى لکمُ

کلاّ وربّي ولا الأملاک في الخلقِ

عذراً فما فکرتي صوّاغةٌ دُرراً

حتى أصوغَ لک الأسلاک في نسقِ

واسلم ودم وتعالى في مشيدِ عُلاً

تستنزل الشهبَ للإنشا فلم تعقِف)

(١)اليقق : القطن. نقس : المداد الذي يکتب به. (المؤلف)
