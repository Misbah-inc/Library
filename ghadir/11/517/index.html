الغدير 11 ٥١٧

إلى حيثُ لا ملکٌ سابقٌ

هناک ولا فلکٌ دائرُ

إذا ساجلَ الناسَ في رتبةٍ

فکلٌّ لدى عزِّه صاغرُ

وإن صالَ فالحتفُ من جندِه

وربُّ السماء له ناصرُ

کأنَّ قلوب العدا إن بدا

من الرعبِ يهفو بها طائرُ

أيا جدُّ إنَّ لسان البل

-يغِ عن حصرِ أوصافِکم قاصرُ

کفاکم عُلىً أنَّ ربَّ السما

ء في الذکر سعيَکم شاکرُ

فجاد ربوعَکَ من لطفِهِ

سَحابٌ برضوانِه ماطرُ

مدى الدهر ما قد طوى سبسباً

لتقبيل أعتابِکم زائرُ

ومن شعره قوله :

يا مخجلاً حَدَقَ المها

أوقعت قلبي بالمهالکْ

ومعيدَ صبحي کالمسا

ضاقت عليَّ به المسالکْ

يا مُنيتي دون الملا

أنحلتَ جسمي في ملالکْ

هبْ لي رقادي إنَّه

مذ بِنْتَ أبخلُ من خيالکْ

لله کم لک هالکٌ

بشبا اللواحظِ إثرَ هالکْ

يا موقفَ التوديعِ کم

دمعٍ نثرتُ على رمالکْ

هل لي مقيلٌ من ضلا

لي أم مقيلٌ في ظلالک

لهفي على عصر مضى

لي بالحبيبِ على تلالکْ

بالله أين غزالُک ال

-فتّانُ ويلي من غزالکْ

لم أنسَه ويدُ النوى

تستلُّ أنفسَنا هنالکْ

أومى يسائل کيف حا

لُک قلت داجي اللونِ حالکْ

فافترّ من عجبٍ وقا

ل بنو الهوی طرّا کذلکْ

فأجبته لو کنت تعلمُ

قدرَ من أصبحت مالکْ

لعلمتَ أنّي عاشقٌ

ما إن يقصّر عن منالکْ
