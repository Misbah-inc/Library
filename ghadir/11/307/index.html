الغدير 11 ٣٠٧

- 79 -

ابن أبي شافين البحراني

اشارة

المتوفّى بعد ⁦(1001)⁩

أجلُّ مصابي في الحياة وأکبرُ

مصابٌ له کلُّ المصائبِ تصغرُ

مصابٌ به الآفاقُ أظلم نورها

ووجه التقى والدين أشعث أغبرُ

مصابٌ به أطواد علمٍ تدکدکت

وأصبح نور الدين وهو مغبّرُ

إلى أن قال فيها :

وسار النبيُّ الطهرُ من أرضِ مکةٍ

وقد ضاق ذرعاً بالذي فيه أضمروا

ولمّا أتى نحو الغدير برحلِهِ

تلقّاه جبريلُ الأمينُ يبشِّرُ

بنصب عليٍّ والياً وخليفةً

فذلک وحيُ اللهِ لا يتأخّرُ

فردَّ من القوم الذين تقدّموا

وحطّ أُناسٌ رحلَهم قد تأخّروا

ولم يک تلک الأرضُ منزلَ راکبٍ

بحرِّ هجيرٍ نارُه تتسعّرُ

رقى منبرَ الأکوارِ طهرٌ مطهّرٌ

ويصدعُ بالأمرِ العظيمِ ويُنذرُ

فأثنى على اللهِ الکريمِ مقدّساً

وثنّى بمدحِ المرتضى وهو مخبرُ

بأن جاءني فيه من اللهِ عزمةٌ

وإن أنا لم أصدعْ فإنّي مقصّرُ

وإنّي على اسمِ الله قمتُ مبلّغاً

رسالتَه واللهُ للحقِّ ينصرُ

عليٌّ أخي في أُمّتي وخليفتي

وناصرُ دينِ اللهِ والحقُّ يُنصرُ

وطاعتُه فرضٌ على کلِّ مؤمنٍ

وعصيانُه الذنبُ الذي ليس يُغفرُ
