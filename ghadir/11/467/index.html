الغدير 11 ٤٦٧

فقم ففي مجلسِنا قد سعى

ساقٍ صغيرٍ بکؤوسٍ کبارْ

تقول عيناه لعشّاقه

من سيفِ أجفاني الحذارَ الحذارْ

واخفض جناحَ العيشِ في قهوةٍ

للهمّ عمّن قد حساها نفارْ

للروح روحٌ فإذا قرّبت

من حجرٍ حدّث صمّ الحجارْ

تطفئُ نارَ الهمِّ منّا وفي ال

کاسات منها مستطيراً شرار

إن قتلت منّا عقولاً فعن

والدها کان لها أخذ ثارْ

من کفّ ألمى ١ ما جلا حسنه

إلاّ وبان العقلُ واللبُّ طارْ

حمراء أعدى لونُها کأسَها

تخالُها من غير کاسٍ تدارْ

قوامُه يطعنُ طعنَ القنا

وفتک ماضي لحظه واقتدارْ

وردفه يشرح لي ثقله

وخصره يسند لي الاختصارْ

قد علّم الفتکَ أُسودَ الشرى

وعلّمَ الغزلانَ کيف النفارْ

عجبتُ من حمرةِ خدّيه إن

بدت لعينيّ علا في اصفرارْ

کأنَّما قد صِيغ من فضّةٍ

سالفة ٢ والخدُّ منّي نضارْ

لي روضةٌ غنّاءُ من وجهه

ولحظه ساقٍ وفيه عقارْ

خدٌّ وثغرٌ مقلةٌ وجنةٌ

وردٌ أقاحٍ نرجسٌ جلّنارْ

له على عشّاقه نصرةٌ

بفاترٍ منه أرى الإنکسارْ

في خدِّه ماءٌ ونارٌ وما

بالماء للنار عهدنا استعارْ

تثبتُ عيناي به لم تزلْ

فلم تحُلْ عنه يميناً يسارْ

کأنّما تلک له قربةٌ

قد عبدت ماء وهاتيک نارْ

يزري إذا ماس بغصنِ النقا

وإن بدا فالبدرُ منه يغارْ

فلو ترى يا لائمي حُسنَه

أقمت فيه حججَ الاعتذارْف)

(١)الألمی : الذي بشفته لمی. غلام ألمی : بارد الريق. (المؤلف)
(٢)السالفة : صفحة العنق عند معلق القرط. (المؤلف)
