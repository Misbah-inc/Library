الغدير 11 ٤٩٧

ولقد غدا صرف الزمان يُصدُّ عن

من نحوه أضحى مريد جوارِ

نعمٌ تعمُّ عمومَ هطّال الحيا

لکنّها جلّت عن الأضرارِ

وشمائلٌ کالروض لو لا أنّه

يذوي لفقد العارض المدرارِ

أقلامه قد قلّمت ما طال لل

أخطابِ والأخطارِ من أظفارِ

ودواته أدوت وداوت کاشحاً

ومؤمّلاً جدواه ذا إعسارِ

من آلِ خاقانَ الذين وجوهُهمْ

عند اسودادِ النقعِ کالأقمارِ

قومٌ إذا شاموا الصوارمَ أغمدتْ

في جيدِ کلِّ مملّکٍ کرّارِ

وإذا همُ اعتقلوا الذوابل في الوغي

آبتْ نواضرُ بالنجيعِ الجاري

أخبارُهمْ بسوادِ کلِّ دجنّةٍ

حُرِّرْنَ فوق بياضِ کلِّ نهارِ

يا من له بأسٌ يحاکي الصخرَ في

خُلُقٍ أرقَّ من النسيمِ الساري

وعُلاً تناسقَ کابراً عن کابرٍ

يحکي أنابيبَ القنا الخطّارِ

وافاک عيدُ النحر طلقاً وجهُهُ

يحکي رقيقُ نسيمِه أشعاري

عيدٌ يعود عليکمُ بمسرّةٍ

محمودةِ الإيرادِ والإصدارِ

لا زالت الأيدي تشيرُ إليکمُ

شبهَ الهلالِ عشيّة الإفطارِ

وبقيت ترفل من علاک بحلّةٍ

فضفاضةٍ قد طرّزت بفخارِ

وله مراسلاً إيّاه لازماً الجناس المذيّل قوله :

لعمرک إنّ دمعَ العين جارٍ

لأنّي حنظل التفريق جارعْ

وما لي غير شهد الوصلِ شافٍ

فهل لي في اجتناءٍ منه شافعْ

وقلبي للوصولِ إليک صادٍ

ونظمي بالثناءِ عليک صادعْ

وهمّي ليثُه الفتّاکُ ضارٍ

ولولاه لما أمسيتُ ضارعْ

ولوني أصفرٌ والدمعُ قانٍ

وطرفي منکمُ بالطيف قانعْ

ومذ غبتم فصبحي شبهُ قارٍ

لديَّ وإصبعي للسنِّ قارعْ
