الغدير 7 ٦٤

يا من تجلّى بالجمالِ

فشقَّ بردةَ کلِّ حالکْ

صلّى عليکَ اللهُ من

هادٍ إلى خيرِ المسالکْ

والحافظ البرسيّ لا

يخشى وأنت له هنالکْ

وله أبيات في أهل البيت خمّسها الشاعر المفلق الشيخ أحمد بن الحسن النحوي نذکرها مع تخميسها :

ولائي لآلِ المصطفى وبنيهمُ

وعترتهم أزکى الورى وذَويهمُ

بهم سمةٌ من جدِّهم وأبيهمُ

همُ القومُ أنوارُ النبوّةِ فيهمُ

تلوحُ وآثارُ الإمامةِ تُلمَحُ

نجومُ سماءِ المجدِ أقمارُ تمِّهِ

معالمُ دينِ اللهِ أطوادُ حلمهِ

منازلُ ذکرِ اللهِ حکّامُ حکمهِ

مهابطُ وحيِ اللهِ خزّانُ علمهِ

وعندهمُ سرُّ المهيمن مودعُ

مديحُهمُ في محکمِ الذکرِ محکمُ

وعندَهمُ ما قد تلقّاه آدمُ

فدَع حکمَ باقي الناسِ فهو تحکّمُ

إذا جلسوا للحکم فالکلُّ أبکمُ

وإن نطقوا فالدهرُ أُذنٌ ومسمعُ

بحبِّهمُ طاعاتُنا تُتَقبّلُ

وفي فضلِهمْ جاءَ الکتابُ المنزَّلُ

يعمُّ شذاهمُ کلَّ أرضٍ ويشملُ

وإن ذکروا فالکون نَدٌّ ومندلُ ١

لهم أرجٌ من طيبهم يتضوعُ

دعا بهمُ موسى ففُرِّجَ کربُهُ

وکلّمه من جانب الطور ربُّهُ

إذا حاولوا أمراً تسهّلَ صعبُهُ

وإن برزوا فالدهرُ يخفقُ قلبُهُ

لسطوتِهمْ والأُسدُ في الغابِ تفزعُ

فلولاهمُ ما سار فلکٌ ولا جرى

ولا ذرأَ اللهُ الأنامَ ولا براف)

(١)الندّ بفتح المعجمة وکسرها : عود يُتبخّر به. المندل : العود الطيّب الرائحة. (المؤلف)
