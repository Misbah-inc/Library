الغدير 7 ١١

- 72 -

ابن العرندس الحلّي

أضحى يميسُ کغصنِ بانٍ في حُلى

قمرٌ إذا ما مرَّ في قلبي حلا

سلبَ العقولَ بناظرٍ في فترةٍ

فيها حرامُ السحرِ بان محلّلا

وانحلَّ شدُّ عزائمي لمَّا غدا

عن خصرِهِ بندُ القباءِ محلّلا

وزها بها کافورُ سالفِ خدِّه

لمّا بريحانِ العذارِ تسلسلا

وتسلسلتْ عبثاً سلاسلُ صدغِه

فلذاک بتُّ مقيّداً ومسلسلا

قمرٌ قويمُ قوامِهِ کقناتِهِ

ولحاظُه في القتلِ تحکي المنصلا

وجناتُه جوريَّةٌ وعيونُه

حوريّةٌ تسبي الغزالَ الأکحلا

أهوى فواترَها المراضَ إذا رنتْ

وأحبُّ جفنيها المراضَ الغُزَّلا

جارت وما صفحت على عشّاقه

فتکاً وعاملُ قدِّه ما أعدلا

ملکت محاسنُهُ ملوکاً طالما

أضحى لها الملکُ العزيزُ مذلّلا

کِسرى بعينيه الصحاحِ وخدُّه

النعمانُ بالخال النجاشي خوّلا

کتب العليُّ على صحائفِ خدِّهِ

نوني قسيِّ الحاجبين ومَثّلا

فرمى بها في عينِ غنجِ عيونِهِ

سهمَ السهامِ أصابَ منِّي المقتلا

فاعجب لعينِ عبيرِ عنبرِ خالِهِ

في جيمِ جمرةِ خدِّه لن تشعلا

وسَلا الفؤادَ بحرِّ نيرانِ الجوى

منِّي فذاب وعن هواهُ ما سلا

فمتى بشيرُ الوصلِ يأتي منجحاً

وأبيتُ مسروراً سعيداً مُقبلا
