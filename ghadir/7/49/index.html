الغدير 7 ٤٩

- 74 -

الحافظ البرسي الحلّي

هو الشمسُ أم نورُ الضريح يلوحُ

هو المسکُ أم طيبُ الوصيِّ يفوحُ

وبحرُ ندى أم روضةٌ حوتِ الهدى

وآدمُ أم سرُّ المهيمنِ نوحُ

وداودُ هذا أم سليمانُ بعدَه

وهارونُ أم موسى العصا ومسيحُ

وأحمدُ هذا المصطفى أم وصيُّه

عليٌّ نماه هاشمٌ وذبيحُ

محيطُ سماءِ المجدِ بدرُ دجنّةٍ

وفلکُ جمالٍ للأنام ويوحُ ١

حبيبُ حبيبِ اللهِ بل سرُّ سرِّه

وجثمانُ أمرٍ للخلائقِ روحُ

له النصُّ في يومِ الغديرِ ومدحُه

من اللهِ في الذکرِ المبينِ صريحُ

إمامٌ إذا ما المرءُ جاءَ بحبِّه

فميزانُه يومَ المعادِ رجيحُ

له شيعةٌ مثلُ النجومِ زواهرٌ

لها بين کلِّ العالمينَ وضوحُ

إذا قاولتَ فالحقُّ فيما تقولهُ

به النورُ بادٍ واللسانُ فصيحُ

وإن جاولتَ أو جادلتَ عن مرامِها

تولّى العدوُّ الجَلْدُ وهو طريحُ

عليک سلامُ اللهِ يا رايةَ الهدى

سلامُ سليمٍ يغتدي ويروحُ

وتأتي له قصيدة منها قوله :

مولىً له بغديرِ خمِّ بيعةٌ

خضعتْ لها الأعناق وهي طوائحُف)

(١)يوح : الشمس. (المؤلف)
