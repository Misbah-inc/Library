الغدير 4 ٢٩٥

هذي المنازلُ بالغميمِ فنادِها

واسکبْ سخيَّ العينِ بعد جمادِها

إن کان دَيْنٌ للمعالمِ فاقضِه

أو مهجةٌ عند الطلولِ ففادِها

ياهل تَبُلُّ من الغليلِ إليهمُ

إشرافةٌ للرکبِ فوق نجادِها

نُؤیٌ کمنعطف الحنيّةِ دونَهُ

سُحُم الخدودِ لهنّ إرثُ رمادِها

ومناطُ أطنابٍ ومقعدُ فِتْيةٍ

تخبو زنادُ الحَيّ غيرَ زنادِها

ومَجَرُّ أرسانِ الجيادِ لِغلمةٍ

سجفوا البيوتَ بشقرِها وورادِها

ولقد حبستُ على الديارِ عصابةً

مضمومةَ الأيدي إلى أکبادِها

حسرى تجاوبُ بالبکاءِ عيونُها

وتعطُّ بالزفراتِ في أبرادِها

وقفوا بها حتى کأنّ مطيَّهمْ

کانت قوائمُهنّ من أوتادِها

ثمّ انثنتْ والدمعُ ماءُ مزادِها

ولواعجُ الأشجانِ من أزوادِها

من کلِّ مشتملٍ حمائِلَ رنّةٍ

قطْرُ المدامعِ من حُليِّ نجادِها

حيّتکَ بل حيّتْ طلوعَکَ ديمةٌ

يشفي سقيمَ الربعِ نفثُ عِهادِها

وغدتْ عليک من الخمائل يَمنةٌ

تستامُ نافقةً على روّادِها ١

هل تطلبونَ من النواظرِ بعدَکم

شيئاً سوى عبراتِها وسُهادِها

لم يبقَ ذُخرٌ للمدامعِ عنکمُ

کلاّ ولا عينٌ جرى لرقادِها

شغلَ الدموعَ عن الديارِ بکاؤنا

لبکاءِ فاطمةٍ على أولادِها

لم يخلُفوها في الشهيدِ وقد رأى

دُفَعَ الفراتِ يُذادُ عن أورادِها ٢

أَتُرى دَرَتْ أنّ الحسينَ طريدةٌ

لِقَنا بني الطرداءِ عند ولادِها

کانت مآتمُ بالعراقِ تعدُّها

أمويَّةٌ بالشام من أعيادِها

ما راقبتْ غضبَ النبيِّ وقد غدا

زرعُ النبيِّ مظنّةً لحصادِها

باعتْ بصائرَ دينِها بضلالِها

وشَرَتْ معاطبَ غيِّها برشادِهات.

(١)الخمائل - جمع خميلة - : القطيفة. اليمنة : بُرد يمنى. تستام : تسأل السوم. (المؤلف).
(٢)دُفَع - جمع دفعة - : دفقة المطر ، استعارها للفرات.
