الغدير 4 ٤١٩

43 - الجبري المصري

يا دارُ غادرني جديدُ بِلاکِ

رثَّ الجديدُ فهل رثيت لذاکِ

أم أنت عمّا أشتکيه من الهوى

عجماءُ مذ عجم البلى مغناکِ

ضفناک نستقري الرسومَ فلم نجدْ

إلاّ تباريحَ الهموم قِراکِ

ورسيسَ شوقٍ تمتري زفراتُه

عبراتِنا حتى تبلّ ثراکِ

ما بالُ ربعکِ لا يبلُّ کأنّما

يشکو الذي أنا من نحوليَ شاکِ

طلّت طلولُکِ دمعَ عيني مثلَما

سفکت دمي يومَ الرحيل دُماکِ

وأرى قتيلَک لا يَديهِ قاتلٌ

وفتورَ ألحاظِ الظباء ظُباکِ

هيّجت لي إذ عجتُ ساکنَ لوعةٍ

بالساکنيک تشبّهاً ذکراکِ

لمّا وقفتُ مسلّماً وکأنّما

ريّا الأحبّةِ سفت من رياکِ ١

وکفت عليکِ سماءُ عيني صيِّباً

لو کفَّ صوبُ المزنِ عنک کفاکِ

سقياً لعهدي والهوى مقضيّةٌ

أوطارُهُ قبل احتکامِ نواکِ

والعيش غضٌّ والشبابُ مطيّةٌ

للهوِ غيرُ بطيئةِ الإدراکِ

أيّام لا واشٍ يُطاعُ ولا هوىً

يُعصى فنقصى عنکِ إذْ زرناکِ

وشفيعنا شرخُ الشبيبةِ کلّما

رُمنا القصاصَ من اقتناصِ مهاکِا.

(١)کذا.
