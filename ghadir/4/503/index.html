الغدير 4 ٥٠٣

هو العالمُ الحَبرُ الذي ليس مثلُهُ

هو البطلُ القرمُ الهزبرُ الغشمشمُ

وما زال في بدرٍ وأُحدٍ وخيبرٍ

يفلُّ جيوشَ المشرکين ويحطمُ

يکرُّ ويعلوهم بقائمِ سيفِهِ

إلى أن أطاعوا مکرهين وأسلموا

وما دخلوا الإسلام ديناً وإنّما

منافقةً کي يُرفعَ السيفُ عنهمُ

وقالوا عليٌّ کان في الحکمِ ظالماً

ليکثر بالدعوى عليه التظلّمُ

وقالوا دماءُ المسلمين أراقَها

وقد کان في القتلى بريءٌ ومجرمُ

فقلتُ لهم مهلاً عدمتمْ صوابَکمْ

وصيُّ النبيّ المصطفى کيف يظلِمُ

أراقَ دماءَ المسلمين فوالذي

هدانا به ما کان في القومِ مسلمُ

ولکنّه للناکثين بعهدِه

وممّن تعدّى منهم کان ينقمُ

أما قال أقضاکم عليٌّ محمدٌ

کذا قد رواه الناقدُ المتقدّمُ

فإن جار ظلماً في القضايا بزعمِکمْ

عليٌّ فمن زکّاهُ لا شکّ أظلمُ

فيا ليتني قد کنت بالأمسِ حاضراً

فأُشرِکهُ في قتلِهمْ وأُصمّمُ

وألقى إلهي دونهمْ بدمائِهمْ

فننظرُ عند اللهِ من يتندّمُ

فمن کعليٍّ عند کلّ ملمّةٍ

إذا ما التقى الجمعانِ والنقعُ مفعمُ

ومَن ذا يُساميه بعلمٍ ولم يَزَلْ

يقول سلوني ما يحلُّ ويحرمُ

سلوني ففي جنبيَّ علمٌ ورثتُهُ

عن المصطفى ما فاه منّي به الفمُ

سلونيَ عن طرقِ السموات إنّني

بها من سلوکِ الأرضِ والطرقِ أعلمُ

ولو کشفَ اللهُ الغطا لم أزِدْ به

يقيناً على ما کنت أدري وأعلمُ

وکأين له من آيةٍ وفضيلةٍ

ومن مکرماتٍ ما تعمُّ وتکتمُ

فمن ختمتْ أعمالُهُ عند موتِهِ

بخيرٍ فأعمالي بحبِّيه تُختَمُ

فيا ربّ بالأشباحِ آلِ محمدٍ

نجومِ الهدى للناسِ والأفق مظلمُ

وبالقائمِ المهديِّ من آل أحمدٍ

وآبائِه الهادين والحقُّ معصمُ

تفضّل على العوديّ منک برحمةٍ

فأنت إذا استُرحمتَ تعفو وترحمُ
