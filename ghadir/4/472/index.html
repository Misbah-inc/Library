الغدير 4 ٤٧٢

ولا تغترر منهمْ بحسنِ بشاشةٍ

فأکثرُ إيماضِ البوارقِ خُلّبُ

وأصغِ إلى ما قلتُهُ تنتفعْ به

ولا تطّرحْ نصحي فإنّي مجرِّبُ

فما تنکُر الأيّامُ معرفتي بها

ولا إنّني أدرى بهنّ وأدربُ

وإنّي لأقوامٍ جذيلٌ محکّکٌ

وإنّي لأقوامٍ عُذيقٌ مرجّبُ

عليمٌ بما ترضى المروءةُ والتقى

خبيرٌ بما آتي وما أتجنّبُ

حلبتُ أفاويقَ الزمانِ براحةٍ

تدرُّ بها أخلافُه حين تحلبُ

وصاحبت هذا الدهَر حتى لقد غدتْ

عجائبُه من خبرتي تتعجّبُ

ودوّختُ أقطارَ البلادِ کأنّني

إلى الريح أُعزي أو إلى الخضر أُنسبُ

وعاشرتُ أقواماً يزيدون کثرةً

على الألف أو عدّ الحصى حين يحسبُ

فما راقني في روضهم قطُّ مرتعٌ

ولا شاقني في وِردِهم قطُّ مشربُ

تراني وإيّاهمْ فريقين کلّنا

بما عنده من عزّة النفسِ معجبُ

فعندهمُ دنيا وعندي فضيلةٌ

ولا شکَّ أنّ الفضل أعلى وأغلبُ

على أنّ ما عندي يدومُ بقاؤه

عليَّ ويفني المالُ عنهمْ ويذهبُ

أُناسٌ مضى صدرٌ من العمرِ عندَهمْ

أُصعّدُ ظنّي فيهمُ وأُصوِّبُ

رجوتُ بهم نيلَ الغني فوجدتُهُ

کما قيلَ في الأمثالِ عنقاء مغربُ

وکسّل عزمَ المدحِ بعد نشاطِهِ

ندى ذمِّه عندي من المدحِ أوجبُ

کأنّ القوافي حين تُدعى لشکرِهمْ

على الجمرِ تمشي أو على الشوکِ تُسحبُ

أفوه بحقٍّ کلّما رمتُ ذمَّهمْ

وما غيرُ قولِ الحقِّ لي قطُّ مذهبُ

وأصدق إلاّ أن اريد مديحَهُمْ

فإنّي على حکمِ الضرورةِ أکذبُ

ولو علموا صدقَ المدائحِ فيهمُ

لکانتْ مساعيهمْ تهشُّ وتطربُ

ولکن دروا أنّ الذي جاء مادحاً

بغيرِ الذي فيهم يسبّ ويثلبُ

وما زال هذا الأمرُ دأبي ودأبَهم

أغالب لومي فيهمُ وهو أغلبُ

إلى أن أذلّتني الليالي وأعتبتْ

وما خلتُها بعد الإساءةِ تعتبُ

فهاجرتُ نحو الصالحِ الملکِ هجرةً

غدتْ سبباً للأمن وهو المسبّبُ
