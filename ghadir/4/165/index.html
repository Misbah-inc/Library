الغدير 4 ١٦٥

من به استکفى أذى ال

أيّام والدهرَ کفاهُ

کيف لا أمدحُ من لم

يخلُ خلقٌ من نداهُ

ومن غرر محاسنه قوله يمدح من قصيدة ، أوّلها :

قد سمعنا مقالَه واعتذاره

وأقلناه ذنبَهُ وعِثاره

والمعاني لمن عنيتُ ولکن

بک عرّضتُ فاسمعي يا جاره

من مراديه أنّه أبد الده

-ر تراه محلّلاً أزراره

عالمٌ أنّه عذابٌ من اللّ

-ه مباحٌ لأعينِ النظّاره

هتکَ اللهُ سترَهُ فَلَکَم ههت

تّک من ذي تستّرٍ أستار

سحرتني ألحاظُه وکذا ک

-لُّ مليحٍ ألحاظُه سحّاره

ما على مؤثر التباعدِ والإ

عراضِ لو آثرَ الرضا والزياره

وعلى أنّني وإن کان قد عذّ

بَ بالهجرِ مؤثرٌ إيثاره

لم أزل لا عَدمتُه من حبيبٍ

أشتهي قربه وآبى نفاره

يقول في مدحها :

لم يدَعْ للعزيزِ في سائرِ الأر

ض عدوّا إلاّ وأخمدَ ناره

فلهذا اجتباه دون سوا

هُ واصطفاه لنفسِهِ واختاره

لم تشيِّد له الوزارةُ مجداً

لا ولا قيل رفّعَتْ مقداره

بل کساها وقد تخرّمها الده

-رُ جلالاً وبهجةً ونضاره

کلَّ يومٍ له على نُوَب الده

-ر وکرِّ الخطوبِ بالبذلِ غاره

ذو يدٍ شأنُها الفرارُ من البُخ

-لِ وفي حومةِ الوغى کرّاره

هيَ فلّتْ عن العزيز عِداه

بالعطايا وکثّرتْ أنصاره

هکذا کلُّ فاضلٍ يدُه تُم

-سی وتُضحي نفّاعةً ضرّاره

فاستجرْهُ فليس يأمنُ إلاّ

من تَفَيّا بظلّه واستجاره
