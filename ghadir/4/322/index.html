الغدير 4 ٣٢٢

وطروسي سودٌ فکيف بيَ الآ

نَ ومنکمْ بياضُها والصقالُ

حبّکم کان فکَّ أسري من الشر

کِ وفي منکبي له أغلالُ

کم تزمّلتُ بالمذلّة حتى

قمتُ في ثوبِ عزِّکمْ أختالُ

برکاتٌ لکم محت من فؤادي

ما أملَّ الضلالَ عَمٌّ وخالُ

ولقد کنتُ عالماً أنّ إقبا

لي بمدحي عليکمُ إقبالُ

- 3 -

وله من قصيدة يرثي بها أهل البيت عليهم السلام وهي ⁦(63)⁩ بيتاً ، توجد في ديوانه ⁦(4 / 198)⁩ مطلعها :

لو کنتُ دانيتُ المودّة قاصيا

ردَّ الحبائبُ يوم بِنَّ فؤاديا

إلى أن قال :

وبحيِّ آلِ محمدٍ إطراؤه

مدحاً وميِّتهمْ رضاه مراثيا

هذا لهمْ والقومُ لا قومي هُمُ

جنساً وعُقرُ ديارِهمْ لا داريا

إلاّ المحبّةَ فالکريمُ بطبعِهِ

يجدُ الکرامَ الأبعدينَ أدانيا

يا طالبيّين اشتفى من دائِهِ ال

-مجدُ الذي عدِمَ الدواءَ الشافيا

بالضاربين قبابَهمْ عَرض الفلا

عقل الرکائبِ ذاهباً أو جائيا

شرعوا المحجّةَ للرشادِ وأرخصوا

ما کان من ثمنِ البصائرِ غاليا

وأما وسيِّدِهمْ عليٍّ قولةٌ

تُشجي العدوَّ وتُبهجُ المتواليا

لقد ابتنى شرفاً لهم لو رامه

زُحلٌ بباعٍ کان عنهُ عالياً

وأفادهم رقَّ الأنامِ بوقفةٍ

في الروعِ بات بها عليهمْ واليا

ما استدرک الإنکارَ منهم ساخطٌ

إلاّ وکان بها هنالک راضياً

أضحوا أصادقَهُ فلمّا سادهمْ

حسدوا فأمسَوا نادمين أعاديا

فارحمْ عدوَّکَ ما أفادک ظاهراً

نصحاً وعالجَ فيکَ خِلاّ خافياً
