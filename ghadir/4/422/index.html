الغدير 4 ٤٢٢

ولَتُسأَلِنَّ عن الولاءِ لحيدرٍ

وهو النعيمُ شقاکِ عنه ثناکِ ١

قست المحيطَ بکلِّ علمٍ مشکلٍ

وعرٍ مسالکُه على السُلاّکِ

بالمعتريهِ کما حکي شيطانُهُ

وکفاهُ عنه بنفسِهِ من حاکِ

والضاربَ الهاماتِ في يومِ الوغى

ضرباً يقدُّ به إلى الأوراکِ

إذ صاح جبريلٌ به متعجِّباً

من بأسِهِ وحسامِهِ البتّاکِ

لا سيفَ إلاّ ذو الفقارِ ولا فتى

إلاّ عليٌّ فاتکُ الفتّاکِ

بالهاربِ الفرّارِ من أقرانِهِ

والحربُ يذکيها قناً ومذاکِ

والقاطع الليلِ البهيمِ تهجّداً

بفؤادِ ذي روعٍ وطرفٍ باکِ

بالتارکِ الصلواتِ کفراناً بها

لو لا الرياءٌ لطالَ ما راباکِ

أبعدْ بهذا من قياسٍ فاسدٍ

لم تأت فيه أُمّةٌ مأتاکِ

أوَما شهدتِ له مواقفَ أذهبتْ

عنکِ اعتراکَ الشکِّ حين عراکِ

من معجزاتٍ لا يقومُ بمثلِها

إلاّ نبيٌّ أو وصيٌّ زاکي

کالشمسِ إذ رُدّت عليه ببابلٍ

لقضاء فرضٍ فائتِ الإدراکِ

والريحِ إذ مرّتْ فقال لها احملي

طوعاً وليَّ اللهِ فوق قواکِ

فجرتْ رجاءً بالبساط مطيعةً

أمرَ الإله حثيثةَ الإيشاکِ ٢

حتى إذا وافى الرقيمَ بصحبِهِ

ليزيلَ عنه مِريةَ الشکّاکِ

قال السلام عليکمُ فتبادروا

بالردِّ بعد الصمت والإمساکِ

عن غيرِهِ فبدتْ ضغائنُ صدرِ ذي

حنقٍ لسترِ نفاقِهِ هتّاکِ

والميتُ حين دعا به من صرصرٍ

فأجابَهُ وأبيتِ حين دعاکِ

لا تدّعي ما ليس فيکِ فتندمي

عند امتحان الصدقِ من دعواکِف)

(١)ثناکِ عنه شقاکِ. کذا في نسخة. (المؤلف)
(٢)وفی نسخة : فغدتْ رخاءً بالبساطِ مطيعةً أمرَ الإله حثيثةَ الإدراکِ (المؤلف)
