الغدير 4 ٥٥٥

يا عاذلي في هوى أبناءِ فاطمةٍ

لک الملامةُ إن قصّرتَ في عذلي

بالله دُر ساحة القصرين وابک معي

عليهما لا على صفّينَ والجملِ

وقل لأهليهما والله ما التحمتْ

فيکم جراحي ولا قرحي بمندملِ

وما ذا عسى کانت الإفرنجُ فاعلةً

في نسلِ آلِ أميرِ المؤمنين علي

هل کان في الأمرِ شيءٌ غير قسمةِ ما

ملکتمُ بين حکمِ السبي والنقلِ

وقد حصلتمْ عليها واسمُ جدّکمُ

محمدٌ وأبوکم غيرُ منتقلِ

مررتُ بالقصرِ والأرکانُ خاليةٌ

من الوفودِ وکانت قبلةَ القبلِ

فملتُ عنها بوجهي خوفَ منتقدٍ

من الأعادي ووجه الودِّ لم يملِ

أسلتُ من أسفي دمعي غداةَ خلتْ

رحابُکمْ وغدتْ مهجورةَ السبلِ

أبکي على ما تراءتْ من مکارمِکمْ

حالَ الزمانُ عليها وهي لم تحلِ

دارُ الضيافةِ کانت أُنسَ وافدِکمْ

واليومَ أوحشُ من رسمٍ ومن طللِ

وفطرةُ الصومِ إذ أضحتْ مکارمُکمْ

تشکو من الدهرِ حيفاً غيرَ محتملِ

وکسوةُ الناسِ في الفصلين قد درستْ

ورثَّ منها جديدٌ عندهم وبلِي

وموسمٌ کان في يومِ الخليجِ لکم

يأتي تجمّلُکمْ فيه على الجملِ

وأوّلُ العامِ والعيدين کم لکمُ

فيهنّ من وَبلِ جودٍ ليس بالوشلِ

والأرض تهتزُّ في يومِ الغديرِ کما

يهتزُّ ما بين قصريکم من الأسلِ

والخيلُ تُعرَضُ في وشيٍ وفي شيةٍ

مثلَ العرائسِ في حُليٍ وفي حللِ

ولا حملتم قِرى الأضيافِ من سعةٍ ال

أطباقِ إلاّ على الأکتاف والعجلِ

وما خصصتمْ ببرٍّ أهل ملّتِکمْ

حتى عممتمْ به الأقصى من المللِ

کانت رواتبُکمْ للذمّتين ولل

-ضيفِ المقيمِ وللطاري من الرسلِ

ثمّ الطراز بتنّيس الذي عظمتْ

منه الصلاتُ لأهلِ الأرضِ والدولِ

وللجوامعِ من إحسانِکمْ نعمٌ

لمن تصدّر في علمٍ وفي عملِ

وربّما عادت الدنيا فمعقلُها

منکم وأضحتْ بکم محلولةَ العَقَلِ
