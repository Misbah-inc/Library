16 - ابکِ ما عشت بالدموعِ الغزارِ

لذراري محمد المختارِ 37

17 - أآمرتي بالصبرِ أسرفتِ في أمري

أيُؤمرُ مثلي لا أبا لک بالصبرِ 29

18 - سلامي على قبرٍ تضمّن حيدرا

سلام مشوقٍ ما يطيقُ التصبّرا 60

ويقول في آخرها :

ولا أغلُ في ديني کمن کان قد غلا

وما کنتُ في حبِّ الوصيّ مقصِّرا

بذلک يلقى الله في يومِ بعثِه

عليُّ بن حمّادٍ إذا هو أُنشرا

19 - يالائمی دع ملامي في الهوى وذرِ

فإنّ حبَّ عليٍّ قام في عذري 28

20 - دعا قلبَهُ داعي الوعيدِ فأسمعا

وداعٍ لبادي شيبِهِ فتورّعا 62

21 - فرّقتَ يا بينُ شملاً کان مجتمعا

أبعدتَ عنّي حبيبي والسرورَ معا 77

22 - خليلي عُجْ بنا نُطِلِ الوقوفا

على مَن نورُهُ شملَ الطفوفا 25

23 - خواطرُ فکري فيِ الحشاءِ تجولُ

24 - أهجرتِ يا ذاتَ الجمالِ دلالا

25 - ألا إنّ زينَ المرءِ في عمرِهِ العقلُ

26 - يا عليَّ بن أبي طالب يا ابنَ المفضلِ

27 - ناجتْکَ أعلامُ الهدايةِ فاعلمِ

28 - النومُ بعدَکمُ عليّ حرامُ

29 - أرضِ الإلهَ وأسخطِ الشيطانا

تُعْطَ الرضا في الحشر والرضوانا 27

يقول فيها وهي ناقصة الآخر :

من أنزلَ اللهُ الکتابَ عليه في

کلِّ العلوم ليفتدی برهانا

من بلّغ الدنيا بنصبِ وصيِّه

يومَ الغدير ليکملَ الإيمانا
