الغدير 4 ٢٩٤

أقودُ إليهما نفسي وأهدي

سلاماً لا يحيدُ عن الجوابِ

لقاؤهما يطهِّر من جناني

ويدرأُ عن ردائي کلَّ عابِ

قسيمُ النارِ جدّي يوم يلقى ١

به باب النجاة من العذابِ

وساقي الخلقِ والمهجاتُ حرّى

وفاتحةُ الصراطِ إلى الحسابِ

ومن سمحتْ بخاتمِهِ يمينٌ ٢

تضِنُّ بکلّ عاليةِ الکعابِ

أما في بابِ خيبرَ معجزاتٌ

تُصَدَّقُ أو مناجاةُ الحُبابِ ٣

أرادت کيدَهُ والله يأبى

فجاءَ النصرُ من قِبَلِ الغرابِ ٤

أهذا البدرُ يُکسَفُ بالدياجي

وهذي الشمسُ تُطمَسُ بالضبابِ

وکان إذا استطال عليه جانٍ

يرى ترکَ العقابِ من العقابِ

أرى شعبانَ يُذْکِرُني اشتياقي

فمن لي أن يذکّرکم ثوابي

بکم في الشعرِ فخري لا بشعري

وعنکم طالَ باعي في الخطابِ

أُجَلّ عن القبائحِ غير أنّي

لکم أرمي وأُرمى بالسبابِ

فأجهرُ بالولاءِ ولا أُوَرّي

وأنطقُ بالبراءِ ولا أحابي

ومن أولى بکم منّي وليّا

وفي أيديکمُ طَرَفُ انتسابي

محبّکمُ ولو بغضتْ حياتي

وزائرُکمْ ولو عقرتْ رکابي

تُباعِدُ بيننا غِيَرُ الليالي

ومرجِعُنا إلى النسب القرابِ ٥

وقال يرثي الإمام السبط المفدّى الحسين بن عليّ عليهما السلام في يوم عاشوراء سنة ⁦(391)⁩: 3.

(١)أشار إلى حديث مرّ بيانه في : 3 / 299. (المؤلف)
(٢)أشار إلى تصدّقه بخاتمه ، وقد مرّ حديثه : 2 / 47 و 3 / 155 - 162. (المؤلف)
(٣)الحُباب : الحيّة.
(٤)أشار إلى حديث الحُباب الذي أسلفناه : 2 / 241 - 242. (المؤلف)
(٥)ديوان الشريف الرضي : 1 / 113.
