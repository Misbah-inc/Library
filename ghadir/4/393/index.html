الغدير 4 ٣٩٣

ودمعٌ إذا حبّستَهُ عن سبيلِهِ

يعود هتوناً في الجفون هطولا

فيا ليتَ أسرابَ الدموعِ التي جرت

أَسَوْنَ کليماً أو شَفَيْنَ غليلا

أُخالُ صحيحاً کلَّ يومٍ وليلةٍ

ويأبى الجوى ألاّ أکونَ عليلا

کأنّي وما أحببت أهوى ممنّعاً

وأرجو ضنيناً بالوصال بخيلا

فقل للذي يبکي نُؤيّا ودمنةً

ويندب رسماً بالعراء محيلا

عداني دمٌ لي طُلَّ بالطفِّ أن أُرى

شجيّا أُبکّي أربُعاً وطلولا

مصابٌ إذا قابلت بالصبرِ غربَهُ

وجدت کثيري في العزاءِ قليلا

ورزءٌ حملتُ الثقلَ منه کأنّني

مدى الدهر لم أحمل سواه ثقيلا

وجدتم عداةَ الدينِ بعد محمدٍ

إلى کلمِهِ في الأقربين سبيلا

کأنّکمُ لم تنزعوا بمکانِهِ

خشوعاً مبيناً في الورى وخمولا

وأيّکمُ ما عزّ فينا بدينِهِ

وقد عاش دهراً قبل ذاک ذليلا

فقل لبني حربٍ وآلِ أُميّةٍ

إذا کنتَ ترضى أن تکون قؤولا

سللتمْ على آل النبيّ سيوفَهُ

مُلئِنَ ثلوماً في الطلى وفلو لا

وقدتم إلی من قادکم من ضلالِکمْ

فأخرجَکمْ من وادييه خيولا

ولم تغدروا إلاّ بمن کان جدُّهُ

إليکم لتحظَوا بالنجاةِ رسولا

وترضون ضدَ الحزمِ إن کان ملککُمْ

ضئيلاً وديناً دنتمُ لهزيلا

نساءُ رسولِ اللهِ عُقْرَ ديارِکمْ

يرجِّعنَ منکم لوعةً وعويلا

لهنّ ببوغاءِ الطفوفِ أعزّةٌ

سُقوا الموتَ صِرفاً صبيةً وکهولا

کأنّهمُ نوّار روضٍ هوتْ به

رياحٌ جَنوباً تارةً وقَبولا

وأنجمُ ليلٍ ما علونَ طوالعاً

لأعينِنا حتى هبطنَ أُفولا

فأيّ بدورٍ ما مُحين بکاسفٍ

وأيّ غصونٍ ما لقينَ ذبولا

أمن بعد أن أعطيتموه عهودَکمْ

خِفافاً إلى تلک العهودِ عُجولا

رجعتمْ عن القصدِ المبينِ تناکصا

وحُلتمْ عن الحقِّ المنيرِ حؤولا
