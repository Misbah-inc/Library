الغدير 4 ٣٨٩

وکم أراکم بأجوازِ الفلا جُزُراً

مبدّدين ولکن أيَّ تبديدِ

لو کان ينصفُکمْ من ليس ينصفُکمْ

ألقى إليکم مطيعاً بالمقاليدِ

حُسدتمُ الفضلَ لم يُحرِزْهُ غيرُکمُ

والناسُ ما بين محرومٍ ومحسودِ

جاءوا إليکم وقد أَعطَوا عهودَهمُ

في فيلقٍ کزُهاء الليلِ ممدودِ

مُستمرحينَ بأيديهمْ وأرجلِهمْ

کما يشاءون رکضَ الضمّرِ القودِ

تهوي بهم کلُّ جرداءٍ مطهّمةٍ

هويّ سَجْلٍ من الأوذام مجدودِ ١

مستشعرين لأطرافِ الرماحِ ومن

حدِّ الظبا أدرُعاً من نسجِ داودِ

کأنّ أصواتَ ضربِ الهامِ بينهمُ

أصواتُ دوحٍ بأيدي الريحِ مبدودِ

حمائمُ الأيکِ تبکيهمْ على فننٍ

مرنّحٍ بنسيمِ الريحِ أُملودِ ٢

نوحِي فذاک هديرٌ منکِ محتسبٌ

على حسينٍ فتعديدٌ کتغريدِ

أُحبّکمْ والذي طافَ الحجيجُ به

بمبتنى بإزاءِ العرشِ مقصودِ

وزمزمٍ کلّما قسنا مواردَها

أوفى وأربى على کلِّ المواريدِ

والموقِفَيْنِ وما ضحّوا على عجلٍ

عند الجمارِ من الکوم المقاحيدِ ٣

وکلّ نسکٍ تلقّاه القبولُ فما

أمسى وأصبحَ إلاّ غيرَ مردودِ

وأرتضي أنّني قد متُّ قبلَکمُ

في موقفٍ بالردينيّاتِ مشهودِ

جمّ القتيل فهاماتُ الرجالِ به

في القاعِ ما بين متروکٍ ومحصودِ

فقلْ لآلِ زيادٍ أيُّ معضلةٍ

رکبتموها بتخبيبٍ وتخويدِ ٤

کيف استلبتمْ من الشجعانِ أمرَهمُ

والحربُ تغلي بأوغادٍ عراديدِ ٥ق.

(١)السجْل : الدلو العظيمة. الأوذام : جمع الوذمة ، وهی السير بين آذان الدلو والخشبة المعترضة عليها. المجدود : المقطوع.
(٢)الأُملود : الناعم الليّن.
(٣)الکوم - جمع الکوماء والأکوم - : الإبل الضخمة السنام. المقاحيد - جمع المقحاد - : النياق العظيمة السنام.
(٤)التخبيب والتخويد : الإسراع في السير.
(٥)العراديد - جمع العرديد - : المنحرف عن القتال أو الطريق.
