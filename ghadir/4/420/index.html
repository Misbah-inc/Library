الغدير 4 ٤٢٠

ولئن أصارتکِ الخطوبُ إلى بلىً

ولحاک ريبُ صروفِها فمحاکِ

فلطالما قضّيتُ فيک مآربي

وأبحتُ ريعانَ الشبابِ حماکِ

ما بين حورٍ کالنجومِ تزيّنتْ

منها القلائدُ للبدورِ حواکي

هيفُ الخصورِ من القصورِ بدت لنا

منها الأهلّةُ لا من الأفلاکِ

يجمعنَ من مرحِ الشبيبةِ خفّة ال

-متغزِّلين وعفّةَ النسّاکِ

ويصِدنَ صاديةَ القلوبِ بأعينٍ

نُجْلٍ کصيدِ الطيرِ بالأشراکِ

من کلِّ مخطفةِ الحشا تحکي الرشا

جيداً وغصنَ البان لينَ حراکِ

هيفاءَ ناطقةِ النطاقِ تشکّياً

من ظلمِ صامتة البُرين ضناکِ ١

وکأنّما من ثغرِها من نحرها

درٌّ تباکره بعود أراکِ

عذبُ الرضابِ کأنّ حشو لثاتِها

مسکٌ يعلُّ به ذرى المسواکِ

تلک التي ملکتْ عليَّ بدلِّها

قلبي فکانت أعنفَ المُلاّکِ

إنّ الصبا يا نفسُ عزَّ طلابُهُ

ونهتْکِ عنه واعظاتُ نهاکِ

والشيبُ ضيفٌ لا محالةَ مؤذنٌ

برداکِ فاتّبعي سبيلَ هداکِ

وتزوّدي من حبِّ آلِ محمدٍ

زاداً متى أخلصتِهِ نجّاکِ

فلنعم زادٌ للمعادِ وعدّةٌ

للحشرِ إن علقتْ يداک بذاکِ ٢

وإلى الوصيِّ مهمُّ أمرِکِ فوِّضي

تَصِلِي بذاکَ إلى قصيِّ مُناکِ

وبه ادرئي في نحرِ کلِّ ملمّةٍ

وإليه فيها فاجعلي شکواکِ

وبحبِّه فتمسّکي أن تسلکي

بالزيغِ عنه مسالکَ الهلاّکِ

لا تجهلي وهواه دأبُکِ فاجعلي

أبداً وهجرَ عداه هجرَ قلاکِ

فسواءٌ انحرف امرؤٌ عن حبِّهِ

أو باتَ منطوياً على الإشراکِ

وخذي البراءةَ من لظىً ببراءةٍ

من شانئيهِ وامحضيهِ هواکِف)

(١)البرين - بالضم جمع بره - : الخلخال. (المؤلف)
(٢)للحشر إن ظفرت بذاک يداکِ. کذا في نسخة. (المؤلف)
