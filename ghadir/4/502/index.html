الغدير 4 ٥٠٢

أما قال إنّي اليوم أکملتُ دينکمْ

وأتممتُ بالنعماءِ منّي عليکمُ

وقال أطيعوا الله ثمّ رسولَه

تفوزوا ولا تعصوا أُولي الأمر منکمُ

فلِمْ حرّموا ما کان حِلاّ وحلّلوا

بفتواهمُ ما جازَ وهو محرّمُ

ترى اللهَ فيما قال قد زلَّ أم هَذي

نبيُّ الهدى أم کان جبريلُ يوهمُ

لقد أبدعوا ممّا نووا من خلافِهمْ

وقال اقبلوا ممّا يقولُ وسلّموا

وإلاّ ترکتمْ إنْ أبيتم رماحَنا

وأسيافَنا فيکم تسدّى وتلحمُ

وما ماتَ حتى أکملَ اللهُ دينَهُ

ولم يبقَ أمرٌ بعد ذلک مبهمُ

ولکنْ حقودٌ أُظهِرتْ وضغائنٌ

وبغيٌ وجورٌ بيّنُ الظلم منهمُ

يُقرَّب مفضولٌ ويُبعَدُ فاضلٌ

ويَسکتُ منطيقٌ وينطقُ أبکمُ

وما أخّروا فيها عليّا لموجبٍ

ولکن تعدٍّ منهمُ وتظلّمُ

وکم شرعوا في نقضِ ما شاد أحمدٌ

ولکنّ دينَ اللهِ لا يتهدّمُ

وحاشا لدينٍ شيّدَ الحقُّ رکنَهُ

بسيفِ عليٍّ يعتريه التهدُّمُ

فحسبهمُ في ظلمِ آلِ محمدٍ

من اللهِ في العقبى عقابٌ ومأثمُ

فإن غصبوهمْ أمرَ دنياً دنيّةٍ

فما لهمُ في الحشر أبقى وأدوَمُ

فهل عظُمتْ في الدهر قطُّ مصيبةٌ

على الناسِ إلاّ وهي في الدينِ أعظمُ

تولّى بإجماعٍ على الناس أوّلٌ

ونصَّ على الثاني بها وهو مُغرمُ

وقال أقيلوني فلستُ بخيرِکمْ

فلِمْ نصّها لو صحَّ ما کان يزعمُ

وأثبتها في جوره بعد موته

صهاکيّةً خشناءَ للخصمِ تکلمُ

ولو أدرک الثاني لمولى حذيفةٍ

لولاّه دون الغير والأنفُ يُرغمُ

وقد نالها شورى من القومِ ثالثٌ

وجُرِّد سيفٌ للوصيّ ولهذمُ

أشورى وإجماعٌ ونصٌّ خلافةٌ

تعالوا على الإسلام نبکي ونلطمُ

وصاحبُها المنصوصُ عنها بمعزلٍ

يُديمُ تلاواتِ الکتابِ ويختمُ

ولو أنّه کان المولّى عليهمُ

إذن لهداهمُ فهو بالأمرِ أعلمُ
