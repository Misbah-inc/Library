الغدير 4 ٤٢١

وتجنّبي إن شئتِ أن لا تعطبي

رأيَ ابنِ سلمى فيه وابنِ صهاکِ

وإذا تشابهتِ الأمور فعوِّلي

في کشف مشکلِها على مولاکِ

خيرُ الرجالِ وخيرُ بعلِ نسائِها

والأصلُ والفرعُ التقيّ الزاکي

وتعوّذي بالزُّهرِ من أولادِهِ

من شرِّ کلِّ مُضلّل أفّاکِ

لا تعدلي عنهمْ ولا تستبدلي

بهمُ فتحظَى بالخسارِ هناکِ

فهمُ مصابيحُ الدجى لذوي الحجا

والعروةُ الوثقى لذي استمساکِ

وهم الأدلّةُ کالأهلّةِ نوُرها

يجلو عمى المتحيّرِ الشکّاکِ

وهم الصراطُ المستقيمُ فأرغمي

بهواهمُ أنفَ الذي يلحاکِ

وهم الأئمّةُ لا إمامَ سواهمُ

فدعي لِتيم وغيرِها دعواکِ

يا أُمّةً ضلّت سبيلَ رشادِها

إنّ الذي استرشدتِه أغواکِ

لئن ائتمنتِ على البريّةِ خائناً

للنفس ضيّعَها غداةَ رعاکِ

أعطاکِ إذ وطّاکِ عشوةَ رأيِه

خدعاً بحبلِ غرورِها دلاّکِ

فتبعتِهِ وسخيفَ دينِک بعتِهِ

مغترّةً بالنزرِ من دنياکِ

لقد اشتريتِ به الضلالة بالهدى

لمّا دعاکِ بمَکرهِ فدهاکِ

وأطعتِهِ وعصيتِ قولَ محمدٍ

فيما بأمرِ وصيِّه وصّاکِ

خلّفتِ واستخلفتِ من لم يرضَهُ

للدينِ تابعةً هوىً هوّاکِ

خلتِ اجتهادَکِ للصواب مؤدِّياً

هيهاتَ ما أدّاکِ بل أرداکِ

لقد اجتريتِ على اجتراح عظيمةٍ

جعلتْ جهنَّم في غدٍ مثواکِ

ولقد شققتِ عصا النبيِّ محمدٍ

وعققتِ من بعد النبيِّ أباکِ

وغدرتِ بالعهدِ المؤکَّدِ عقدُهُ

يومَ الغديرِ له فما عذراکِ

فلتعلمنَّ وقد رجعت به على ال

أعقاب ناکصةً على عقباکِ

أعَن الوصيِّ عدلت عادلةً به

مَن لا يساوي منه شسعَ شراکِ
