الغدير 4 ٣٨٦

ماضٍ لما أمَّ فلو جادَ في ال

-هيجاءِ بالحوباءِ لم يندَمِ

وکالفٍ بالحربِ لو أنّه

أُطعِمَ يَوم السلم لم يطعمِ

مثلَّمِ السيفِ ومن دونه

عرْضٌ صحيحُ الحدِّ لم يُثْلَمِ

فلم يزالوا يُکرعون الظُّبا

بين تراقي الفارسِ المُعْلَمِ

فمُثخَنٌ يحمل شهّاقةً

تحکي لراءٍ فغرة الأعلمِ ١

کأنّما الوَرسُ بها سائلٌ

أو أُنبِتَتْ من قُضُبِ العَنَدمِ

ومستزلٌّ بالقنا عن قَرا

عبلِ الشوى أو عن مطا أدهمِ ٢

لو لم يکيدوهم بها کيدَةً

لانقلبوا بالخِزي والمرغمِ

فاقتُضِبتْ بالبيضِ أرواحُهمْ

في ظلِّ ذاک العارضِ الأسحمِ

مصيبةٌ سيقت إلى أحمدٍ

ورهطِهِ في الملأ الأعظمِ

رزءٌ ولا کالرزءِ من قبلِهِ

ومؤلمٌ ناهيک من مؤلِمِ

ورميةٌ أصمتْ ولکنّها

مصميةٌ من ساعدٍ أجذَمِ

قل لبني حربٍ ومن جمّعوا

من حائرٍ عن رشده أو عَمِي

وکلِّ عانٍ في إسارِ الهوى

يُحسَب يقظانَ من النوَّمِ

لا تحسبوها حلوةً إنّها

أمرُّ في الحلق من العلقمِ

صرّعهمْ أنّهمُ أقدموا

کم فُدي الُمحجِمُ بالمُقدِمِ

هل فيکمُ إلاَّ أخو سوءَةٍ

مجرّحُ الجلدِ من اللوّمِ

إن خاف فقراً لم يجُد بالندى

أو هاب وشکَ الموتِ لم يُقدمِ

يا آل ياسينَ ومَن حبُّهمْ

منهجُ ذاک السَّنَنِ الأقومِ

مهابطُ الأملاکِ أبياتُهمْ

ومستقرُّ المُنزَلِ المحکَمِر.

(١)المثخن : الذي أثخنته الجراحات. الشهّاقة : الرمح. الفغرة : الفتحة. الأعلم : الذي شُقّت شفته العليا.
(٢)القرا : الظهر. العبل : الضخم. الشوى : الأعضاء. المطا : الظهر.
