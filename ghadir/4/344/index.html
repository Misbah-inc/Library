الغدير 4 ٣٤٤

وانتزعَ العاملَ ١ من قناتِهِ

فرُدَّ بالکرهِ فشدَّ فحملْ

والحال تُنبي أنّ ذاک لم يکنْ

عن توبةٍ وإنّما کان فشلْ

ومنهمُ من تابَ بعد موتِهِ

وليس بعدَ الموتِ للمرءِ عملْ

خ وإن تکن ذاتُ الغبيطِ أقلعتْ

برغم مَن أسند ذاک ونقلْ

فما لها تمنعُ من دفنِ ابنِهِ

لو لا هناتُ جرحِها لم يندملْ ٢

وما الخبيثانِ ابنُ هند وابنُهُ

وإن طغى خطبُهما بعدُ وجلْ

بمُبدِعَينِ في الذي جاءا به

وإنّما تقفّيا تلک السبلْ

إن يحسدوکَ فلفرطِ عجزهمْ

في المشکلاتِ ولما فيک کملْ

الصنوُ أنت والوصيُّ دونهمْ

ووارثُ العلمِ وصاحبُ الرسُلْ

وآکلُ الطائرِ والطاردُ للصِ

-لّ ومن کلّمهُ قبلک صلْ ٣

وخاصفُ النعلِ وذو الخاتمِ وال

-مُنهلُ في يومِ القليبِ والمُعِلْ

وفاصلُ القضيّةِ العسراءِ في

يومِ الجنينِ وهو حُکمٌ ما فصلْ

ورجعةُ الشمسِ عليکَ نبأٌ

تشعّبُ الألباب فيه وتضِلْ

فما ألومُ حاسداً عنک انزوى

غيظاً ولا ذا قدَمٍ فيک تزلْ

يا صاحبَ الحوضِ غداً لا حُلِّئتْ

نفسٌ تواليک عن العذب النَّهِلْ ٤

ولا تُسلَّطْ قبضةُ النارِ على

عُنقٍ إليک بالوداد ينفتلْ

عاديتُ فيک الناسَ لم أحفلْ بهمْ

حتى رموني عن يدٍ إلاّ الأقلْ

تفرّغوا يعترقون غيبةً

لحمي وفي مدحِکَ عنهم لي شُغُلْ ٥

عدلتُ أن ترضى بأن يسخطَ من

تُقِلُّهُ الأرضُ علیَّ فاعتدلْن.

(١)العامل : صدر الرمح وهو ما يلي السنان. (المؤلف)
(٢)هذا البيت غير موجود في الديوان.
(٣)الصلُّ : الثعبان. (المؤلف)
(٤)حلِّئت : مُنِعتْ من الورد. (المؤلف)
(٥)يعترقون : ينزعون ما علی العظم من لحم ، وهی هنا بمعنى يأکلون.
