الغدير 4 ١١٣

نعى کافي الکفاة فکلُّ حرٍّ

عزيزٍ بعد مصرعه ذليلُ

نعى کهفَ العُفاةِ فکلُّ عينٍ

بما تقذي العيون به کحيلُ

کأنّ نسيمَ تربتِهِ سحيراً

نسيمُ الروضِ تقبلُه القبولُ

إذا وافى أُنوفَ الرکبِ قالوا

سحيقُ المسکِ أم تُربٌ مَهيلُ

أيا قمرَ المکارمِ والمعالي

أبِن لي کيف عاجلَکَ الأُفولُ

أبِن لي کيف هالَکَ ما يهول

وغالَکَ بعد عزِّک ما يغولُ

ويا من ساس أشتات البرايا

وألجم من يقول ومن يصولُ

أَدَلتَ على الليالي من شکاها

وقد جارت عليک فمن يُديلُ

بکاک الدينُ والدنيا جميعاً

وأهلهما کما يُبکى الحمولُ ١

بکتکَ البيضُ والسمْرُ المواضي

وکنتَ تعولها فيمن تعولُ

بکتک الخيلُ مُعوِلةً ولکن

بُکاها حين تندبُکَ الصهيلُ

قلوبُ العالمينَ عليکَ قلبٌ

وحظُّکَ من بکائهمُ قليلُ

ولي قلبٌ لصاحبه وفيٌ

يسيلُ وتحتَهُ روحٌ تسيلُ

إذا نظمتْ يدي في الطرسِ بيتاً

محاهُ منه منتظمٌ هَطُولُ

فإن يکُ رکَّ شعري من ذهولي

فذلک بعضُ ما يجني الذهولُ

کتبتُ بما بکيتُ لأنّ دمعي

عليک الدهرَ فيّاضٌ هَمولُ

وکنتُ أَعُدُّ من روحي فداءً

لروحِکَ إن أُريدَ لها بديلُ

أأحيا بعدَهُ وأقرُّ عيناً

حياتي بعدَهُ هدرٌ غلولُ

حياتي بعدَهُ موتٌ وَحِيٌ

وعيشي بعده سمٌّ قَتولُ ٢

عليک صلاةُ ربِّک کلَّ حينٍ

تهبُّ بها من الخلد القبولُع.

(١)الحمول : الميت الذي يُحمَل.
(٢)الوَحيّ : السريع.
