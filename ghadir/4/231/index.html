أحبُّهمُ وأمنحهمْ مديحاً

وأمنحُ من يسبُّهمُ سبابا

ولم أمدحهُمُ قطُّ اکتساباً

ولکنّي مدحتهمُ ارتغابا

ولن يرجو ابنُ حمّادٍ عليٌ

بحسنِ مديحِهم إلاّ الثوابا

8 - هل لجسمي من السقامِ طبيبُ

أم لعيني من الرقادِ نصيبُ 26

9 - يا أهلَ بيتِ رسول الله إنکُمُ

لأشرفُ الخلقِ جَدّا غاب أو أبی 30

10 - الدهرُ فيه طرائفٌ وعجائبُ

تترى وفيه فوائدٌ ومصائبُ 60

11 - أيا من لقلبٍ دائمِ الحسراتِ

ومن لجفونٍ تسکبُ العبراتِ 34

وهي على رويِّ تائية دعبل ، يقول في آخرها :

إليک أمينَ اللهِ نظمَ قصيدةٍ

إماميّةٍ تزهو بحسنِ صفاتِ

عليُّ بنُ حمّادٍ دعاها فأقبلتْ

وهمّتُهُ من أعظمِ الهمماتِ

شبيهٌ لما قال الخزاعيُّ دعبلٌ

تضمّنه الرحمنُ بالغرفاتِ

مدارسُ آياتٍ خلتْ من تلاوةٍ

ومهبطُ وحيٍ مقفرُ العرصاتِ

12 - بقاعٌ في البقيعِ مقدّساتُ

ر وأکنافٌ بطيبةَ طيّباتُ 95

13 - دعني أنوح وأُسعد النوّاحا

مثلي بکى يومَ الحسينِ وناحا 28

14 - أرى الصبرَ يفنى والهمومَ تزيد

وجسميَ يبلى والسقامُ جديدُ 23

15 - ما ضرّ عهدَ الصبا لو أنَّه عادا

يوماً يزوِّدني من طيبةٍ زادا 86

جارى بها السيّد إسماعيل الحميري في قصيدة له أوّلها :

طافَ الخيالُ علينا منک عبّادا

فقال العبديُّ في آخر قصيدته :

وازنتُ ما قال إسماعيلُ مبتدئاً

طافَ الخيالُ علينا منک عبّادا
