الغدير 4 ٣٩٦

في موقفٍ فيه الصوا

رم والذوابل والکماةُ

وأتاهمُ من حيث لم

يخشوا لِحَيْنِهُمُ المماتُ

وطوتهمُ طيَّ البرو

د لهم قبورٌ مظلماتُ

فهمُ بها مثل الهشی

-م تعيثُ فيها العاصفاتُ

شُعثٌ وسائدُهم بها

من غير تکرمةٍ علاةُ

قل للذين لهم إلى

الدنيا دواع مسمعاتُ

وکأنّهم لم يسمعوا

ما ذا تقول الناعياتُ

أو ما تقول لهم إذا اج

-تازوا الديارُ الخالياتُ

فالضاحکاتُ وقد نعِم

-نَ بهنّ هنّ الباکياتُ

حتى متى وإلى متى

تأوي عيونکم السناتُ

کم ذا تفرِّجُ عنکمُ

أبدَ الزمانِ الموعظاتُ

کم ذا وُعظتم لو تکو

نُ لکم قلوبٌ مصغياتُ

لکمُ عقولٌ معرضا

تٌ أو عيونٌ عاشياتُ

عُجْ بالديارِ فنادِها

أينَ الجبالُ الراسياتُ

أين العصاةُ على المکا

رمِ للعواذلِ والأُباةُ

تجري المنايا من روا

جبهمْ جميعاً والصلاتُ ١

وإذا لَقَوا يوم الوغى

أقرانَهمْ کانت هناةُ

والدهرُ طوعَ يمينِهمْ

وهمُ على الدنيا الولاةُ

أعطاهمُ متبرِّعاً

ثمّ استردَّ فقال هاتوا

کانت جميعاً ثمّ مزّق

شملَ بينِهمُ الشتاتُ

فأکفُّهمْ من بعد أن

سُلِبوا المواهبَ مقفراتُ

وسيوفُهمْ ورماحُهمْ

منبوذةٌ والضامراتُ

أمِنوا الصباحَ وما لهم

علمٌ بما يجنى البياتُع.

(١)الرواجب : مفاصل أصول الأصابع.
