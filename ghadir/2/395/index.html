الغدير 2 ٣٩٥

أنْذِرْ عَشيرتَکَ الدنيا بشِرْعَتِک ال

غرّا وأظهرْ لها أسنى معانيها

وَمُذْ تُبَلَّغُ أمرَ اللهِ هِمَّ به

بهمّةٍ ما اعتدا الکفّارِ يَثنيها

ولم يَجِدْ عَضُداً کي يستعينَ به

على مُجاهرةٍ قد کان خاشيها

إلاّ العليَّ فناداهُ وأخبرَه

ببغيةٍ حسبَ أمرِ اللهِ باغيها

وقال هيِّئ لنا في الحال مأدُبةً

وَلْيُتْقِنَنَّ لها الألوانَ طاهيها

فَرِجْلُ شاةٍ على صاع الطعام وأع

ساسٌ لها اللبنُ النوقيُّ يُمليها

وادعُ الهواشمَ باسمي کي أُشافِهَها

بأمرِ ربِّيَ باريّ وباريها

قام العليُّ بأمرِ المصطفى ودعا

إلى وليمتِهِ أکرِمْ بداعيها

أبناءَ هاشم هم کانوا عشيرتَهُ

ولم يکنْ فيهمُ إلاّ مُلَبِّيها

وعَدُّهمْ کان عند الأربعين وهمْ

رجّالة العُرْبِ في إحصاءِ مُحصيها

هذي عشيرة طه بل قرابتُهُ ال

دنيا التي کان للإسلامِ راجيها

وإذ أتتْهُ تلقّاها على رَحِبٍ

ببِشرِهِ وانثنى صفواً يُحيّيها

حتى إذا ما استوى فيها المقامُ لها

مدَّ السِّماط وفيه ما يُشهِّيها

فأقبلتْ ورسول الله يخدمُها

على الطعامِ ويُعني کي يُهنِّيها

حتى إذا أکَلتْ ذاک الطعامَ ومِن

ألبانِهِ سُقِيَتْ واللهُ کافيها

ظلَّ الطعامُ کما قد کان وهو واي

مُ اللهِ ما کان يکفي مُستجيعيها

وتلک معجزةٌ للمصطفى وبها

قام العليُّ وعنه نحن نَروِيها

وثَمَّة ابتدرَ القومَ الرسولُ بذک

ري يُمْنِ بعثتهِ يُبدِي خَوافيها

وإذ أبو لَهَبٍ في الحالِ قاطَعَه

وموّه الحقّ بالتضليل تمويها

وقال يا ناس طه جاء يسحرکم

بذا الطعامِ احذروا الإضلال والتِّيها

هيّا انهضوا ودعوهُ أن يَغُشَّ نفو

س الغيرِ في هذه الدعوى ويُصبِيها

وهکذا ارْفَضَّ ذاک الاجتماعُ وأن

فسَ الجَمْع داجي الکفر غاشيها

وعادَ طه إلى تِکرارِ دعوَتِهِ

وکان حيدرةُ المِقْدامُ راعيها
