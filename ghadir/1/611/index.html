الغدير 1 ٦١١

ألا إنَّ من أنا مولىً له

فمولاه هذا قضاً لن يجورا

فهل أنا بلّغتُ قالوا نعم

فقال اشهدوا غُيَّباً أو حضورا

يُبلِّغ حاضرُکم غائباً

وأُشهِدُ ربّي السميعَ البصيرا

فقوموا بأمرِ مليکِ السما

يبايعْه کلٌّ عليه أميرا

فقاموا لبيعته صافقين

أکفّاً فأوجس منهم نکيرا

فقال إلهيَ والِ الوليَّ

وعادِ العدوَّ له والکفورا

وکن خاذلاً للأُلی يخذُلون

وکن للأُلی ينصرون نصيرا

فکيف ترى دعوةَ المصطفى

مُجاباً بها أم هباءً نثيرا

أُحبّکَ يا ثانيَ المصطفى

ومن أُشْهِدَ الناسُ فيه الغديرا

ومنهم : العبدي الکوفي من شعراء القرن الثاني في بائيّته الکبيرة بقوله :

وکان عنها لهمْ في خُمّ مُزدَجرٌ

لَمّا رقى أحمدُ الهادي على قَتَبِ

وقال والناسُ من دانٍ إليه ومن

ثاوٍ لديه ومن مُصغٍ ومُرتقبِ

قم يا عليُّ فإنّي قد أُمِرتُ بأن

أُبلِّغَ الناسَ والتبليغُ أجدرُ بي

إنّي نصبتُ عليّاً هادياً علماً

بعدي وإنَّ عليّاً خيرُ مُنتصَبِ

فبايعوک وکلٌّ باسطٌ يَدَهُ

إليک من فوقِ قلبٍ عنک مُنقلبِ

ومنهم : شيخ العربيّة والأدب أبو تمام المتوفّى ⁦(231)⁩ في رائيّته بقوله :

ويومَ الغديرِ استوضح الحقَّ أهلُهُ

بصحياءَ لا فيها حجاب ولا سترُ

أقام رسول الله يدعوهُمُ بها

ليقْرَبَهُمْ عُرفٌ وينآهُمُ نُکْرُ

يمدّ بضَبْعيهِ ويُعلِمُ أنَّه

وليٌّ ومولاکمْ فهل لَکُمُ خُبْرُ

يروحُ ويغدو بالبيان لِمَعْشَرٍ

يروحُ بهم غَمْرٌ ويغدو بهم غَمْرُ

فکان لهم جهرٌ بإثباتِ حقِّهِ

وکان لهم في بَزِّهمْ حقّه جهرُ
