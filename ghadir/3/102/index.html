الغدير 3 ١٠٢

کأنما القُبلةُ في خدّهِ

للحُسنِ من رقّته عضّه

يهتزُّ أعلاه إذا ما مشى

وکلّه في يمنه قبضه

ارحم فتىً لمّا تملّکتَهُ

أقرّ بالرقّ فلم تَرضَه

وله في الأنوار (ص 480) قوله :

بأبي فمٌ شهد الضميرُ لهُ

قبل المذاقِ بأنّه عَذْبُ

کشهادتي للهِ خالصةً

قبلَ العيان بأنّه الربُ

والعينُ لا تُغني بنظرتِها

حتى يکونَ دليلَها القلبُ

وله في (ص 481) قوله :

کأنّ همومَ الناسِ في الأرض کلّها

عليَّ وقلبي بينهم قلبُ واحدِ

ولي شاهدا عدلٍ سُهادٌ وعَبرةٌ

وکم مدّعٍ للحقّ من غير شاهدِ

وله في (ص 528) قوله :

وجهٌ هو البدر إلاّ أنّ بينهما

فضلاً تحيّر عن حافاته النورُ

في وجهِ ذاکَ أخاليطٌ مُسوّدةٌ

وفي مضاحِکِ هذا الدرُّ منثورُ

وذکر له في نشوة السکران (ص 79) قوله :

عَريتُ عن الشبابِ وکنتُ غضّا

کما يعرى عن الورقِ القضيبُ

ونُحتُ على الشبابِ بدمعِ عيني

فما نفعَ البکاءُ ولا النحيبُ

ألا ليتَ الشبابَ يعود يوماً

فأخبرَهُ بما فعلَ المشيبُ ١

ولادته ووفاته :

لم نقف على تاريخ ولادة المترجَم سيّدنا الحِمّاني ، غير أنّ المستفاد من وفاته ف)

(١)توجد هذه الأبيات بتغيير يسير في ديوان أبي العتاهية : ص 23. (المؤلف)
