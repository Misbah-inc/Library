الغدير 3 ٣٣٠

أما إنّه أوّلُ العابدين

بمکّةَ واللهُ لا يُعبدُ ١

50 - جندب بن زهير ، کان يرتجز يوم صفِّين بقوله :

هذا عليٌّ والهدى حقّا مَعَهْ

يا ربّ فاحفظْهُ ولا تضيِّعَهْ

فإنّه يخشاکَ ربّي فارفعَهْ

نحن نصرناهُ على من نازعَهْ

صهرُ النبيِّ المصطفى قد طاوعَهْ

أوّل من بايَعَهُ وتابعَهْ ٢

51 - زُفَر بن يزيد ٣ بن حذيفة الأسديّ ، قال :

فحوطوا عليّا وانصروه فإنّه

وصيٌّ وفي الإسلامِ أوّلُ أوّلُ

وإن تخذلوهْ والحوادثُ جمّةٌ

فليس لکمْ عن أرضکم مُتَحوّلُ ٤

52 - النجاشي بن الحارث بن کعب ، قال :

فقل للمضلّلِ من وائلٍ

ومن جعلَ الغَثَّ يوماً سمينا

جعلتَ ابنَ هندٍ وأشياعَهُ

نظيرَ عليٍّ أما تستحونا

إلى أوّل الناسِ بعد الرسول

أجابَ النبيَّ من العالمينا

وصهرِ الرسولِ ومنْ مثلُه

إذا کان يومٌ يشيبُ القرونا ٥

53 - جرير بن عبد الله البجليّ ، قال :

فصلّى الإلهُ على أحمدٍ

رسولِ المليکِ تمامَ النعَمْ

وصلّى على الطهرِ من بعدِهِ
