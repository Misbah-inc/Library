الغدير 3 ٥٤٢

تُلفى جموعُکمُ شتّى مفرَّقةً

بين العبادِ وشملُ الناسِ مجتمعُ

وتُستباحون أقماراً مُنکَّسةً

تهوى وأرؤسُها بالسمرِ تُقتَرعُ

ألستمُ خيرَ من قامَ الرشادُ بکم

وقوِّضت سُننُ التضليلِ والبدعُ

ووُحِّد الصمدُ الأعلى بهداکمُ

إذ کنتمُ عَلَماً للرشد يُتّبعُ

ما للحوادثِ لا تجري بظالمکم

ما للمصائب عنکم ليس ترتدعُ

منکم طريدٌ ومقتولٌ على ظمأ

ومنکمُ دنفٌ بالسمرِ مُنصرعُ

وهاربٌ في أقاصي الغربِ مغتربٌ

ودارعٌ بدمِ اللبّاتِ مندرعُ

ومقصدٌ من جدارٍ ظلّ مُنکدراً

وآخر تحت ردمٍ فوقه يقعُ

ومن محرّقِ جسمٍ لا يُزارُ له

قبرٌ ولا مشهدٌ يأتيه مرتدعُ

وإن نسيتُ فلا أنسى الحسينَ وقد

مالت إليهِ جنودُ الشرکِ تقترعُ

فجسمُهُ لحوامي الخيل مطّردٌ

ورأسه لسنان السُّمر مرتفعُ

وله في رثائهم - سلام الله عليهم - قوله :

بنو المصطفى تَفْنَوْن بالسيفِ عنوةً

ويُسلمني طيفُ الهجوع فأهجعُ

ظُلِمتم وذبِّحتمْ وقُسِّمَ فيئُکمْ

وجارَ عليکمْ من لکمْ کانَ يخضعُ

فما بقعةٌ في الأرض شَرقاً ومغرباً

وإلاّ لکمْ فيها قتيلٌ ومصرَعُ

وله في رثاء الإمام السبط الشهيد عليه السلام قوله :

أعاتبُ عيني إذا أقصرتْ

وأفني دموعي إذا ما جَرَتْ

لذکراکمُ يا بني المصطفى

دموعي على الخطِّ قد سُطّرتْ

لکم وعليکم جَفَتْ غَمْضَها

جفوني عن النومِ واستشعرتْ

أُمثِّل أجسادکم بالعراقِ

وفيها الأسنّةُ قد کُسِّرتْ

أُمثّلکم في عراصِ الطفوفِ

بدوراً تکسّفُ إذ أقمرتْ

غدتْ أرضُ يثربَ من جمعکمْ

کخطِّ الصحيفةِ إذ أقفرتْ
