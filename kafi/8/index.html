الروضة من الكافي

٥محتویات الکتاب٥كتاب الروضة٦الرسالة من أبي عبد الله(عليه السلام)إلى جماعۀ الشیعه١٧صحيفة علي بن الحسين(عليه السلام)وكلامه في الزهد٢٠خطبة لأمير المؤمنين(عليه السلام)وهي خطبة الوسيلة٣٠خطبة لأمیر المومنین(علیه السلام)وهی خطبۀ الطَّالوتِيَّة٣٢مقامات الشیعه وفضائلهم وبشارتهم بخیر المآل٣٥حدیث أبی عبدالله (علیه السلام)مع المنصور فی موکبه وفیه علامات آخر الزمان، تناهز المائۀ والخمسین من الفتن والأشراط٤٠حدیث موسی(علیه السلام)وما خاطبه الله عزوجل به٤٧ان الله تعالی اختار من بنی هاشم سبعۀ لم یخلق مثلهم٤٧معنی قوله تعالی: ﴿هَذَا كِتَابُنَا يَنْطِقُ عَلَيْكُمْ بِالْحَقِّ﴾٤٨تأویل قوله تعالی:﴿وَالشَّمْسِ وَضُحَاهَا﴾٤٨تأویل قوله تعالی: ﴿هَلْ أَتَاكَ حَدِيثُ الْغَاشِيَةِ﴾٤٨تأویل قوله تعالی: ﴿وَأَقْسَمُوا بِاللَّهِ جَهْدَ أَيْمَانِهِمْ لَا يَبْعَثُ اللَّهُ مَنْ يَمُوتُ﴾٤٩تأویل قوله تعالی: ﴿فَلَمَّا أَحَسُّوا بَأْسَنَا إِذَا هُمْ مِنْهَا يَرْكُضُونَ﴾٥٠رسالة أبي جعفر(عليه السلام)إلى سعد الخير٥٣کان امیر المومنین (علیه السلام)یشبه عیسی بن مریم(علیه السلام)٥٤تفسیر قوله تعالی : ﴿ظَهَرَ الْفَسَادُ فِي الْبَرِّ وَالْبَحْرِ بِمَا كَسَبَتْ أَيْدِي النَّاسِ..الأیۀ﴾٥٥تفسیر قوله تعالی: ﴿وَلَا تُفْسِدُوا فِي الْأَرْضِ بَعْدَ إِصْلَاحِهَا﴾٥٥خطبة لأمير المؤمنين(عليه السلام)فی ذم اتباع الهوی٥٥تأسفه علی بعض ما حدث بعد رسول الله(صلی الله علیه وآله وسلم)٥٧خطبة لامير المؤمنين(عليه السلام)فی تأسفه علی ما سیحدث٦٠خطبة لامير المؤمنين(عليه السلام)فی عاقبۀ الظلم والبغی٦١حديث علي بن الحسين(عليه السلام)وفیه حثُّ علی التقوی٦١علامات آخر الزمان أو أشراط الساعۀ٦٢تسویۀ أمیرالمومنین(عليه السلام)بین المسلمین فی تقسیم بیت المال٦٢حديث النبي (صلی الله علیه وآله وسلم)حين عُرِضَتْ عليه الخَيْلُ٦٤نصیحۀ أمیرالمومنین(عليه السلام)لمولی له فرّ منه إلی معاویۀ٦٤خطبۀ علي بن الحسين(عليه السلام)وموعظۀ الناس فی کل یوم جمعۀ٦٧حديث الشيخ مع أبی جعفرالباقر(عليه السلام)٦٨قصة صاحب الزيت مع رسول الله(صلی الله علیه وآله وسلم)٦٩وصية النبي(صلی الله علیه وآله وسلم)لأمير المؤمنين(عليه السلام)٧٠الدین هو الحب وأنت مع من أحببتَ٧٠فضل أهل البیت وشیعتهم وأن علیاً(عليه السلام)أفضل الناس بعد النبی(صلی الله علیه وآله وسلم)٧١إحیاء أمرهم وانتظار فَرَجِهِم(عجل الله تعالی فرجه)٧٢فضل الشیعۀ وتفسیر قوله تعالی : ﴿ولقد أرسلنا رسلاً من قبلك﴾٧٢الشقی من الشقی فی بطن أمه والسعید من وُعِظَ بغیره٧٢تفسیر قوله تعالی :﴿ كَانَ النَّاسُ أُمَّةً وَاحِدَةً ﴾٧٣حديث البحر مع الشمس٧٤لکل أهل بیت حجۀ یحتج الله بها یوم القیامۀ٧٤تفسیر قوله تعالی :﴿ وَأَرْسَلَ عَلَيْهِمْ طَيْرًا أَبَابِيلَ...الآیۀ﴾٧٥عُوذَۀ للریح والوجه٧٥حدیث نبوی(صلی الله علیه وآله وسلم)فیه وصیۀ نافعۀ٧٥ادعاء الرجل الهمدانی بغلۀ موسی بن جعفر(عليه السلام)٧٦تعریض العاشرلأبی عبدالله(عليه السلام) وسلوکه معه٧٦کیفیۀ معاشرۀ أبی عبدالله(عليه السلام)مع غلامه٧٦حديث الطبيب وبیان وجه التسمیۀ٧٦فی أن غالب الأدواء له مادۀ فی الجسد٧٧الإستشفاء بالبُرّ وکیفیتۀ٧٧حديث الحوت على أيّ شيءٍ هو ؟٧٧خلق الأرض وإرسال الماء المالح إلیها وأصل الخلق٧٨حديث الأحلام والحجّة على أهل ذلك الزمان٧٨رؤیا المؤمن فی آخر الزمان علی سبعین جزاً من أجزاء النبوۀ٧٨سؤال النبی(صلی الله علیه وآله وسلم):«هل من مبشّرات»٧٨تفسیر قوله تعالی :﴿لَهُمُ الْبُشْرَى فِي الْحَيَاةِ الدنيا﴾٧٩الرؤیا علی ثلاثۀ وجوه٧٩الرؤیا الصادقۀ والکاذبۀ مخرجها من موضع واحد٧٩حديث الرِّياح وهی أربعۀ أقسام : الشمال والجنوب والصبا والدبور٧٩إن لله عزوجل ریاح رحمۀ وریاح عذاب٨١علاج الهم والفقر والسقم٨١فی معنی ذوی القربی٨٢حديث الرجل الشاميّ مع أبي جعفر(عليه السلام)وما سأله عنه٨٢کان کل شیء ماءاً وکان عرشه تعالی علی الماء٨٣حديث الجنان والنُّوق ووصف أهل الجنۀ٨٧کلامهم (عليه السلام)علی سبعین وجهاً لهم منها المخرج٨٨حديث أبي بصير مع المرأة٨٨الناصب لأهل البیت شرّ من تارک الصلاۀ٨٩من استخفّ بمؤمن فیهم، ومن ذبّ عنهم(عليه السلام)٨٩حدیث عبدالرحمن مع أبی عبدالله(عليه السلام)٩٠ما قال عمر لعلی بن أبی طالب (عليه السلام)فی بنی أمیۀ٩٠فی قوله تعالی:﴿ الَّذِينَ بَدَّلُوا نِعْمَتَ اللَّهِ كُفْرًا﴾٩٠نزول قوله تعالی :﴿فَتَوَلَّ عَنْهُمْ فَمَا أَنْتَ بِمَلُومٍ﴾٩١احوال یوم القیامۀ وبَعْث الخلایق٩٢من أحب أهل البیت (عليهم السلام)کان معهم یوم القیامۀ٩٣ردٌّ علی من زعم دن الکمال کله فی عفۀ البطن والفرج٩٤إذا بلغ المؤمن أربعین سنة٩٤فی جواز الفرار من الوباء٩٥ثلاثۀ لم ینج منها نبی فمن دونه٩٦غزوۀ أحد ومواساۀ أمیرالمومنین لرسول الله(صلی الله علیه وآله وسلم)٩٧حدیث آدم(عليه السلام)مع الشجرة٩٨قصۀ قابیل وَهِبۀ الله٩٩قصّه نوح (عليه السلام)١٠١أمره سبحانه وتعالی رسوله بالوصیۀ لعلی صلوات الله علیهما١٠٥حديث نصراني الشام مع أبی جعفر الباقر(عليه السلام)١٠٦کتاب أبي الحسن موسى(عليه السلام)إلی علی بن سوید١٠٩حدیث نادر فی أبی ذر مع رسول الله(صلی الله علیه وآله وسلم)١١٠لایقبل الله تعالی عملاً إلا بولایۀ أهل البیت (عليهم السلام)١١٣فی زهد النبی(صلی الله علیه وآله وسلم)وأدبه وزهد علی(عليه السلام)١١٣فیما ناجی الله عزوجل عیسی بن مریم(عليه السلام)١٢٣حديث إبليس لعنۀ الله١٢٣دعاء علمۀ رسول الله(صلی الله علیه وآله وسلم)فاطمۀ (عليه السلام)فی رؤیاها التی رأتها١٢٤حديث محاسبة النفس١٢٤مثل الناس یوم القیامۀ١٢٤حدیث حفص وسجود أبی عبدالله(عليه السلام)١٢٤فی مذمۀ الدنیا١٢٥فی ذم شکایۀ المومن حاجته عند الکافر١٢٥حدیث المشرکین مع رسول الله(صلی الله علیه وآله وسلم)١٢٥إن الله تعالی خلق الجنۀ قبل أن یخلق النار١٢٦فی قوله تعالی :﴿ خَلَقَ السَّمَاوَاتِ وَالْأَرْضَ وَمَا بَيْنَهُمَا فِي سِتَّةِ أَيَّامٍ﴾١٢٦حدیث فیه مدح لزرارۀ بن أعین وأصحابه١٢٧فضل الشیعۀ، ووصیۀ أبی عبدالله (عليه السلام)لهم١٢٧من مات ولم یکن له إمام مات میته جاهلیۀ١٢٧إن رسول الله(صلی الله علیه وآله وسلم)إذا ذهب من طریق رجع من غیره١٢٨تکذیب المغتاب وحملُ فعل المومن علی أحسنه١٢٨حديث من وُلد في الإسلام١٢٨عرف الله تعالی نفسه إلی خلقۀ بالکلام والدلالات١٢٨ما خلق الله عزوجل شیئاً إلا وخلق شیئاً یغلبه١٢٩وصیّۀ رسول الله (صلی الله علیه وآله وسلم)لرجل استوصاه١٢٩أمر النبی(صلی الله علیه وآله وسلم)بالترحم علی ثلاث١٢٩نهی عن تجسّس عیوب من کان أقبل إلینا بمودته١٣٠جعل المتعۀ للإمامیۀ عوضاً من الأشربۀ١٣٠ما شرط الرضا(عليه السلام)علی المأمون فی قبول ولایۀ العهد١٣٠بعض حقوق المسلم مع إخوانه١٣١النهی عن تعریض الإنسان نفسه للتهمۀ١٣١صفۀ نهر فی الجنۀ یقال له: جعفر١٣١النصر مع من أحسن الرعایۀ والحفظ للإسلام١٣١موعظۀ نافعۀ لعلی بن الحسین(عليه السلام)١٣١كان كل شيء ماءاً وكان عرشه على الماء١٣٢حديث زينب العطّارة١٣٣حديث الذي أَضَافَ رسول الله(صلی الله علیه وآله وسلم)بالطائف١٣٣حمل عظام یوسف(علیه السلام)وخبر عجوز بنی إسرائیل١٣٥تأویل قول تعالی :﴿ويستبشرون بالذين لم يلحقوا...الآیۀ﴾١٣٥تفسیر قوله تعالی:﴿فيهِنَّ خَيْراتُ حِسان﴾١٣٥للشمس ثلاثمأئۀ وستون برجاً١٣٥نهی أبی جعفر(عليه السلام)جابر الجعفی عن إفشاء سبعین حدیثاً علّمه١٣٧حديث الناس يوم القيامة١٣٨حدیث سلیمان بن خالد مع أبی عبدالله(عليه السلام)فی الزیدیۀ١٣٨لم سُمّی المؤمن مؤمناً(عليه السلام)١٣٨الناصب لا یبالی صلّی أم زنا١٣٩من لم یوال علیاً(عليه السلام)١٣٩مدح بالغ لزید بن علی بن الحسین١٣٩هلاک بنی أمیۀ بعد زید بن علی بن الحسین(عليه السلام)١٣٩إیاب الخلق إلیهم وحسابهم علیهم(عليهم السلام)١٤٠وجوب الإجتناب عن فاعل المنکر١٤٠إن الله يعذب الستة بالستة١٤٠أحب الأشیاء ألی رسول الله(صلی الله علیه وآله وسلم)١٤١سیره علی(عليه السلام) وزهد وأن ولیّه لایأکل الحرام١٤١کراهیۀ أکل الطعام الحارّ وآکل التمر علی الطعام١٤٢سیره علی وفاطمۀ(عليهم السلام)١٤٣فی معنی قوله تعالی:﴿سَنُرِيهِمْ آيَاتِنَا فِي الْآفَاقِ وَفِي أَنْفُسِهِمْ﴾١٤٣طاعۀ علی(عليه السلام)ومعصیته١٤٤مدح الشیعۀ وذمّ مخافیهم١٤٤الحکمۀ ضالّۀ المؤمن فحیثما وجد أخذ١٤٤أشعث بن قیس وبنته وإبنه لعنهم الله١٤٤الرقۀ والبکاءعند سماع قراءۀ القرآن وموعظۀ نافعۀ١٤٥وصیۀ أبی عبدالله (عليه السلام)لعمرو بن سعید بن هلال١٤٥وصیۀ رسول الله(صلی الله علیه وآله وسلم)لأصحابه١٤٦النهی عن الشکوی إلی أهل الخلاف١٤٦خطبة لأمير المؤمنين(عليه السلام)فی الموعظۀ١٤٨خطبة لأمير المؤمنين(عليه السلام)أیضاً فی الوصیۀ بتقوی الله تعالی فی یوم الجمعۀ١٥١لكل مؤمن حافظ من الله عزوجلوسائب١٥١الناس معادن کمعادن الذهب والفضۀ١٥١حدیث الزُّوار وما یُقتل فیها١٥٢تفسیر قوله تعالی:﴿ وَلَا يُؤْذَنُ لَهُمْ فَيَعْتَذِرُونَ ﴾١٥٢تأویل قوله تعالی:﴿ وَمَنْ يَتَّقِ اللَّهَ يَجْعَلْ لَهُ مَخْرَجًا﴾١٥٣تأویل قوله تعالی:﴿مَا يَكُونُ مِنْ نَجْوَى ثَلَاثَةٍ...﴾١٥٣الذین خرجوا یوم البصرۀ وهم الباغون١٥٤إیذاء بعض الصحابۀ سلمان الفارسی رضی الله عنه١٥٤تسویۀ أمیرالمومنین(عليه السلام)فی العطائ بین الأسود والأبیض١٥٥موعظۀ رسول الله(صلی الله علیه وآله وسلم) بنی عبدالمطلب١٥٥رؤیا رآها أبو جعفر(عليه السلام)فی میسّر بن عبدالعزیز وعبدالله بن عجلان١٥٥ان الملائکۀ تغسل أبا جعفر(عليه السلام)فی البقیع١٥٦بیان قوله تعالی:﴿وَلَوْ أَنَّا كَتَبْنَا عَلَيْهِمْ أَنِ اقْتُلُوا...الآیۀ﴾١٥٦بیان قوله تعالی:﴿أُولَئِكَ الَّذِينَ يَعْلَمُ اللَّهُ مَا فِي قُلُوبِهِمْ...الایۀ ﴾١٥٦لایوجب الله طاعۀ أولی الأمر ویرخّص فی منازعتهم١٥٧حديث قوم صالح(عليه السلام)١٦٠حدیث فَرْوَۀ عن أبی جعفر(عليه السلام)١٦١معالجة بعض الأمراض١٦١معالجۀ من تغیّر علیه ماءالظهر١٦٣دواء الضرس والفم١٦٤فی النظر فی علم النجوم١٦٥لا عدوى ولا طيرة ولا هامّة ولاشوم ولاصفَر١٦٥قصۀ الذين خرجوا من ديارهم وهم ألوف حذرَ الموت١٦٦هل یعلم یعقوب (عليه السلام)أن یوسف حیّ؟١٦٦تدویل قوله تعالی: ﴿ فَعَمُوا وصَمّوا﴾١٦٦معنی قوله تعالی:﴿ لُعِنَ الَّذِينَ كَفَرُوا مِنْ بَنِي إِسْرَائِيلَ..الآیۀ﴾١٦٧قراءۀ قوله تعالی: ﴿ فَإِنَّهُمْ لَا السلام)فی إنذاره بما یأتی من زمانالسوء٢٩٩حديث إبراهيم(عليه السلام)والرجل العابد٢٩٩إن الله اتخذ ابراهیم(عليه السلام)خلیلاً٣٠٠كلام لعلي بن الحسين(عليه السلام)٣٠١قول أبی عبدالله(عليه السلام): لا يخرج على هشام أحد إلا قتله، وحدیث زید(عليه السلام)٣٠٢خبرمحمد بن عبدالله بن الحسن بن علی(عليه السلام)٣٠٢تفسیر قوله تعالی :﴿َقَالُوا رَبَّنَا بَاعِدْ بَيْنَ أَسْفَارِنَا﴾٣٠٢صفۀ أهل البیت(عليهم السلام)٣٠٣فهرست
