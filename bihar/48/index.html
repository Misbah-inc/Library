٢تتمة كتاب تاريخ علي بن الحسين و محمد بن علی و جعفر بن محمد الصادق و موسی بن جعفر الكاظم عليهم السلام٢أبواب تاريخ الإمام العليم أبی إبراهيم موسى بن جعفر الكاظم الحليم صلوات اللّٰه عليه و علی آبائه الكرام و أولاده الأئمة الأعلام ما تعاقب النور و الظلام٢باب 1 ولادته عليه السلام و تاريخه و جمل أحواله١٠باب 2 أسمائه و ألقابه و كناه و حليته و نقش خاتمه صلوات اللّٰه عليه١٢باب 3 النصوص عليه صلوات اللّٰه عليه٢٩باب 4 معجزاته و استجابة دعواته و معالي أموره و غرائب شأنه صلوات اللّٰه عليه١٠٠باب 5 عبادته و سيره و مكارم أخلاقه و وفور علمه صلوات عليه عليه١٢١باب 6 مناظراته عليه السلام مع خلفاء الجور و ما جرى بينه و بينهم و فيه بعض أحوال علي بن يقطين١٥٩باب 7 أحوال عشائره و أصحابه و أهل زمانه و ما جرى بينه و بينهم و ما جرى من الظلم على عشائره صلوات اللّٰه عليه١٨٩باب 8 احتجاجات هشام بن الحكم في الإمامة و بدو أمره و ما آل إليه أمره إلى وفاته صلوات اللّٰه عليه٢٠٦باب 9 أحواله عليه السلام في الحبس إلى شهادته و تاريخ وفاته و مدفنه صلوات اللّٰه عليه و لعنة اللّٰه علی من ظلمه٢٥٠باب 10 رد مذهب الواقفية و السبب الذي لأجله قيل بالوقف على موسى عليه السلام٢٧٦باب 11 وصاياه و صدقاته صلوات اللّٰه عليه٢٨٣باب 12 أحوال أولاده و أزواجه صلوات اللّٰه عليه٢٩٣شذرات في ما يتعلق بأحوال إخوانه و أولاده عليهم السلام٢٩٣فيما يتعلق بأحوال إخوانه و أخواته عليه الصلاة و السلام.٣٠٣فيما يتعلق بأحوال أولاده عليه الصلاة و السلام.٣١٨نبذة فيما يتعلق ببقعته عليه السلام٣٢٠نبذة فيما يتعلق بالإمام علي بن موسى عليهما السلام.٣٢١خاتمة شريفة في فضيلة بقعة الرضا صلوات اللّٰه عليه.٣٢٩كلمة المحقّق٣٣١كلمة المصحّح٣٣٢فهرس ما في هذا الجزءمن الأبواب٣٣٥رموز الكتاب
