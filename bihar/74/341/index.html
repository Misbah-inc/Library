العلامة محمد باقر المجلسي 74 ٣٤١

بِكُلِّ مُؤْمِنٍ رَحِیمٌ سَخِیٌّ رَضِیٌّ وَلِیٌّ زَكِیٌّ عَلَیْهِ رَحْمَةٌ وَ تَسْلِیمٌ وَ بَرَكَةٌ وَ تَعْظِیمٌ وَ تَكْرِیمٌ مِنْ رَبٍّ غَفُورٍ رَحِیمٍ قَرِیبٍ مُجِیبٍ حَلِیمٍ وَصَّیْتُكُمْ مَعْشَرَ مَنْ حَضَرَ بِوَصِیَّةِ رَبِّكُمْ- وَ ذَكَّرْتُكُمْ سُنَّةَ نَبِیِّكُمْ فَعَلَیْكُمْ بِرَهْبَةٍ تُسْكِنُ قُلُوبَكُمْ وَ خَشْیَةٍ تدری [تُذْرِی] دُمُوعَكُمْ وَ تَقِیَّةٍ تُنْجِیكُمْ قَبْلَ یَوْمِ یُذْهِلُكُمْ وَ یَبْتَلِیكُمْ یَوْمَ یَفُوزُ فِیهِ مَنْ ثَقُلَ وَزْنُ حَسَنَتِهِ وَ خَفَّ وَزْنُ سَیِّئَتِهِ وَ عَلَیْكُمْ بِمَسْأَلَةِ١

ذُلٍّ وَ خُضُوعٍ وَ تَمَلُّقٍ وَ خُشُوعٍ وَ تَوْبَةٍ وَ نُزُوعٍ وَ لْیَغْنَمْ كُلٌ ٢ مِنْكُمْ صِحَّتَهُ قَبْلَ سُقْمِهِ وَ شَیْبَتَهُ قَبْلَ هَرَمِهِ وَ سَعَتَهُ قَبْلَ فَقْرِهِ ٣

وَ فَرْغَتَهُ قَبْلَ

شُغُلِهِ وَ حَضَرَهُ قَبْلَ سَفَرِهِ وَ حَیَاتَهُ قَبْلَ مَوْتِهِ قَبْلَ یَهِنُ وَ یَهْرَمُ وَ یَمْرَضُ وَ یَسْقَمُ وَ یَمِلُّهُ طَبِیبُهُ وَ یُعْرِضُ عَنْهُ حَبِیبُهُ وَ یَنْقَطِعُ عُمُرُهُ وَ یَتَغَیَّرُ عَقْلُهُ ثُمَّ قِیلَ هُوَ مَوْعُوكٌ وَ جِسْمُهُ مَنْهُوكٌ ثُمَّ جَدَّ فِی نَزْعٍ شَدِیدٍ وَ حَضَرَهُ كُلُّ قَرِیبٍ وَ بَعِیدٍ فَشَخَصَ بِبَصَرِهِ وَ طَمَحَ بِنَظَرِهِ وَ رَشَحَ جَبِینُهُ وَ خَطَفَتْ عَرِینُهُ وَ جَدَبَتْ نَفْسُهُ وَ بَكَتْ عِرْسُهُ وَ حَضَرَ رَمْسُهُ وَ یَتَمَ مِنْهُ وَلَدُهُ وَ تَفَرَّقَ عَنْهُ عَدَدُهُ وَ فُصِمَ جَمْعُهُ وَ ذَهَبَ بَصَرُهُ وَ سَمْعُهُ وَ جُرِّدَ وَ غُسِّلَ وَ عُرِیَ وَ نُشِّفَ وَ سُجِّیَ وَ بُسِطَ لَهُ وَ هُیِّئَ وَ نُشِرَ عَلَیْهِ كَفَنُهُ ٤

وَ شُدَّ مِنْهُ ذَقَنُهُ وَ حُمِلَ فَوْقَ سَرِیرٍ وَ صُلِّیَ عَلَیْهِ بِتَكْبِیرٍ بِغَیْرِ سُجُودٍ وَ تَعْفِیرٍ وَ نُقِلَ مِنْ دُورٍ مُزَخْرَفَةٍ وَ قُصُورٍ مُشَیَّدَةٍ وَ فُرُشٍ مُنَجَّدَةٍ٥

فَجُعِلَ فِی ضَرِیحٍ مَلْحُودٍ ضَیِّقٍ مَرْصُودٍ بِلَبِنٍ مَنْضُودٍ مُسَقَّفٍ بِجَلْمُودٍ وَ هِیلَ عَلَیْهِ عَفْرُهُ وَ حُشِیَ مَدَرُهُ وَ تَحَقَّقَ حَذَرُهُ وَ نُسِیَ خَبَرُهُ وَ رَجَعَ عَنْهُ وَلِیُّهُ وَ نَدِیمُهُ وَ نَسِیبُهُ وَ حَمِیمُهُ وَ تَبَدَّلَ بِهِ قَرِینُهُ وَ حَبِیبُهُ فَهُوَ حَشْوُ قَبْرٍ وَ رَهِینُ حَشْرٍ یَدِبُّ فِی جِسْمِهِ دُودُ قَبْرِهِ وَ یَسِیلُ صَدِیدُهُ مِنْ

(١)1- 1. فی بعض نسخ المصدر« و لتكن مسألتكم مسئلة».
(٢)2- 2. زاد فی كف« و ندم و رجوع، و لیغتنم كل مغتنم».
(٣)3- 3. فی كف« عدمه و خلوته قبل فقره».
(٤)4- 4. زاد فی كف« و قمص و عمم و لف و ودع و سلم».
(٥)5- 5. زاد فی كف« و حجر منضدة».
