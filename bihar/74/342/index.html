العلامة محمد باقر المجلسي 74 ٣٤٢

مَنْخِرِهِ وَ تَسْحَقُ تُرْبَتُهُ لَحْمَهُ وَ یُنْشَفُ دَمُهُ وَ یُرَمُّ عَظْمُهُ حَتَّی یَوْمِ حَشْرِهِ فَیَنْشُرُهُ مِنْ قَبْرِهِ وَ یُنْفَخُ فِی الصُّورِ وَ یُدْعَی لِحَشْرٍ وَ نُشُورٍ فَثَمَّ بُعْثِرَتْ قُبُورٌ وَ حُصِّلَتْ سَرِیرَةٌ فِی صُدُورٍ وَ جِی ءَ بِكُلِّ نَبِیٍّ وَ صِدِّیقٍ وَ شَهِیدٍ وَ مِنْطِیقٍ وَ قَعَدَ لِفَصْلِ حُكْمِهِ قَدِیرٌ١ بِعَبْدِهِ خَبِیرٌ بَصِیرٌ فَكَمْ حَسْرَةٍ تُضْنِیهِ ٢ فِی مَوْقِفٍ مَهِیلٍ وَ مَشْهَدٍ جَلِیلٍ بَیْنَ یَدَیْ مَلِكٍ عَظِیمٍ بِكُلِّ صَغِیرَةٍ وَ كَبِیرَةٍ عَلِیمٍ- فَحِینَئِذٍ یُلْجِمُهُ عَرَقُهُ وَ یَخْفِرُهُ قَلَقُهُ فَعَبْرَتُهُ غَیْرُ مَرْحُومَةٍ وَ صَرْخَتُهُ غَیْرُ مَسْمُوعَةٍ٣

وَ بَرَزَتْ صَحِیفَتُهُ وَ تَبَیَّنَتْ جَرِیرَتُهُ فَنَظَرَ فِی سُوءِ عَمَلِهِ ٤

وَ شَهِدَتْ عَیْنُهُ بِنَظَرِهِ وَ یَدُهُ بِبَطْشِهِ وَ رِجْلُهُ بِخَطْوِهِ وَ جِلْدُهُ بِلَمْسِهِ وَ فَرْجُهُ بِمَسِّهِ وَ یُهَدِّدُهُ مُنْكَرٌ وَ نَكِیرٌ وَ كُشِفَ لَهُ حَیْثُ یَصِیرُ فَسُلْسِلَ جِیدُهُ وَ غُلَّتْ یَدُهُ فَسِیقَ یُسْحَبُ وَحْدَهُ فَوَرَدَ جَهَنَّمَ بِكُرْهٍ شَدِیدٍ وَ ظَلَّ یُعَذَّبُ فِی جَحِیمٍ وَ یُسْقَی شَرْبَةً مِنْ حَمِیمٍ تَشْوِی وَجْهَهُ وَ تَسْلَخُ جِلْدَهُ ٥

یَسْتَغِیثُ فَیُعْرِضُ عَنْهُ خَزَنَةُ جَهَنَّمَ وَ یَسْتَصْرِخُ فَیَلْبَثُ حُقْبَهُ بِنَدَمٍ- نَعُوذُ بِرَبٍّ قَدِیرٍ مِنْ شَرِّ كُلِّ مَصِیرٍ وَ نَسْأَلُهُ عَفْوَ مَنْ رَضِیَ عَنْهُ وَ مَغْفِرَةَ مَنْ قَبِلَ مِنْهُ وَ هُوَ وَلِیُّ مَسْأَلَتِی وَ مُنْجِحُ طَلِبَتِی فَمَنْ زُحْزِحَ عَنْ تَعْذِیبِ رَبِّهِ جُعِلَ فِی جَنَّتِهِ بِقُرْبِهِ وَ خُلِّدَ فِی قُصُورٍ٦

وَ نِعَمِهِ وَ مُلِكَ بِحُورٍ عِینٍ وَ حَفَدَةٍ وَ تَقَلَّبَ فِی نَعِیمٍ وَ سُقِیَ مِنْ تَسْنِیمٍ ٧ مَخْتُومٍ بِمِسْكٍ وَ عَنْبَرٍ٨ یَشْرَبُ مِنْ خَمْرٍ مَعْذُوبٍ شُرْبُهُ لَیْسَ یُنْزَفُ لُبُّهُ.

(١)1- 1. فی بعض نسخ المصدر« قعد و تولی لفصل حكمه عند ربّ قدیر».
(٢)2- 2. أی تهزله و تضعفه، و فی بعض نسخ المصدر« فكم زمرة تغنیه».
(٣)3- 3. زاد فی كف« و حجته مقبولة».
(٤)4- 4. زاد فی كف« فنطق كل عضو منه بسوء عمله».
(٥)5- 5. زاد فی كف« یضرب زبینه بمقمع من حدید یعود جلده بعد نضجه بجلد جدید» و الزبینة: الشرطی.
(٦)6- 6. زاد فی كف« و طیف علیه بكؤوس و سكن حضیرة مشیدة و مكن فردوس».
(٧)7- 7. زاد فی كف« و یشرب من عین سلسبیل، ممزوجة بزنجبیل».
(٨)8- 8. زاد فی كف« مستدیم للحبور مستشعر للسرور یشرب من خمور فی روض مشرق مغدق لیس یصدع من شربه». و الحبور: السرور.
