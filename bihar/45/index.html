٢تتمة كتاب تاريخ فاطمة و الحسن و الحسين عليهم السلام٢تتمة أبواب ما يختص بتاريخ الحسين بن علي صلوات اللّٰه عليهما٢بقية الباب 37 سائر ما جرى عليه بعد بيعة الناس ليزيد بن معاوية إلى شهادته صلوات اللّٰه عليه٩٦تذنيب١٠٠باب 38 شهادة ولدى مسلم الصغيرين رضي اللّٰه عنهما١٠٧باب 39 الوقائع المتأخرة عن قتله صلوات اللّٰه عليه إلى رجوع أهل البيت عليهم السلام إلى المدينة و ما ظهر من إعجازه صلوات اللّٰه عليه في تلك الأحوال٢٠١باب 40 ما ظهر بعد شهادته من بكاء السماء و الأرض عليه ص و انكساف الشمس و القمر و غيرها٢٢٠باب 41 ضجيج الملائكة إلى اللّٰه تعالى في أمره و أن اللّٰه بعثهم لنصره و بكائهم و بكاء الأنبياء و فاطمة عليهم السلام عليه صلوات اللّٰه عليه٢٣٠باب 42 رؤية أم سلمة و غيرها رسول اللّٰه صلى اللّٰه عليه وآله في المنام و إخباره بشهادة الكرام٢٣٣باب 43 نوح الجن عليه صلوات اللّٰه عليه٢٤٢باب 44 ما قيل من المراثي فيه صلوات اللّٰه عليه٢٩٥باب 45 العلة التي من أجلها أخر اللّٰه العذاب عن قتلته صلوات اللّٰه عليه و العلة التي من أجلها يقتل أولاد قتلته عليه السلام و إن اللّٰه ينتقم له في زمن القائم عليه السلام٣٠٠باب 46 ما عجل اللّٰه به قتلة الحسين صلوات اللّٰه عليه من العذاب في الدنيا و ما ظهر من إعجازه و استجابة دعائه في ذلك عند الحرب و بعده٣٢٣باب 47 أحوال عشائره و أهل زمانه صلوات اللّٰه عليه و ما جرى بينهم و بين يزيد من الاحتجاج و قد مضى أكثرها في الأبواب السابقة و سيأتي بعضها٣٢٩باب 48 عدد أولاده صلوات اللّٰه عليه و جمل أحوالهم و أحوال أزواجه عليه السلام٣٣٢باب 49 أحوال المختار بن أبي عبيد الثقفي و ما جرى علی يديه و أيدي أوليائه٣٩٠باب 50 جور الخلفاء على قبره الشريف و ما ظهر من المعجزات عند ضريحه و من تربته و زيارته صلوات اللّٰه عليه٤١٠كلمة المصحّح٤١١فهرس ما في هذا الجزء من الأبواب٤١٣رموز الكتاب
