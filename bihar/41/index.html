٢تتمة كتاب تاريخ أمير المؤمنين عليه السلام٢تتمة أبواب كرائم خصاله و محاسن أخلاقه و أفعاله صلوات اللّٰه عليه و علی آله٢باب 99 يقينه صلوات اللّٰه عليه و صبره على المكاره و شدة ابتلائه٨باب 100 تنمره في ذات اللّٰه و تركه المداهنة في دين اللّٰه١١باب 101 عبادته و خوفه عليه السلام٢٤باب 102 سخائه و إنفاقه و إيثاره صلوات اللّٰه عليه و مسابقته فيها علی سائر الصحابة٤٤باب 103 خبر الناقة٤٨باب 104 حسن خلقه و بشره و حلمه و عفوه و إشفاقه و عطفه صلوات اللّٰه عليه٥٤باب 105 تواضعه صلوات اللّٰه عليه٥٩باب 106 مهابته و شجاعته و الاستدلال بسابقته في الجهاد على إمامته و فيه بعض نوادر غزواته١٠٢باب 107 جوامع مكارم أخلاقه و آدابه و سننه و عدله و حسن سياسته صلوات اللّٰه عليه١٦٤باب 108 علة عدم اختضابه عليه السلام١٦٦أبواب معجزاته صلوات اللّٰه و سلامه عليه١٦٦باب 109 رد الشمس له و تكلم الشمس معه عليه السلام١٩١باب 110 استجابة دعواته صلوات اللّٰه عليه في إحياء الموتى و شفاء المرضى و ابتلاء الأعداء بالبلايا و نحو ذلك٢٣٠باب 111 ما ظهر من معجزاته في استنطاق الحيوانات و انقيادها له صلوات اللّٰه عليه٢٤٨باب 112 ما ظهر من معجزاته عليه الصلاة و السلام في الجمادات و النباتات٢٧٤باب 113 قوته و شوكته صلوات اللّٰه عليه في صغره و كبره و تحمله للمشاق و ما يتعلق من الإعجاز ببدنه الشريف٢٨٣باب 114 معجزات كلامه من إخباره بالغائبات و علمه باللغات و بلاغته و فصاحته صلوات اللّٰه عليه٣٦١مراجع التصحيح و التخريج و التعليق٣٦٧كلمة المصحّح٣٦٨فهرس ما في هذا الجزء من الأبواب٣٧١رموز الكتاب
