بِسْمِ ٱللَّهِ ٱلرَّحْمَٰنِ ٱلرَّحِيمِ

٥٨٦

إِذَا ٱلشَّمْسُ كُوِّرَتْ۝١

1آنگاه که خورشید به هم درپیچد،

وَإِذَا ٱلنُّجُومُ ٱنكَدَرَتْ۝٢

2و آنگه که ستارگان همی‌تیره شوند،

وَإِذَا ٱلْجِبَالُ سُيِّرَتْ۝٣

3و آنگاه که کوهها به رفتار آیند.

وَإِذَا ٱلْعِشَارُ عُطِّلَتْ۝٤

4وقتی شتران ماده وانهاده شوند،

وَإِذَا ٱلْوُحُوشُ حُشِرَتْ۝٥

5و آنگه که وحوش را همی‌گرد آرند،

وَإِذَا ٱلْبِحَارُ سُجِّرَتْ۝٦

6دریاها آنگه که جوشان گردند،

وَإِذَا ٱلنُّفُوسُ زُوِّجَتْ۝٧

7و آنگاه که جانها به هم درپیوندند،

وَإِذَا ٱلْمَوْءُۥدَةُ سُئِلَتْ۝٨

8پرسند چو زان دخترک زنده به‌گور:

بِأَىِّ ذَنۢبٍ قُتِلَتْ۝٩

9به کدامین گناه کشته شده است؟

وَإِذَا ٱلصُّحُفُ نُشِرَتْ۝١٠

10و آنگاه که نامه‌ها زهم بگشایند،

وَإِذَا ٱلسَّمَآءُ كُشِطَتْ۝١١

11و آنگاه که آسمان زجا کنده شود،

وَإِذَا ٱلْجَحِيمُ سُعِّرَتْ۝١٢

12و آنگه که جحیم را برافروزانند،

وَإِذَا ٱلْجَنَّةُ أُزْلِفَتْ۝١٣

13و آنگه که بهشت را فرا پیش آرند،

عَلِمَتْ نَفْسٌ مَّآ أَحْضَرَتْ۝١٤

14هر نَفْس بداند چه فراهم دیده.

فَلَآ أُقْسِمُ بِٱلْخُنَّسِ۝١٥

15نه، نه! سوگند به اختران گردان،

ٱلْجَوَارِ ٱلْكُنَّسِ۝١٦

16[کز دیده‌] نهان شوند و از نو آیند،

وَٱلَّيْلِ إِذَا عَسْعَسَ۝١٧

17سوگند به شب چون پشت گرداند،

وَٱلصُّبْحِ إِذَا تَنَفَّسَ۝١٨

18سوگند به صبح چون دمیدن گیرد،

إِنَّهُۥ لَقَوْلُ رَسُولٍ كَرِيمٍ۝١٩

19که [قرآن‌] سخن فرشته بزرگواری است.

ذِى قُوَّةٍ عِندَ ذِى ٱلْعَرْشِ مَكِينٍ۝٢٠

20نیرومند [که‌] پیش خداوند عرش، بلندپایگاه است.

مُّطَاعٍ ثَمَّ أَمِينٍ۝٢١

21در آنجا [هم‌] مطاع [و هم‌] امین است.

وَمَا صَاحِبُكُم بِمَجْنُونٍ۝٢٢

22و رفیق شما مجنون نیست؛

وَلَقَدْ رَءَاهُ بِٱلْأُفُقِ ٱلْمُبِينِ۝٢٣

23و قطعاً آن [فرشته وحی‌] را در افق رخشان دیده.

وَمَا هُوَ عَلَى ٱلْغَيْبِ بِضَنِينٍ۝٢٤

24و او در امر غیب بخیل نیست.

وَمَا هُوَ بِقَوْلِ شَيْطَٰنٍ رَّجِيمٍ۝٢٥

25و [قرآن‌] نیست سخن دیو رجیم.

فَأَيْنَ تَذْهَبُونَ۝٢٦

26پس به کجا می‌روید؟

إِنْ هُوَ إِلَّا ذِكْرٌ لِّلْعَٰلَمِينَ۝٢٧

27این [سخن‌] بجز پندی برای عالمیان نیست؛

لِمَن شَآءَ مِنكُمْ أَن يَسْتَقِيمَ۝٢٨

28برای هر یک از شما که خواهد به راه راست رود.

وَمَا تَشَآءُونَ إِلَّآ أَن يَشَآءَ ٱللَّهُ رَبُّ ٱلْعَٰلَمِينَ۝٢٩

29و تا خدا، پروردگار جهانها، نخواهد، [شما نیز] نخواهید خواست.
